    ```java
    package com.paralife.engine;

    import com.paralife.world.*;
    import com.paralife.world.Entity.*;
    import org.junit.jupiter.api.BeforeEach;
    import org.junit.jupiter.api.Test;

    import java.util.*;
    import java.util.concurrent.ThreadLocalRandom;

    import static org.assertj.core.api.Assertions.assertThat;

    /**
     * Integration test: 300-tick population dynamics with full metabolism system.
     * Validates that per-type metabolic profiles, starvation, fertility, and
     * seasonal cycles produce stable population dynamics.
     */
    class MetabolismIntegrationTest {

        private static final int WIDTH = 64;
        private static final int HEIGHT = 64;
        private static final int TICKS = 300;
        private static final int ENTITIES_PER_TYPE = 20;

        private WorldGrid grid;
        private BotRegistry botRegistry;
        private SimulationEngine engine;

        @BeforeEach
        void setUp() {
            grid = new WorldGrid(new GridConfig(WIDTH, HEIGHT));
            botRegistry = new BotRegistry(); // or mock if needed

            // Create all config records with defaults
            var simConfig = SimulationConfig.defaults();
            var bondConfig = BondingConfig.defaults();
            var compRegistry = new CompositeRegistry();
            var compConfig = CompositeConfig.defaults();
            var metabProfile = MetabolicProfile.defaults();
            var starvConfig = StarvationConfig.defaults();
            var seasonsConfig = SeasonsConfig.defaults();
            var seasonTracker = new SeasonTracker(seasonsConfig);

            engine = new SimulationEngine(grid, simConfig, botRegistry, bondConfig,
                compRegistry, compConfig, metabProfile, starvConfig, seasonTracker);

            // Initialize fertility patches
            var fertilityConfig = FertilityConfig.defaults();
            var fertilityInit = new FertilityInitializer(grid, fertilityConfig);
            fertilityInit.initializeFertility();

            // Seed entities: 20 of each type at random positions
            seedEntities();
        }

        private void seedEntities() {
            ThreadLocalRandom rng = ThreadLocalRandom.current();
            int id = 0;
            for (ParticleType type : ParticleType.values()) {
                var profile = MetabolicProfile.defaults().forType(type);
                for (int i = 0; i < ENTITIES_PER_TYPE; i++) {
                    for (int attempt = 0; attempt < 100; attempt++) {
                        int x = rng.nextInt(WIDTH);
                        int y = rng.nextInt(HEIGHT);
                        if (grid.getCell(x, y).isEmpty()) {
                            String entityId = type.name().toLowerCase() + "-" + (id++);
                            Particle p = Particle.spawn(entityId, type, profile.maxEnergy());
                            grid.setEntity(x, y, p);
                            break;
                        }
                    }
                }
            }
        }

        @Test
        void allTypesServiceOverMetabolismRun() {
            boolean starvationObserved = false;
            Set<Integer> nutrientSpawnCounts = new HashSet<>();

            for (long tick = 1; tick <= TICKS; tick++) {
                engine.processTick(tick);

                // Check for starvation flags
                if (!starvationObserved) {
                    for (int x = 0; x < WIDTH && !starvationObserved; x++) {
                        for (int y = 0; y < HEIGHT && !starvationObserved; y++) {
                            if (grid.getCell(x, y).hasFlag(Cell.FLAG_STARVING)) {
                                starvationObserved = true;
                            }
                        }
                    }
                }
            }

            // Count surviving entities by type
            Map<ParticleType, Integer> counts = new EnumMap<>(ParticleType.class);
            for (ParticleType type : ParticleType.values()) counts.put(type, 0);

            for (int x = 0; x < WIDTH; x++) {
                for (int y = 0; y < HEIGHT; y++) {
                    Entity occ = grid.getCell(x, y).occupant();
                    if (occ instanceof Particle p) {
                        counts.merge(p.type(), 1, Integer::sum);
                    } else if (occ instanceof BondedPair bp) {
                        counts.merge(bp.primaryType(), 1, Integer::sum);
                    }
                }
            }

            // All 3 types must survive
            for (ParticleType type : ParticleType.values()) {
                assertThat(counts.get(type))
                    .as("Population of %s at tick %d", type, TICKS)
                    .isGreaterThan(0);
            }

            // Starvation mechanic must have been active
            assertThat(starvationObserved)
                .as("FLAG_STARVING observed during run")
                .isTrue();
        }

        @Test
        void fertilityPatchesExistAfterInit() {
            int fertileCells = 0;
            for (int x = 0; x < WIDTH; x++) {
                for (int y = 0; y < HEIGHT; y++) {
                    if (grid.getCell(x, y).nutrientLevel() > 0) {
                        fertileCells++;
                    }
                }
            }
            assertThat(fertileCells)
                .as("Number of fertile cells after initialization")
                .isGreaterThan(0);
        }
    }
    ```
