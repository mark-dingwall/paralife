    /**
     * Hybrid vigor formula (D-05): rate trends toward better parent with random bonus.
     * rate = avg(rateA, rateB) + (max(rateA, rateB) - avg(rateA, rateB)) * random(bonusMin, bonusMax)
     */
    private int hybridRate(int rateA, int rateB, ThreadLocalRandom rng) {
        int avg = (rateA + rateB) / 2;
        int max = Math.max(rateA, rateB);
        double bonus = rng.nextDouble(bondingConfig.bondRateBonusMin(), bondingConfig.bondRateBonusMax());
        return avg + (int)((max - avg) * bonus);
    }

    /**
     * Bond decay cost formula (D-06): bonded decay = sum(A, B) * random(costMin, costMax).
     * Bonding reduces total metabolic cost.

    private int bondDecayCost(int decayA, int decayB, ThreadLocalRandom rng) {
        double factor = rng.nextDouble(bondingConfig.bondDecayCostMin(), bondingConfig.bondDecayCostMax());
        return (int)((decayA + decayB) * factor);
    }

    Add a method to SimulationEngine:
    ```java
    /**
     * Compute starvation intensity (D-09). Returns 0.0 if not starving, up to 1.0 at floor.
     * starvationIntensity = (threshold - currentPercent) / (threshold - floorPercent), clamped [0.0, 1.0]
     */
    double computeStarvationIntensity(int energy, int maxEnergy, int thresholdPercent, int floorPercent) {
        double currentPercent = (double) energy / maxEnergy * 100;
        if (currentPercent >= thresholdPercent) return 0.0;
        if (thresholdPercent == floorPercent) return 1.0; // avoid division by zero
        double intensity = (thresholdPercent - currentPercent) / (double)(thresholdPercent - floorPercent);
        return Math.clamp(intensity, 0.0, 1.0);
    }
    ```
