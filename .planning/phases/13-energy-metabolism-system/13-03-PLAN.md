    ```java
    @Component
    public class SeasonTracker {
        public enum Season { SPRING, SUMMER, AUTUMN, WINTER }

        private final SeasonsConfig config;

        public SeasonTracker(SeasonsConfig config) {
            this.config = config;
        }

        /**
         * Seasonal multiplier using COSINE so tick 0 = spring peak (addresses review concern #3).
         *
         * Formula: 1 + amplitude * cos(2 * PI * tick / yearLength)
         *
         * This gives:
         *   tick=0:              cos(0) = 1    -> multiplier = 1.5 (spring peak)
         *   tick=yearLength/4:   cos(PI/2) = 0 -> multiplier = 1.0 (summer equinox)
         *   tick=yearLength/2:   cos(PI) = -1  -> multiplier = 0.5 (autumn trough)
         *   tick=3*yearLength/4: cos(3PI/2)= 0 -> multiplier = 1.0 (winter equinox)
         *
         * Range: [1 - amplitude, 1 + amplitude]
         *
         * Note: CONTEXT.md D-14 specifies sin(), but sin(0)=0 means tick 0 is baseline,
         * not spring peak. Cosine is used instead so tick 0 is peak abundance (spring).
         * Both reviewers flagged this inconsistency. The CONTEXT.md formula is the
         * baseline shape; cosine is the phase-corrected version for intuitive season labels.
         */
        public double getSeasonalMultiplier(long tick) {
            return 1.0 + config.amplitude() * Math.cos(2 * Math.PI * tick / config.yearLengthTicks());
        }

        /**
         * Current season based on tick position in the year cycle (D-15).
         * Year divided into 4 equal quarters.
         * Quarter 0 (tick 0-49):   SPRING — peak abundance, cosine peak
         * Quarter 1 (tick 50-99):  SUMMER — declining toward baseline
         * Quarter 2 (tick 100-149): AUTUMN — trough, cosine minimum
         * Quarter 3 (tick 150-199): WINTER — rising back toward spring
         */
        public Season getSeason(long tick) {
            int yearLength = config.yearLengthTicks();
            int position = (int)(tick % yearLength);
            int quarter = position / (yearLength / 4);
            return Season.values()[Math.min(quarter, 3)]; // clamp for edge case
        }

        public SeasonsConfig getConfig() {
            return config;
        }
    }
    ```

    **Per D-13:** Create `FertilityInitializer.java`:
    ```java
    @Component
    public class FertilityInitializer {
        private final WorldGrid worldGrid;
        private final FertilityConfig config;

        public FertilityInitializer(WorldGrid worldGrid, FertilityConfig config) {
            this.worldGrid = worldGrid;
            this.config = config;
        }

        @PostConstruct
        public void initializeFertility() {
            if (config.patchCount() == 0) return;
            ThreadLocalRandom rng = ThreadLocalRandom.current();
            int width = worldGrid.getWidth();
            int height = worldGrid.getHeight();

            for (int i = 0; i < config.patchCount(); i++) {
                int cx = rng.nextInt(width);
                int cy = rng.nextInt(height);
                int radius = rng.nextInt(config.patchMinRadius(), config.patchMaxRadius() + 1);
                generatePatch(cx, cy, radius, width, height);
            }
        }

        /**
         * Generate a single fertility patch with radial falloff (D-13).
         * Center has max level, edges have proportionally less. Uses max-merge
         * to avoid overwriting higher existing values from overlapping patches.
         * Uses Math.floorMod for toroidal wrapping.
         */
        void generatePatch(int cx, int cy, int radius, int width, int height) {
            for (int dx = -radius; dx <= radius; dx++) {
                for (int dy = -radius; dy <= radius; dy++) {
                    double dist = Math.sqrt(dx * dx + dy * dy);
                    if (dist > radius) continue;
                    double falloff = 1.0 - (dist / radius);
                    int level = (int)(config.maxLevel() * falloff);
                    if (level <= 0) continue;
                    int x = Math.floorMod(cx + dx, width);
                    int y = Math.floorMod(cy + dy, height);
                    Cell cell = worldGrid.getCell(x, y);
                    // Max-merge: keep the higher of existing and new value
                    worldGrid.setCell(x, y, cell.withNutrientLevel(Math.max(cell.nutrientLevel(), level)));
                }
            }
        }
    }
    ```

    private int processNutrientSpawning(int width, int height, long tickNumber) {
        if (config.nutrientSpawnProbability() <= 0) return 0;

        int spawned = 0;
        ThreadLocalRandom rng = ThreadLocalRandom.current();
        double seasonalMultiplier = seasonTracker.getSeasonalMultiplier(tickNumber);

                    // D-12: fertility modulates spawn probability
