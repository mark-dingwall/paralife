void spawnLightning(long tick) {
    Season currentSeason = seasonTracker.getSeason(tick);
    if (currentSeason == Season.WINTER) return;    // D-29
    var cfg = config.lightning();
    double lambda = computeSineLambda(tick, currentSeason, cfg.peakSeason(), cfg.peakLambda(), cfg.offSeasonLambda());
    if (rng.nextDouble() >= lambda) return;

    int w = worldGrid.getWidth(), h = worldGrid.getHeight();
    int cx = rng.nextInt(w);
    int cy = rng.nextInt(h);
    applyLightningAt(cx, cy, cfg);
    log.info("Lightning strike at ({}, {}) tick={}", cx, cy, tick);
}

/** D-22 single-tick instant effect. Package-private for direct testing. */
void applyLightningAt(int cx, int cy, EnvironmentConfig.Lightning cfg) {
    int w = worldGrid.getWidth(), h = worldGrid.getHeight();
    int outer = cfg.outerRadius();
    int inner = cfg.innerRadius();
    int maxLevel = fertilityConfig.maxLevel();

    for (int dx = -outer; dx <= outer; dx++) {
        for (int dy = -outer; dy <= outer; dy++) {
            double dist = Math.sqrt(dx * dx + dy * dy);
            if (dist > outer) continue;
            int x = Math.floorMod(cx + dx, w);
            int y = Math.floorMod(cy + dy, h);
            Cell cell = worldGrid.getCell(x, y);
            if (dist <= inner) {
                // Inner radius: damage any occupant
                if (!cell.isEmpty()) {
                    damageEntityAt(x, y, cell, cfg.damage());
                }
            } else {
                // Outer ring: fertility boost
                int newLevel = Math.min(cell.nutrientLevel() + cfg.fertilityBoost(), maxLevel);
                worldGrid.setCell(x, y, cell.withNutrientLevel(newLevel));
            }
        }
    }
}

private void damageEntityAt(int x, int y, Cell cell, int damage) {
    Entity occupant = cell.occupant();
    switch (occupant) {
        case Particle p ->
                worldGrid.setEntity(x, y, p.withEnergy(Math.max(0, p.energy() - damage)));
        case Entity.BondedPair bp ->
                worldGrid.setEntity(x, y, bp.withEnergy(Math.max(0, bp.energy() - damage)));
        case Entity.CompositeMember cm ->
                worldGrid.setEntity(x, y, cm.withEnergy(Math.max(0, cm.energy() - damage)));
        default -> { /* Rock / Nutrient immune to lightning */ }
    }
}

package com.paralife.engine.environment;

import com.paralife.engine.*;
import com.paralife.world.*;
import com.paralife.world.Entity.Particle;
import com.paralife.world.Entity.ParticleType;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

import java.util.Random;

import static org.assertj.core.api.Assertions.assertThat;

class LightningTest {

    private WorldGrid grid;
    private EnvironmentEngine env;
    private EnvironmentConfig cfg;

    @BeforeEach
    void setup() {
        grid = new WorldGrid(new GridConfig(32, 32));
        cfg = EnvironmentConfig.defaults();
        env = new EnvironmentEngine(grid,
                new SeasonTracker(new SeasonsConfig(200, 0.5)),
                cfg,
                new BuffRegistry(),
                FertilityConfig.defaults(),
                new ToxinPathGenerator(),
                new CompositeRegistry(),
                new Random(42L));
    }

    @Test
    void innerRadiusDamagesOccupants() {
        Particle p = new Particle("p1", ParticleType.CATALYST, 100, 100);
        grid.setEntity(10, 10, p);                // center
        env.applyLightningAtForTest(10, 10);
        Particle after = (Particle) grid.getCell(10, 10).occupant();
        assertThat(after.energy()).isEqualTo(100 - cfg.lightning().damage());
    }

    @Test
    void damageClampsAtZero() {
        Particle p = new Particle("p1", ParticleType.CATALYST, 10, 100);  // low energy
        grid.setEntity(10, 10, p);
        env.applyLightningAtForTest(10, 10);
        Particle after = (Particle) grid.getCell(10, 10).occupant();
        assertThat(after.energy()).isZero();
    }

    @Test
    void outerRingBoostsFertility() {
        int startLevel = grid.getCell(14, 10).nutrientLevel();  // 4 cells east of center -> just inside outer radius
        env.applyLightningAtForTest(10, 10);
        int newLevel = grid.getCell(14, 10).nutrientLevel();
        assertThat(newLevel).isEqualTo(Math.min(startLevel + cfg.lightning().fertilityBoost(),
                FertilityConfig.defaults().maxLevel()));
    }

    @Test
    void fertilityBoostClampsToMaxLevel() {
        int maxLevel = FertilityConfig.defaults().maxLevel();
        grid.setCell(14, 10, grid.getCell(14, 10).withNutrientLevel(maxLevel - 5));
        env.applyLightningAtForTest(10, 10);
        assertThat(grid.getCell(14, 10).nutrientLevel()).isEqualTo(maxLevel);
    }

    @Test
    void noEffectOutsideOuterRadius() {
        int outer = cfg.lightning().outerRadius();
        int farX = 10 + outer + 2;
        int startLevel = grid.getCell(farX, 10).nutrientLevel();
        env.applyLightningAtForTest(10, 10);
        assertThat(grid.getCell(farX, 10).nutrientLevel()).isEqualTo(startLevel);
    }

    @Test
    void innerRadiusCellsDoNotReceiveFertilityBoost() {
        // Inner radius cells are damage-only, not fertility-boosted
        int startLevel = grid.getCell(10, 10).nutrientLevel();  // center
        env.applyLightningAtForTest(10, 10);
        assertThat(grid.getCell(10, 10).nutrientLevel()).isEqualTo(startLevel);
    }

    @Test
    void lightningWorksAcrossToroidalSeam() {
        // Strike near edge — outer radius wraps around torus
        Particle p = new Particle("edge", ParticleType.CATALYST, 100, 100);
        grid.setEntity(1, 1, p);  // within inner radius of center (0,0) via torus wrap (dist sqrt(2))
        env.applyLightningAtForTest(0, 0);
        Particle after = (Particle) grid.getCell(1, 1).occupant();
        assertThat(after.energy()).isLessThan(100);
    }
}
