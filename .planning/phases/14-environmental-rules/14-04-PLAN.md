void spawnLightning(long tick) {
    Season currentSeason = seasonTracker.getSeason(tick);
    if (currentSeason == Season.WINTER) return;
    var cfg = config.lightning();
    double lambda = computeSineLambda(tick, currentSeason, cfg.peakSeason(),
            cfg.peakLambda(), cfg.offSeasonLambda());
    if (rng.nextDouble() >= lambda) return;

    int w = worldGrid.getWidth(), h = worldGrid.getHeight();
    int cx = rng.nextInt(w);
    int cy = rng.nextInt(h);

    // Codex LOW: "attempted-strike" semantics — increment BEFORE apply.
    // If applyLightningAt throws, the counter still records the attempted strike.
    lightningStrikeCount++;

    applyLightningAt(cx, cy, cfg);
    log.info("Lightning strike at ({}, {}) tick={}", cx, cy, tick);
}

/**
 * D-22 single-tick instant effect. Package-private for direct testing.
 * Does NOT call clearEntity — zero-energy occupants are removed by
 * {@link #processEnvDeaths()} via DeathFinalizer (Plan 01 Codex HIGH 1).
 * CompositeMember kills therefore update CompositeRegistry same tick.
 */
void applyLightningAt(int cx, int cy, EnvironmentConfig.Lightning cfg) {
    int w = worldGrid.getWidth(), h = worldGrid.getHeight();
    int outer = cfg.outerRadius();
    int inner = cfg.innerRadius();
    int maxLevel = fertilityConfig.maxLevel();
    boolean anyDamage = false;

    for (int dx = -outer; dx <= outer; dx++) {
        for (int dy = -outer; dy <= outer; dy++) {
            double dist = Math.sqrt(dx * dx + dy * dy);
            if (dist > outer) continue;
            int x = Math.floorMod(cx + dx, w);
            int y = Math.floorMod(cy + dy, h);
            Cell cell = worldGrid.getCell(x, y);
            if (dist <= inner) {
                if (!cell.isEmpty()) {
                    if (damageEntityAt(x, y, cell, cfg.damage())) {
                        anyDamage = true;
                    }
                }
            } else {
                int newLevel = Math.min(cell.nutrientLevel() + cfg.fertilityBoost(), maxLevel);
                worldGrid.setCell(x, y, cell.withNutrientLevel(newLevel));
            }
        }
    }

    if (anyDamage) markEnvDamageApplied();
}

private boolean damageEntityAt(int x, int y, Cell cell, int damage) {
    Entity occupant = cell.occupant();
    return switch (occupant) {
        case Particle p -> {
            worldGrid.setEntity(x, y, p.withEnergy(Math.max(0, p.energy() - damage)));
            yield true;
        }
        case Entity.BondedPair bp -> {
            worldGrid.setEntity(x, y, bp.withEnergy(Math.max(0, bp.energy() - damage)));
            yield true;
        }
        case Entity.CompositeMember cm -> {
            worldGrid.setEntity(x, y, cm.withEnergy(Math.max(0, cm.energy() - damage)));
            yield true;
        }
        default -> false; // Rock / Nutrient immune
    };
}

package com.paralife.engine.environment;

import com.paralife.engine.*;
import com.paralife.world.*;
import com.paralife.world.Entity.*;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

import static org.mockito.ArgumentMatchers.any;
import static org.mockito.ArgumentMatchers.anyString;

class LightningTest {

    private WorldGrid grid;
    private EnvironmentEngine env;
    private EnvironmentConfig cfg;
    private DeathFinalizer finalizer;
    private CompositeRegistry composites;

    @BeforeEach
    void setup() {
        grid = new WorldGrid(new GridConfig(32, 32));
        cfg = EnvironmentConfig.defaults();
        composites = mock(CompositeRegistry.class);
        BotRegistry bots = new BotRegistry();
        BuffRegistry buffs = new BuffRegistry();
        SimulationEngine sim = mock(SimulationEngine.class);
        // Build placeholder env first to wire finalizer then real env
        finalizer = new DeathFinalizer(grid, bots, buffs, composites,
                mock(EnvironmentEngine.class), sim);

                new SeasonTracker(new SeasonsConfig(200, 0.5)),
                cfg, buffs, FertilityConfig.defaults(), finalizer,
                new ToxinPathGenerator(), new Random(42L));

    @Test
    void innerRadiusDamagesParticle() {
        grid.setEntity(10, 10, new Particle("p1", ParticleType.CATALYST, 100, 100));
        env.applyLightningAtForTest(10, 10);
        assertThat(((Particle) grid.getCell(10, 10).occupant()).energy())
                .isEqualTo(100 - cfg.lightning().damage());
    }

    @Test
    void innerRadiusDamagesBondedPair() {
        grid.setEntity(10, 10, new BondedPair("bp1", ParticleType.CATALYST, ParticleType.MEMBRANE,
                100, 100, "a", "b"));
        env.applyLightningAtForTest(10, 10);
        assertThat(((BondedPair) grid.getCell(10, 10).occupant()).energy())
                .isEqualTo(100 - cfg.lightning().damage());
    }

    @Test
    void innerRadiusDamagesCompositeMember() {
        grid.setEntity(10, 10, new CompositeMember("cm1", "comp", ParticleType.CATALYST,
                Role.LOCOMOTOR, 100, 100));
        env.applyLightningAtForTest(10, 10);
        assertThat(((CompositeMember) grid.getCell(10, 10).occupant()).energy())
                .isEqualTo(100 - cfg.lightning().damage());
    }

    @Test
    void damageClampsAtZero() {
        grid.setEntity(10, 10, new Particle("p1", ParticleType.CATALYST, 10, 100));
        env.applyLightningAtForTest(10, 10);
        assertThat(((Particle) grid.getCell(10, 10).occupant()).energy()).isZero();
    }

    @Test
    void innerRadiusLethalParticleKillIsRemovedSameTickViaProcessEnvDeaths() {
        // Plan 01 Codex HIGH 1 integration for lightning
        grid.setEntity(10, 10, new Particle("p-dies", ParticleType.CATALYST, 1, 100));
        env.applyLightningAtForTest(10, 10);
        env.processEnvDeathsForTest();
        assertThat(grid.getCell(10, 10).isEmpty()).isTrue();
    }

    @Test
    void innerRadiusCompositeMemberKillUpdatesCompositeRegistrySameTick() {
        // Codex HIGH 1 inheritance: composite member killed by lightning MUST
        // trigger compositeRegistry.removeMember SAME TICK via DeathFinalizer.
        var state = mock(CompositeRegistry.CompositeState.class);
        when(state.getMemberCount()).thenReturn(2);
        when(composites.getComposite("comp-L")).thenReturn(Optional.of(state));

        grid.setEntity(10, 10, new CompositeMember("cm-dies", "comp-L",
                ParticleType.CATALYST, Role.LOCOMOTOR, 1, 100));

        env.applyLightningAtForTest(10, 10);
        env.processEnvDeathsForTest();

        assertThat(grid.getCell(10, 10).isEmpty()).isTrue();

    @Test

        int startLevel = grid.getCell(14, 10).nutrientLevel();
        env.applyLightningAtForTest(10, 10);
        int newLevel = grid.getCell(14, 10).nutrientLevel();

    }

    @Test

    @Test

        env.applyLightningAtForTest(20, 20);
        assertThat(env.lightningStrikeCount()).isEqualTo(2L);
    }
}
