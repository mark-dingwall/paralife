package com.paralife.engine;

import com.paralife.world.Position;

/**
 * Per-entity infection state (D-14, D-16).
 * `initialTicks` captured at infection time for D-16 buff duration.
 * `position` (REVIEWS HIGH 2) is the entity's grid position, refreshed by the
 * single-pass damage scan in tickBuffsAndInfections — removes the need for a
 * per-infected-entity full-grid lookup.
 */
public record Infection(int ticksLeft, int initialTicks, byte strain, int damagePerTick, Position position) {
    public Infection decrement() {
        return new Infection(ticksLeft - 1, initialTicks, strain, damagePerTick, position);
    }
    public Infection decrementBy(int n) {
        return new Infection(Math.max(0, ticksLeft - n), initialTicks, strain, damagePerTick, position);
    }
    public Infection withPosition(Position newPos) {
        return new Infection(ticksLeft, initialTicks, strain, damagePerTick, newPos);
    }
    public boolean isExpired() {
        return ticksLeft <= 0;
    }
}
```

void advanceMutagen(long tick) {
    var cfg = config.mutagen();
    int w = worldGrid.getWidth(), h = worldGrid.getHeight();

    // Copy src into dst so gossip is additive.
    for (int x = 0; x < w; x++) {
        System.arraycopy(mutagenGrid[x], 0, mutagenGridNext[x], 0, h);
    }

    if (activeMutagen != null) {
        // Gossip spreads new infection; mark reinforced cells as fresh.
        for (int x = 0; x < w; x++) {
            for (int y = 0; y < h; y++) {
                int strain = mutagenGrid[x][y] & 0xFF;
                if (strain == 0) continue;
                mutagenLastReinforcedTick[x][y] = tick;           // self stays fresh
                for (int dx = -1; dx <= 1; dx++) {
                    for (int dy = -1; dy <= 1; dy++) {
                        if (dx == 0 && dy == 0) continue;
                        if (rng.nextDouble() >= cfg.gossipProbability()) continue;
                        int nx = Math.floorMod(x + dx, w);
                        int ny = Math.floorMod(y + dy, h);
                        if ((mutagenGridNext[nx][ny] & 0xFF) != 0) continue;
                        int newStrain = strain;
                        if (rng.nextDouble() < cfg.strainMutationChance()) {
                            newStrain = Math.max(1, Math.min(255, strain + (rng.nextBoolean() ? 1 : -1)));
                        }
                        mutagenGridNext[nx][ny] = (byte) newStrain;
                        mutagenLastReinforcedTick[nx][ny] = tick;    // newly reinforced
                    }
                }
            }
        }
        if (activeMutagen.isExpired(tick)) activeMutagen = null;
    } else {
        // Blocker 3: event expired — age strain cells; clear those past zoneDecayTicks
        long decayLimit = cfg.zoneDecayTicks();
        for (int x = 0; x < w; x++) {
            for (int y = 0; y < h; y++) {
                int strain = mutagenGridNext[x][y] & 0xFF;
                if (strain == 0) continue;
                if (tick - mutagenLastReinforcedTick[x][y] > decayLimit) {
                    mutagenGridNext[x][y] = 0;
                }
            }
        }
    }

    byte[][] tmp = mutagenGrid;
    mutagenGrid = mutagenGridNext;
    mutagenGridNext = tmp;
}

void resolveMutagenCollisions(long tick) {
    var cfg = config.mutagen();
    int w = worldGrid.getWidth(), h = worldGrid.getHeight();
    for (int x = 0; x < w; x++) {
        for (int y = 0; y < h; y++) {
            int strain = mutagenGrid[x][y] & 0xFF;
            if (strain == 0) continue;
            Cell cell = worldGrid.getCell(x, y);
            if (cell.isEmpty()) continue;
            String entityId = entityIdOf(cell.occupant());
            if (entityId == null) continue;
            if (infections.containsKey(entityId)) continue;
            Long immuneUntil = cureImmuneUntil.get(entityId);
            if (immuneUntil != null && immuneUntil >= tick) continue;
            int duration = cfg.infectionDurationMin()
                    + rng.nextInt(cfg.infectionDurationMax() - cfg.infectionDurationMin() + 1);
            infections.put(entityId, new Infection(
                    duration, duration, (byte) strain, cfg.damagePerTick(), new Position(x, y)));
        }
    }
}

private String entityIdOf(Entity e) {
    return switch (e) {
        case Entity.Particle p -> p.id();
        case Entity.BondedPair bp -> bp.id();
        case Entity.CompositeMember cm -> cm.id();
        default -> null;
    };
}

void tickBuffsAndInfections(long tick) {
    var cfg = config.mutagen();
    if (!infections.isEmpty()) {
        // SINGLE per-tick grid pass — build entityId -> (Position, occupant) index for all
        // currently-infected entities. Avoids the N * W * H full-grid-per-infected-entity scan
        // flagged in REVIEWS HIGH 2.
        Map<String, int[]> infectedPositions = new HashMap<>();
        Map<String, Entity> infectedOccupants = new HashMap<>();
        int w = worldGrid.getWidth(), h = worldGrid.getHeight();
        for (int x = 0; x < w; x++) {
            for (int y = 0; y < h; y++) {
                Cell c = worldGrid.getCell(x, y);
                if (c.isEmpty()) continue;
                String id = entityIdOf(c.occupant());
                if (id == null) continue;
                if (!infections.containsKey(id)) continue;
                infectedPositions.put(id, new int[]{x, y});
                infectedOccupants.put(id, c.occupant());
            }
        }

        // Damage + decrement
        Map<String, Infection> cured = new LinkedHashMap<>();
        for (var entry : new HashMap<>(infections).entrySet()) {
            String id = entry.getKey();
            Infection inf = entry.getValue();
            int[] pos = infectedPositions.get(id);
            Entity occ = infectedOccupants.get(id);
            if (pos != null && occ != null) {
                // Apply damage in place; refresh Position on Infection
                int damage = inf.damagePerTick();
                int x = pos[0], y = pos[1];
                switch (occ) {
                    case Entity.Particle p ->
                            worldGrid.setEntity(x, y, p.withEnergy(Math.max(0, p.energy() - damage)));
                    case Entity.BondedPair bp ->
                            worldGrid.setEntity(x, y, bp.withEnergy(Math.max(0, bp.energy() - damage)));
                    case Entity.CompositeMember cm ->
                            worldGrid.setEntity(x, y, cm.withEnergy(Math.max(0, cm.energy() - damage)));
                    default -> { /* ignored */ }
                }
                inf = inf.withPosition(new Position(x, y));
            }
            Infection next = inf.decrement();
            if (next.isExpired()) {
                cured.put(id, inf);   // pre-decrement record retains initialTicks
            } else {
                infections.put(id, next);
            }
        }

        // Grant buffs + cure-immunity
        for (var e : cured.entrySet()) {
            String id = e.getKey();
            Infection expired = e.getValue();
            infections.remove(id);
            Entity occ = infectedOccupants.get(id);
            grantSurvivorBuffs(id, tick, cfg, expired.initialTicks(), occ);
            cureImmuneUntil.put(id, tick + cfg.cureTicks());
        }
    }

    cureImmuneUntil.entrySet().removeIf(ent -> ent.getValue() < tick);
    buffRegistry.expireBuffs(tick);
}
```

private void grantSurvivorBuffs(String entityId, long tick,

                                Entity occupant) {
