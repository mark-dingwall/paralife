```java
/** Per-entity infection state (D-14). Keyed by entity id in EnvironmentEngine. */
public record Infection(int ticksLeft, byte strain, int damagePerTick) {
    public Infection decrement() {
        return new Infection(ticksLeft - 1, strain, damagePerTick);
    }
    public Infection decrementBy(int n) {
        return new Infection(Math.max(0, ticksLeft - n), strain, damagePerTick);
    }
    public boolean isExpired() {
        return ticksLeft <= 0;
    }
}
```

```java
void spawnMutagen(long tick) {
    if (activeMutagen != null) return;
    Season currentSeason = seasonTracker.getSeason(tick);
    if (currentSeason == Season.WINTER) return;
    var cfg = config.mutagen();
    double lambda = computeSineLambda(tick, currentSeason, cfg.peakSeason(), cfg.peakLambda(), cfg.offSeasonLambda());
    if (rng.nextDouble() < lambda) {
        int x = rng.nextInt(worldGrid.getWidth());
        int y = rng.nextInt(worldGrid.getHeight());
        mutagenGrid[x][y] = (byte) (1 + rng.nextInt(255));  // strain 1-255 (0 = clean)
        activeMutagen = new MutagenEvent(tick, new Position(x, y),
                cfg.infectionDurationMax() * 10L);  // outbreak lasts ~10× max infection duration
        log.info("Mutagen outbreak spawned at {} (strain {}) tick={}",
                new Position(x, y), mutagenGrid[x][y] & 0xFF, tick);
    }
}
```

```java
void advanceMutagen(long tick) {
    if (activeMutagen == null) {
        // gossip still runs so existing infection tails fade out — but if no active, let existing zones linger naturally
    }
    var cfg = config.mutagen();
    int w = worldGrid.getWidth(), h = worldGrid.getHeight();
    // Copy src into dst (gossip is ADDITIVE per cell — start from current state)
    for (int x = 0; x < w; x++) {
        System.arraycopy(mutagenGrid[x], 0, mutagenGridNext[x], 0, h);
    }
    for (int x = 0; x < w; x++) {
        for (int y = 0; y < h; y++) {
            int strain = mutagenGrid[x][y] & 0xFF;
            if (strain == 0) continue;
            for (int dx = -1; dx <= 1; dx++) {
                for (int dy = -1; dy <= 1; dy++) {
                    if (dx == 0 && dy == 0) continue;
                    if (rng.nextDouble() >= cfg.gossipProbability()) continue;
                    int nx = Math.floorMod(x + dx, w);
                    int ny = Math.floorMod(y + dy, h);
                    if ((mutagenGridNext[nx][ny] & 0xFF) != 0) continue;  // already infected
                    int newStrain = strain;
                    if (rng.nextDouble() < cfg.strainMutationChance()) {
                        newStrain = Math.max(1, Math.min(255, strain + (rng.nextBoolean() ? 1 : -1)));
                    }
                    mutagenGridNext[nx][ny] = (byte) newStrain;
                }
            }
        }
    }
    byte[][] tmp = mutagenGrid;
    mutagenGrid = mutagenGridNext;
    mutagenGridNext = tmp;

    if (activeMutagen != null && activeMutagen.isExpired(tick)) {
        activeMutagen = null;
    }
}
```

```java
void resolveMutagenCollisions(long tick) {
    var cfg = config.mutagen();
    int w = worldGrid.getWidth(), h = worldGrid.getHeight();
    for (int x = 0; x < w; x++) {
        for (int y = 0; y < h; y++) {
            int strain = mutagenGrid[x][y] & 0xFF;
            if (strain == 0) continue;
            Cell cell = worldGrid.getCell(x, y);
            if (cell.isEmpty()) continue;
            String entityId = entityIdOf(cell.occupant());
            if (entityId == null) continue;
            if (infections.containsKey(entityId)) continue;              // already infected
            Long immuneUntil = cureImmuneUntil.get(entityId);
            if (immuneUntil != null && immuneUntil >= tick) continue;     // cure-immune
            int duration = cfg.infectionDurationMin()
                    + rng.nextInt(cfg.infectionDurationMax() - cfg.infectionDurationMin() + 1);
            infections.put(entityId, new Infection(duration, (byte) strain, cfg.damagePerTick()));
        }
    }
}

private String entityIdOf(Entity e) {
    return switch (e) {
        case Particle p -> p.id();
        case Entity.BondedPair bp -> bp.id();
        case Entity.CompositeMember cm -> cm.id();
        default -> null;
    };
}
```

```java
void tickBuffsAndInfections(long tick) {
    var cfg = config.mutagen();
    // (a) damage + decrement
    List<String> cured = new ArrayList<>();
    for (var entry : infections.entrySet()) {
        String id = entry.getKey();
        Infection inf = entry.getValue();
        // Apply DPS to entity if still alive
        applyInfectionDamage(id, inf.damagePerTick());
        Infection next = inf.decrement();
        if (next.isExpired()) {
            cured.add(id);
        } else {
            infections.put(id, next);
        }
    }
    for (String id : cured) {
        infections.remove(id);
        grantSurvivorBuffs(id, tick, cfg);
        cureImmuneUntil.put(id, tick + cfg.cureTicks());
    }
    // (b) decrement cure-immunity (implicit via expiry check in resolveMutagenCollisions)
    // Clean up expired immunities
    cureImmuneUntil.entrySet().removeIf(e -> e.getValue() < tick);
    // (c) expire buffs
    buffRegistry.expireBuffs(tick);
}

private void applyInfectionDamage(String entityId, int damage) {
    // Scan grid for this entity id. O(W*H) per infection — acceptable for MVP grid sizes.
    // Optimisation: CompositeRegistry/BotRegistry have position lookups; use if available.
    // For now, iterate via grid once per infection map (small N).
    int w = worldGrid.getWidth(), h = worldGrid.getHeight();
    outer:
    for (int x = 0; x < w; x++) {
        for (int y = 0; y < h; y++) {
            Cell cell = worldGrid.getCell(x, y);
            if (cell.isEmpty()) continue;
            if (!entityId.equals(entityIdOf(cell.occupant()))) continue;
            if (cell.occupant() instanceof Particle p) {
                worldGrid.setEntity(x, y, p.withEnergy(p.energy() - damage));
            }
            break outer;
        }
    }
}
```

```java
public void reduceInfection(String entityId, int ticks, long currentTick) {
```
