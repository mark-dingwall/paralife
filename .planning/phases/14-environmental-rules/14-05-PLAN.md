static CellView cellToView(Cell cell, byte cellStatus, byte entityStatus) {
    int flags = cell.flags();
    if (cell.isEmpty()) {
        return new CellView(null, null, cell.nutrientLevel(), flags, cellStatus, entityStatus);
    }
    Entity occupant = cell.occupant();
    return switch (occupant) {
        case Particle p -> new CellView(p.type().name(), p.id(), cell.nutrientLevel(), flags, cellStatus, entityStatus);
        case Entity.Rock r -> new CellView("ROCK", r.id(), cell.nutrientLevel(), flags, cellStatus, entityStatus);
        case Entity.Nutrient n -> new CellView("NUTRIENT", n.id(), cell.nutrientLevel(), flags, cellStatus, entityStatus);
        case Entity.BondedPair bp -> new CellView(
                "BONDED_" + bp.primaryType() + "_" + bp.secondaryType(),
                bp.id(), cell.nutrientLevel(), flags, cellStatus, entityStatus);
        case Entity.CompositeMember cm -> new CellView(
                "COMPOSITE_" + cm.type() + "_" + cm.role(),
                cm.id(), cell.nutrientLevel(), flags, cellStatus, entityStatus);
    };
}

private byte computeVisionScopedOvercrowded(Position cellPos, Position botPos, int radius) {
    int count = 0;
    for (Position n : worldGrid.getNeighbors(cellPos.x(), cellPos.y())) {
        // Only count neighbors within bot's visible range
        int ndx = Math.floorMod(n.x() - botPos.x() + worldGrid.getWidth() / 2, worldGrid.getWidth()) - worldGrid.getWidth() / 2;
        int ndy = Math.floorMod(n.y() - botPos.y() + worldGrid.getHeight() / 2, worldGrid.getHeight()) - worldGrid.getHeight() / 2;
        if (Math.abs(ndx) > radius || Math.abs(ndy) > radius) continue;
        if (worldGrid.getCell(n.x(), n.y()).hasOccupant()) count++;
    }
    // Use same threshold as SimulationEngine.processOvercrowding — confirm via its constant.
    // If it is 5 (common for Moore-8), hardcode. If different, expose via SimulationConfig.
    if (count >= 5) return 0b0001;
    return 0;
}

}

/**
 * Walk `range` cells in `dir` from `from`, falling back to `range-1` on blockage (FN-9).
 * Returns null if no clear target within the range.
 */
Position findTargetAtRange(Position from, Direction dir, int range,
                            Set<Position> claimedCells, int w, int h) {
    int minCandidate = range > 1 ? range - 1 : 1;
    for (int candidate = range; candidate >= minCandidate; candidate--) {
        Position t = from;
        for (int step = 0; step < candidate; step++) {
            t = dir.apply(t, w, h);
        }
        if (claimedCells.contains(t)) continue;
        if (worldGrid.getCell(t.x(), t.y()).hasOccupant()) continue;
        return t;
    }
    return null;
}
