```java
/**
 * D-40: count occupied neighbors of `cellPos` that also lie within the bot's visible square.
 * Toroidal-wrap aware. Returns 0b0001 if count >= threshold, else 0.
 * Package-private static so VisionScopedOvercrowdingTest can invoke directly.
 */
static byte computeVisionScopedOvercrowded(WorldGrid grid, Position cellPos, Position botPos,
                                           int radius, int threshold) {
    int count = 0;
    int w = grid.getWidth(), h = grid.getHeight();
    for (int dx = -1; dx <= 1; dx++) {
        for (int dy = -1; dy <= 1; dy++) {
            if (dx == 0 && dy == 0) continue;
            int nx = Math.floorMod(cellPos.x() + dx, w);
            int ny = Math.floorMod(cellPos.y() + dy, h);
            int bdx = toroidalDelta(nx, botPos.x(), w);
            int bdy = toroidalDelta(ny, botPos.y(), h);
            if (Math.abs(bdx) > radius || Math.abs(bdy) > radius) continue;
            if (grid.getCell(nx, ny).hasOccupant()) count++;
        }
    }
    return count >= threshold ? (byte) 0b0001 : (byte) 0;
}

static int toroidalDelta(int to, int from, int size) {
    return Math.floorMod(to - from + size / 2, size) - size / 2;
}
```
