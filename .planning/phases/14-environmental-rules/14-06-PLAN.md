```java
package com.paralife.engine.environment;

import com.paralife.engine.*;
import com.paralife.world.*;
import com.paralife.world.Entity.*;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.context.ApplicationEventPublisher;
import org.springframework.test.context.ActiveProfiles;
import org.springframework.test.context.TestPropertySource;

import java.util.ArrayList;
import java.util.List;

import static org.assertj.core.api.Assertions.assertThat;

/**
 * Codex HIGH 6 — engine-only deterministic harness. Runs a 300-tick seeded
 * simulation TWICE within the same JVM, resetting all stateful collaborators
 * between runs via explicit reset hooks. Asserts every observable is equal
 * across runs.
 *
 * <p>This test does NOT attempt to exercise WebSocket delivery or real
 * BotClient loops — that's the job of {@link EnvironmentFullStackSmokeTest}.
 * Instead it publishes TickEvents directly, which is sufficient for
 * engine-level determinism validation.
 */
@SpringBootTest
@ActiveProfiles("test")
@TestPropertySource(properties = {
        "paralife.simulation.events.enabled=true",
        "paralife.simulation.events.seed=42",
        "paralife.simulation.seasons.year-length-ticks=100",
        "paralife.simulation.events.lightning.peak-lambda=0.06",
        "paralife.tick.auto-start=false"
})
class EnvironmentDeterminismTest {

    @Autowired WorldGrid worldGrid;
    @Autowired EnvironmentEngine environmentEngine;
    @Autowired EnvironmentConfig environmentConfig;
    @Autowired BuffRegistry buffRegistry;
    @Autowired CompositeRegistry compositeRegistry;
    @Autowired BotRegistry botRegistry;
    @Autowired DeathFinalizer deathFinalizer;
    @Autowired ApplicationEventPublisher publisher;

    private static final class RunObservables {
        int initialPopulation;
        int finalPopulation;
        int toxinEvents;
        int mutagenEvents;
        long lightningStrikes;
        long deathEvents;
        boolean everBuffed;
        boolean nonZeroStatusSeen;
    }

    @Test
    void observablesFireDuringSingleRun() {
        resetAll();
        RunObservables obs = driveRun();
        assertThat(obs.toxinEvents).isGreaterThanOrEqualTo(1);
        assertThat(obs.mutagenEvents).isGreaterThanOrEqualTo(1);
        assertThat(obs.lightningStrikes).isGreaterThanOrEqualTo(1L);
        assertThat(obs.deathEvents)
                .as("Codex HIGH 6: explicit death counter replaces disappeared-position heuristic")
                .isGreaterThanOrEqualTo(1L);
        assertThat(obs.everBuffed).isTrue();
        assertThat(obs.nonZeroStatusSeen).isTrue();
        assertThat(obs.finalPopulation)
                .isBetween((int) (obs.initialPopulation * 0.2),
                          (int) (obs.initialPopulation * 1.2));
    }

    @Test
    void runsAreDeterministicAcrossTwoInvocations() {
        resetAll();
        RunObservables a = driveRun();

        resetAll();   // Codex HIGH 6: explicit reset of env + composite + bot + buff + death counter
        RunObservables b = driveRun();

        assertThat(b.finalPopulation).as("population deterministic").isEqualTo(a.finalPopulation);
        assertThat(b.toxinEvents).as("toxinEvents deterministic").isEqualTo(a.toxinEvents);
        assertThat(b.mutagenEvents).as("mutagenEvents deterministic").isEqualTo(a.mutagenEvents);
        assertThat(b.lightningStrikes).as("lightningStrikes deterministic").isEqualTo(a.lightningStrikes);
        assertThat(b.deathEvents).as("deathEvents deterministic").isEqualTo(a.deathEvents);
        assertThat(b.everBuffed).as("everBuffed deterministic").isEqualTo(a.everBuffed);
    }

    /** Codex HIGH 6: explicit reset of EVERY stateful collaborator. */
    private void resetAll() {
        int w = worldGrid.getWidth(), h = worldGrid.getHeight();
        for (int x = 0; x < w; x++) {
            for (int y = 0; y < h; y++) worldGrid.clearEntity(x, y);
        }
        environmentEngine.resetForTest();   // env shadow grids + active events + infection maps + rng + counters
        buffRegistry.clear();
        compositeRegistry.clear();
        botRegistry.clear();
        deathFinalizer.resetCountForTest();
    }

    private RunObservables driveRun() {
        RunObservables obs = new RunObservables();
        seedInitialPopulation(60);
        obs.initialPopulation = countPopulation();
        long lightningBaseline = environmentEngine.lightningStrikeCount();
        long deathBaseline = deathFinalizer.getDeathEventCount();

        boolean toxinPrev = false, mutagenPrev = false;
        for (long tick = 1; tick <= 300; tick++) {
            publisher.publishEvent(new TickEvent(tick));

            boolean toxinActive = environmentEngine.activeToxinEvent() != null;
            boolean mutagenActive = environmentEngine.activeMutagenEvent() != null;
            if (toxinActive && !toxinPrev) obs.toxinEvents++;
            if (mutagenActive && !mutagenPrev) obs.mutagenEvents++;
            toxinPrev = toxinActive;
            mutagenPrev = mutagenActive;

            if (!obs.everBuffed && anyEntityBuffed()) obs.everBuffed = true;
            if (!obs.nonZeroStatusSeen && anyNonZeroStatus()) obs.nonZeroStatusSeen = true;
        }

        obs.finalPopulation = countPopulation();
        obs.lightningStrikes = environmentEngine.lightningStrikeCount() - lightningBaseline;
        obs.deathEvents = deathFinalizer.getDeathEventCount() - deathBaseline;
        return obs;
    }

    private void seedInitialPopulation(int count) {
        java.util.Random fixed = new java.util.Random(7L);
        int w = worldGrid.getWidth(), h = worldGrid.getHeight();
        ParticleType[] types = ParticleType.values();
        int placed = 0, tries = 0;
        while (placed < count && tries < count * 10) {
            int x = fixed.nextInt(w);
            int y = fixed.nextInt(h);
            tries++;
            if (!worldGrid.getCell(x, y).isEmpty()) continue;
            ParticleType t = types[placed % types.length];
            worldGrid.setEntity(x, y, new Particle("p-" + placed, t, 80, 100));
            placed++;
        }
    }

    private int countPopulation() {
        int n = 0;
        for (int x = 0; x < worldGrid.getWidth(); x++) {
            for (int y = 0; y < worldGrid.getHeight(); y++) {
                Entity occ = worldGrid.getCell(x, y).occupant();
                if (occ instanceof Particle || occ instanceof BondedPair || occ instanceof CompositeMember) n++;
            }
        }
        return n;
    }

    private boolean anyEntityBuffed() {
        for (int x = 0; x < worldGrid.getWidth(); x++) {
            for (int y = 0; y < worldGrid.getHeight(); y++) {
                Cell c = worldGrid.getCell(x, y);
                if (c.isEmpty()) continue;
                String id = switch (c.occupant()) {
                    case Particle p -> p.id();
                    case BondedPair bp -> bp.id();
                    case CompositeMember cm -> cm.id();
                    default -> null;
                };
                if (id != null && !buffRegistry.getBuffs(id).isEmpty()) return true;
            }
        }
        return false;
    }

    private boolean anyNonZeroStatus() {
        for (int x = 0; x < worldGrid.getWidth(); x++) {
            for (int y = 0; y < worldGrid.getHeight(); y++) {
                Position p = new Position(x, y);
                if (environmentEngine.getCellStatus(p) != 0) return true;
                Cell c = worldGrid.getCell(x, y);
                if (c.isEmpty()) continue;
                String id = switch (c.occupant()) {
                    case Particle pa -> pa.id();
                    case BondedPair bp -> bp.id();
                    case CompositeMember cm -> cm.id();
                    default -> null;
                };
                if (id != null && environmentEngine.getEntityStatus(id) != 0) return true;
            }
        }
        return false;
    }
}
```

```java
@ActiveProfiles("test")
@TestPropertySource(properties = {
        "paralife.simulation.events.enabled=true",

class EnvironmentFullStackSmokeTest {

    @LocalServerPort int port;

    @Autowired WorldGrid worldGrid;
    @Autowired EnvironmentEngine environmentEngine;
    @Autowired org.springframework.context.ApplicationEventPublisher publisher;

    @Test
    void realWebSocketDeliversPerceptionWithNonZeroStatusByte() throws Exception {
        // Seed a dense population so some entity ends up in a toxic cell or infected.
        seedInitialPopulation(80);

        // Connect a real BotClient.
        String url = "ws://localhost:" + port + "/ws/world";
        AtomicBoolean nonZeroStatusSeen = new AtomicBoolean(false);
        AtomicInteger perceptionCount = new AtomicInteger(0);

        // Instrumented BotClient — if the production BotClient does not expose a
        // per-Perception callback, use a test-only subclass or a package-private hook.
        // The minimum test contract: when a Messages.Perception arrives, inspect the
        // neighbourhood for any non-zero cellStatus / entityStatus byte.
        try (var client = new com.paralife.bot.TestInstrumentedBotClient(url, perception -> {
            perceptionCount.incrementAndGet();
            for (var row : perception.neighbourhood()) {
                for (var view : row) {
                    if (view.cellStatus() != 0 || view.entityStatus() != 0) {
                        nonZeroStatusSeen.set(true);
                        return;
                    }
                }
            }
        })) {
            client.connect();

            // Drive ~60 ticks to give env effects time to fire and bots time to receive.
            for (long tick = 1; tick <= 60; tick++) {
                publisher.publishEvent(new TickEvent(tick));
                Thread.sleep(20);     // allow async WebSocket delivery
                if (nonZeroStatusSeen.get()) break;
            }
        }

        assertThat(perceptionCount.get())
                .as("real BotClient received at least one Perception frame (Codex HIGH 6)")
                .isGreaterThanOrEqualTo(1);
        assertThat(nonZeroStatusSeen.get())
                .as("at least one CellView in a received frame had a non-zero status byte "
                        + "(Codex HIGH 6: actual wire-path validation, not direct cache probe)")
                .isTrue();
    }

    private void seedInitialPopulation(int count) {
        java.util.Random fixed = new java.util.Random(7L);
        int w = worldGrid.getWidth(), h = worldGrid.getHeight();
        ParticleType[] types = ParticleType.values();
        int placed = 0, tries = 0;
        while (placed < count && tries < count * 10) {
            int x = fixed.nextInt(w);
            int y = fixed.nextInt(h);
            tries++;
            if (!worldGrid.getCell(x, y).isEmpty()) continue;
            ParticleType t = types[placed % types.length];
            worldGrid.setEntity(x, y, new Particle("p-" + placed, t, 80, 100));
            placed++;
        }
    }
}
```
