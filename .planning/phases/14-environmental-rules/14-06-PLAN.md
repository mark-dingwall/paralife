package com.paralife.engine.environment;

import com.paralife.engine.BuffRegistry;
import com.paralife.engine.EnvironmentConfig;
import com.paralife.engine.EnvironmentEngine;
import com.paralife.engine.TickEvent;
import com.paralife.world.Cell;
import com.paralife.world.Entity;
import com.paralife.world.Entity.BondedPair;
import com.paralife.world.Entity.CompositeMember;
import com.paralife.world.Entity.Particle;
import com.paralife.world.Entity.ParticleType;
import com.paralife.world.Position;
import com.paralife.world.WorldGrid;
import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.context.ApplicationEventPublisher;
import org.springframework.test.annotation.DirtiesContext;
import org.springframework.test.context.ActiveProfiles;
import org.springframework.test.context.TestPropertySource;

import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

import static org.assertj.core.api.Assertions.assertThat;

/**
 * Phase 14 integration: 300-tick seeded full-stack validation with direct observables.
 *
 * <p>Env overrides live on this class via @TestPropertySource (REVIEWS HIGH 1) — NOT in
 * a global application-test.yml. This prevents env-enabled behavior from bleeding into
 * unrelated integration tests.
 */
@SpringBootTest
@ActiveProfiles("test")
@TestPropertySource(properties = {
        "paralife.simulation.events.enabled=true",
        "paralife.simulation.events.seed=42",
        "paralife.simulation.seasons.year-length-ticks=100",
        "paralife.simulation.events.lightning.peak-lambda=0.06",
        "paralife.tick.auto-start=false"
})
class EnvironmentIntegrationTest {

    @Autowired WorldGrid worldGrid;
    @Autowired EnvironmentEngine environmentEngine;
    @Autowired BuffRegistry buffRegistry;
    @Autowired EnvironmentConfig environmentConfig;
    @Autowired ApplicationEventPublisher publisher;

    /** Aggregate observables captured during a 300-tick run. */
    private static final class RunObservables {
        int initialPopulation;
        int finalPopulation;
        int toxinEvents;
        int mutagenEvents;
        long lightningStrikes;
        boolean everBuffed;
        boolean nonZeroStatusSeen;
        List<Position> deathPositions = new ArrayList<>();
    }

    @Test
    void fullPhase14PipelineOver300Ticks() {
        RunObservables obs = driveRun();

        // 1. Toxin fires at least once
        assertThat(obs.toxinEvents)
                .as("rising-edge count on activeToxinEvent")
                .isGreaterThanOrEqualTo(1);

        // 2. Mutagen fires at least once
        assertThat(obs.mutagenEvents)
                .as("rising-edge count on activeMutagenEvent")
                .isGreaterThanOrEqualTo(1);

        // 3. Lightning fires at least once — DIRECT counter (no proxy)
        assertThat(obs.lightningStrikes)
                .as("direct lightningStrikeCount delta")
                .isGreaterThanOrEqualTo(1L);

        // 4. Some entity was buffed at some tick (REVIEWS HIGH 5: latch across ticks)
        assertThat(obs.everBuffed)
                .as("everBuffed latched true during the run")
                .isTrue();

        // 5. Localised compost — at least one recorded death position has nutrientLevel >= fullStrength
        int fullStrength = environmentConfig.compost().fullStrength();
        boolean anyDeathCellComposted = obs.deathPositions.stream()
                .anyMatch(p -> worldGrid.getCell(p.x(), p.y()).nutrientLevel() >= fullStrength);
        assertThat(anyDeathCellComposted)
                .as("at least one recorded death position has composted nutrientLevel >= compost.fullStrength "
                        + "(localised observable — NOT a total-nutrient-sum proxy)")
                .isTrue();

        // 6. Status byte was observed non-zero at some tick
        assertThat(obs.nonZeroStatusSeen)
                .as("nonZeroStatusSeen latched true during the run (perception status coverage)")
                .isTrue();

        // 7. Population within [20%, 120%] — counts Particle + BondedPair + CompositeMember
        assertThat(obs.finalPopulation)
                .as("population within stable range; includes Particle + BondedPair + CompositeMember")
                .isBetween((int) (obs.initialPopulation * 0.2), (int) (obs.initialPopulation * 1.2));
    }

    /**
     * Deterministic double-run: run the harness twice with the same seed, assert key
     * observables match. Uses @DirtiesContext to force a fresh Spring context before
     * the second invocation so EnvironmentEngine Random is re-seeded.
     */
    @Test
    @DirtiesContext(methodMode = DirtiesContext.MethodMode.BEFORE_METHOD)
    void deterministicAcrossRunsWithSameSeed() {
        RunObservables a = driveRun();

        // Reset mutable state for a second run within the same Spring context.
        // If Spring context caching makes this insufficient, this test SHOULD be marked
        // @Disabled("TODO: cross-context determinism — Phase 16") and the blocker documented.
        resetState();
        RunObservables b = driveRun();

        assertThat(b.finalPopulation).as("population deterministic").isEqualTo(a.finalPopulation);
        assertThat(b.toxinEvents).as("toxinEvents deterministic").isEqualTo(a.toxinEvents);
        assertThat(b.mutagenEvents).as("mutagenEvents deterministic").isEqualTo(a.mutagenEvents);
        assertThat(b.lightningStrikes).as("lightningStrikes deterministic").isEqualTo(a.lightningStrikes);
        assertThat(b.everBuffed).as("everBuffed deterministic").isEqualTo(a.everBuffed);
    }

    private RunObservables driveRun() {
        RunObservables obs = new RunObservables();
        seedInitialPopulation(60);
        obs.initialPopulation = countPopulation();
        long lightningBaseline = environmentEngine.lightningStrikeCount();

        boolean toxinActivePrev = false, mutagenActivePrev = false;
        Set<Position> prevOccupied = snapshotOccupiedPositions();

        for (long tick = 1; tick <= 300; tick++) {
            publisher.publishEvent(new TickEvent(tick));

            // Rising-edge event counting
            boolean toxinActive = environmentEngine.activeToxinEvent() != null;
            boolean mutagenActive = environmentEngine.activeMutagenEvent() != null;
            if (toxinActive && !toxinActivePrev) obs.toxinEvents++;
            if (mutagenActive && !mutagenActivePrev) obs.mutagenEvents++;
            toxinActivePrev = toxinActive;
            mutagenActivePrev = mutagenActive;

            // Death-site capture (localised compost observable)
            Set<Position> currentOccupied = snapshotOccupiedPositions();
            Set<Position> disappeared = new HashSet<>(prevOccupied);
            disappeared.removeAll(currentOccupied);
            obs.deathPositions.addAll(disappeared);
            prevOccupied = currentOccupied;

            // Latch everBuffed
            if (!obs.everBuffed && anyEntityBuffed()) {
                obs.everBuffed = true;
            }

            // Latch nonZeroStatusSeen
            if (!obs.nonZeroStatusSeen && anyNonZeroStatus()) {
                obs.nonZeroStatusSeen = true;
            }
        }

        obs.finalPopulation = countPopulation();
        obs.lightningStrikes = environmentEngine.lightningStrikeCount() - lightningBaseline;
        return obs;
    }

    private void resetState() {
        // Clear grid + env + buff state for second deterministic run.
        int w = worldGrid.getWidth(), h = worldGrid.getHeight();
        for (int x = 0; x < w; x++) for (int y = 0; y < h; y++) worldGrid.clearEntity(x, y);
        buffRegistry.clear();
        // EnvironmentEngine does not expose a general reset; its Random is re-seeded on
        // @DirtiesContext re-creation, which is this test's path. If shared-state bleed remains,
        // document in SUMMARY and mark @Disabled per the plan's fallback.
    }

    private void seedInitialPopulation(int count) {
        java.util.Random fixed = new java.util.Random(7L);
        int w = worldGrid.getWidth(), h = worldGrid.getHeight();
        ParticleType[] types = ParticleType.values();
        int placed = 0, tries = 0;
        while (placed < count && tries < count * 10) {
            int x = fixed.nextInt(w);
            int y = fixed.nextInt(h);
            tries++;
            if (!worldGrid.getCell(x, y).isEmpty()) continue;
            ParticleType t = types[placed % types.length];
            worldGrid.setEntity(x, y, new Particle("p-" + placed, t, 80, 100));
            placed++;
        }
    }

    /** REVIEWS MEDIUM: counts Particle + BondedPair + CompositeMember. */
    private int countPopulation() {
        int n = 0;
        for (int x = 0; x < worldGrid.getWidth(); x++) {
            for (int y = 0; y < worldGrid.getHeight(); y++) {
                Entity occ = worldGrid.getCell(x, y).occupant();
                if (occ instanceof Particle || occ instanceof BondedPair || occ instanceof CompositeMember) n++;
            }
        }
        return n;
    }

    private Set<Position> snapshotOccupiedPositions() {
        Set<Position> out = new HashSet<>();
        for (int x = 0; x < worldGrid.getWidth(); x++) {
            for (int y = 0; y < worldGrid.getHeight(); y++) {
                if (!worldGrid.getCell(x, y).isEmpty()) out.add(new Position(x, y));
            }
        }
        return out;
    }

    private boolean anyEntityBuffed() {
        for (int x = 0; x < worldGrid.getWidth(); x++) {
            for (int y = 0; y < worldGrid.getHeight(); y++) {
                Cell c = worldGrid.getCell(x, y);
                if (c.isEmpty()) continue;
                String id = switch (c.occupant()) {
                    case Particle p -> p.id();
                    case BondedPair bp -> bp.id();
                    case CompositeMember cm -> cm.id();
                    default -> null;
                };
                if (id != null && !buffRegistry.getBuffs(id).isEmpty()) return true;
            }
        }
        return false;
    }

    private boolean anyNonZeroStatus() {
        for (int x = 0; x < worldGrid.getWidth(); x++) {
            for (int y = 0; y < worldGrid.getHeight(); y++) {
                Position p = new Position(x, y);
                if (environmentEngine.getCellStatus(p) != 0) return true;
                Cell c = worldGrid.getCell(x, y);
                if (c.isEmpty()) continue;
                String id = switch (c.occupant()) {
                    case Particle pa -> pa.id();
                    case BondedPair bp -> bp.id();
                    case CompositeMember cm -> cm.id();
                    default -> null;
                };
                if (id != null && environmentEngine.getEntityStatus(id) != 0) return true;
            }
        }
        return false;
    }
}
