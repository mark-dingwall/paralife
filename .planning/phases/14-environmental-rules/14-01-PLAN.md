 * Phase 14 environmental events config (D-48). All rates, radii, durations bound
 * under `paralife.simulation.events.*`. Nested records follow MetabolicProfile
 * pattern; validation in compact constructors throws IllegalArgumentException
 * matching SeasonsConfig/FertilityConfig style.
 *
 * <p>Season fields use `SeasonTracker.Season` enum — Spring Boot 3 binds
 * YAML strings (e.g. "SUMMER") to enum values automatically.
 */
@ConfigurationProperties(prefix = "paralife.simulation.events")
public record EnvironmentConfig(
        boolean enabled,
        Long seed,               // null = non-deterministic Random; set in test profile
        Lightning lightning,
        Toxin toxin,
        Mutagen mutagen,
        Compost compost
) {

    public record Lightning(
            Season peakSeason,
            double peakLambda,
            double offSeasonLambda,
            int innerRadius,
            int outerRadius,
            int damage,
            int fertilityBoost
    ) {
        public Lightning {
            if (peakSeason == null) throw new IllegalArgumentException("lightning.peakSeason required");
            if (peakLambda < 0.0 || peakLambda > 1.0) throw new IllegalArgumentException("lightning.peakLambda must be in [0,1]: " + peakLambda);
            if (offSeasonLambda < 0.0 || offSeasonLambda > 1.0) throw new IllegalArgumentException("lightning.offSeasonLambda must be in [0,1]: " + offSeasonLambda);
            if (peakLambda < offSeasonLambda) throw new IllegalArgumentException("lightning.peakLambda < offSeasonLambda: " + peakLambda + " < " + offSeasonLambda);
            if (innerRadius < 0) throw new IllegalArgumentException("lightning.innerRadius negative: " + innerRadius);
            if (outerRadius <= innerRadius) throw new IllegalArgumentException("lightning.outerRadius must be > innerRadius: " + outerRadius + " <= " + innerRadius);
            if (damage < 0) throw new IllegalArgumentException("lightning.damage negative: " + damage);
            if (fertilityBoost < 0) throw new IllegalArgumentException("lightning.fertilityBoost negative: " + fertilityBoost);
        }
    }

    public record Toxin(
            Season peakSeason,
            double peakLambda,
            double offSeasonLambda,
            int pathPointsMin,
            int pathPointsMax,
            int pathOffsetMin,
            int pathOffsetMax,
            int speed,

            int diffusionRadius,
            double decayRate,

            public Resistance {
                if (catalyst < 0) throw new IllegalArgumentException("toxin.resistance.catalyst negative: " + catalyst);
                if (membrane < 0) throw new IllegalArgumentException("toxin.resistance.membrane negative: " + membrane);
                if (spore < 0) throw new IllegalArgumentException("toxin.resistance.spore negative: " + spore);
            }
            /** Per-type multiplier (damage = baseDamage * intensity * multiplier). */
            public double multiplierFor(Entity.ParticleType type) {
                return switch (type) {
                    case CATALYST -> catalyst;
                    case MEMBRANE -> membrane;
                    case SPORE -> spore;
                };
            }
        }
        public Toxin {
            if (peakSeason == null) throw new IllegalArgumentException("toxin.peakSeason required");
            if (peakLambda < offSeasonLambda) throw new IllegalArgumentException("toxin.peakLambda < offSeasonLambda");
            if (pathPointsMin < 2) throw new IllegalArgumentException("toxin.pathPointsMin must be >= 2: " + pathPointsMin);
            if (pathPointsMax < pathPointsMin) throw new IllegalArgumentException("toxin.pathPointsMax < pathPointsMin");
            if (pathOffsetMin < 0) throw new IllegalArgumentException("toxin.pathOffsetMin negative");
            if (pathOffsetMax < pathOffsetMin) throw new IllegalArgumentException("toxin.pathOffsetMax < pathOffsetMin");
            if (speed < 1) throw new IllegalArgumentException("toxin.speed must be >= 1: " + speed);
            if (lifetimeTicks < 1) throw new IllegalArgumentException("toxin.lifetimeTicks must be >= 1: " + lifetimeTicks);
            if (diffusionRadius < 1) throw new IllegalArgumentException("toxin.diffusionRadius must be >= 1");
            if (decayRate < 0.0 || decayRate > 1.0) throw new IllegalArgumentException("toxin.decayRate must be in [0,1]: " + decayRate);
            if (splashDamageFraction < 0.0) throw new IllegalArgumentException("toxin.splashDamageFraction negative");

            if (resistance == null) throw new IllegalArgumentException("toxin.resistance required");
        }
    }

            int cureTicks,

            int damagePerTick,
            double strainMutationChance,

            if (infectionDurationMin < 1) throw new IllegalArgumentException("mutagen.infectionDurationMin must be >= 1");
            if (infectionDurationMax < infectionDurationMin) throw new IllegalArgumentException("mutagen.infectionDurationMax < min");
            if (buffDurationMultiplier < 1) throw new IllegalArgumentException("mutagen.buffDurationMultiplier must be >= 1");
            if (cureTicks < 0) throw new IllegalArgumentException("mutagen.cureTicks negative");
            if (attackCureReduction < 0) throw new IllegalArgumentException("mutagen.attackCureReduction negative");

package com.paralife.engine;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;

import java.util.List;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.CopyOnWriteArrayList;

/**
 * Shadow registry for mutagen survivor buffs (D-19, D-47). Keyed by entity ID.
 * Mirrors BotRegistry / CompositeRegistry pattern. CopyOnWriteArrayList makes
 * per-entity buff lists safe to iterate from perception/action threads while
 * the tick thread mutates.
 */
@Component
public class BuffRegistry {

    private static final Logger log = LoggerFactory.getLogger(BuffRegistry.class);

    public enum BuffType {
        ATTACK_PLUS_1,
        MOVEMENT_PLUS_1,
        SENSOR_PLUS_1,
        UPKEEP_MINUS_1
    }

    /** Immutable buff record (D-19). */
    public record ActiveBuff(BuffType type, long expiryTick) {}

    private final ConcurrentHashMap<String, CopyOnWriteArrayList<ActiveBuff>> byEntity =
            new ConcurrentHashMap<>();

    public void grant(String entityId, BuffType type, long expiryTick) {
        byEntity.compute(entityId, (k, existing) -> {
            var list = existing == null ? new CopyOnWriteArrayList<ActiveBuff>() : existing;
            list.add(new ActiveBuff(type, expiryTick));
            return list;
        });
        log.debug("Buff granted: entity={} type={} expiresAt={}", entityId, type, expiryTick);
    }

    public List<ActiveBuff> getBuffs(String entityId) {
        var list = byEntity.get(entityId);
        return list == null ? List.of() : list;
    }

    public boolean hasBuff(String entityId, BuffType type) {
        var list = byEntity.get(entityId);
        if (list == null) return false;
        for (ActiveBuff b : list) {
            if (b.type() == type) return true;
        }
        return false;
    }

    /**
     * Remove all buffs whose expiryTick <= currentTick. Called each tick from
     * EnvironmentEngine.
     */
    public void expireBuffs(long currentTick) {
        byEntity.forEach((id, list) -> list.removeIf(b -> b.expiryTick() <= currentTick));
        byEntity.entrySet().removeIf(e -> e.getValue().isEmpty());
    }

    /**
     * Called from SimulationEngine.processDeaths on every entity clear (Pitfall 5).
     */
    public void unregisterEntity(String entityId) {
        byEntity.remove(entityId);
    }

    /** Test-only: reset all state. */
    public void clear() {
        byEntity.clear();
    }
}
