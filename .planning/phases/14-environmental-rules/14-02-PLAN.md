## Revision Notes (reviews cycle 3)

This revision addresses cycle-3 Codex feedback:

- **Codex MEDIUM — BondedPair toxin resistance rule explicit:** resistance is `max(multFor(primaryType), multFor(secondaryType))`. Stated inline in `resolveToxinCollisions` and enforced by a dedicated test in ToxinTest. Rationale documented: toxin is an environmental hazard; the pair takes the worst-case (least-resistant) member's multiplier so that a MEMBRANE+SPORE pair is damaged as SPORE (high multiplier) rather than averaged. Avoids order-sensitive balance from arbitrarily picking primary or secondary.
- **Codex LOW — non-zero toxin cell counter:** `nonZeroToxinCellCount` is maintained inside `advanceToxin` by capturing the return value of `CellularAutomaton.diffuseStep` (which now returns an int count of non-zero destination cells — free-of-charge, since the step already iterates every cell). Idle-tick fast path is O(1): `activeToxin == null && nonZeroToxinCellCount == 0` ⇒ return. The prior `toxinGridIsAllZero()` O(area) scan is removed.
- **Codex LOW — consolidate entityIdOf:** a new `EntityIds` utility class hosts `public static String entityIdOf(Entity)`. `EnvironmentEngine`, `PerceptionBroadcaster` (Plan 05), and `EnvDeathSweepTest` all route through it. Prevents drift when new entity types are added.

Everything the prior review endorsed — normalised damage (intensity/255.0), SplashDelta deferred-delta routing, exhaustive attack-family enumeration, markEnvDamageApplied integration, TOXIN_PRESENT vs TOXIC threshold split — is preserved unchanged.

```java
package com.paralife.engine;

/**
 * Double-buffered CA diffusion helper. Configurable Moore-N radius, decay,
 * and below-threshold clear.
 *
 * <p>Codex LOW cycle 3: {@link #diffuseStep} returns the count of non-zero
 * destination cells so callers (EnvironmentEngine's toxin advance) can
 * maintain an O(1) fast-path guard instead of an O(W*H) scan every idle tick.
 */
public final class CellularAutomaton {
    private CellularAutomaton() {}

    /**
     * @return count of cells in {@code dst} with a post-step value != 0.
     */
    public static int diffuseStep(byte[][] src, byte[][] dst,
                                   int w, int h,
                                   double diffusionRate,
                                   double decayRate,
                                   int threshold,
                                   int radius) {
        int nonZero = 0;
        for (int x = 0; x < w; x++) {
            for (int y = 0; y < h; y++) {
                // Moore-N kernel of given radius
                double sum = 0.0;
                int samples = 0;
                for (int dx = -radius; dx <= radius; dx++) {
                    for (int dy = -radius; dy <= radius; dy++) {
                        int nx = Math.floorMod(x + dx, w);
                        int ny = Math.floorMod(y + dy, h);
                        sum += (src[nx][ny] & 0xFF);
                        samples++;
                    }
                }
                double avg = sum / samples;
                double self = src[x][y] & 0xFF;
                double mixed = self * (1.0 - diffusionRate) + avg * diffusionRate;
                double decayed = mixed * (1.0 - decayRate);
                int value = (int) Math.round(decayed);
                if (value < 0) value = 0;
                if (value > 255) value = 255;
                if (value <= threshold) value = 0;
                dst[x][y] = (byte) value;
                if (value != 0) nonZero++;
            }
        }
        return nonZero;
    }
}
```

```java
    // Codex LOW cycle 3: O(1) fast-path; no O(W*H) scan on idle ticks.
    if (activeToxin == null && nonZeroToxinCellCount == 0) return;

    if (activeToxin != null) {
        int speed = config.toxin().speed();
        int newHeadIdx = Math.min(activeToxin.headIdx() + speed, activeToxin.prePath().size() - 1);
        for (int i = activeToxin.headIdx(); i <= newHeadIdx; i++) {
            Position p = activeToxin.prePath().get(i);
            // Maintain the counter on stamp: count only fresh transitions 0 -> non-zero.
            if ((toxinGrid[p.x()][p.y()] & 0xFF) == 0) nonZeroToxinCellCount++;
            toxinGrid[p.x()][p.y()] = (byte) 255;
        }
        activeToxin = activeToxin.withHeadIdx(newHeadIdx);
        if (activeToxin.isExpired(tick)) activeToxin = null;
    }

    // Codex LOW cycle 3: CA step returns the count of non-zero destination cells.
    nonZeroToxinCellCount = CellularAutomaton.diffuseStep(
            toxinGrid, toxinGridNext,

            config.toxin().decayRate(),
            config.toxin().intensityThreshold(),
            config.toxin().diffusionRadius());
    byte[][] tmp = toxinGrid;
    toxinGrid = toxinGridNext;
    toxinGridNext = tmp;
}
```

```java
void resolveToxinCollisions(long tick) {
    var cfg = config.toxin();
    int w = worldGrid.getWidth(), h = worldGrid.getHeight();
    boolean anyDamage = false;
    for (int x = 0; x < w; x++) {
        for (int y = 0; y < h; y++) {
            int intensity = toxinGrid[x][y] & 0xFF;
            if (intensity == 0) continue;
            Cell cell = worldGrid.getCell(x, y);
            if (cell.isEmpty()) continue;
            Entity occ = cell.occupant();
            double fraction = intensity / 255.0;
            switch (occ) {
                case Particle p -> {
                    double mult = cfg.resistance().multiplierFor(p.type());
                    int damage = Math.max(1, (int) Math.round(cfg.baseDamage() * fraction * mult));
                    worldGrid.setEntity(x, y, p.withEnergy(Math.max(0, p.energy() - damage)));
                    anyDamage = true;
                }
                case Entity.BondedPair bp -> {
                    // Codex MEDIUM cycle 3: BondedPair resistance is the MAX of the two member-type multipliers.
                    // Rationale: toxin is an environmental hazard; the pair takes the worst-case multiplier
                    // (least-resistant member drives damage). Avoids order-sensitive balance from picking
                    // primary or secondary arbitrarily.
                    double mult = Math.max(
                            cfg.resistance().multiplierFor(bp.primaryType()),
                            cfg.resistance().multiplierFor(bp.secondaryType()));
                    int damage = Math.max(1, (int) Math.round(cfg.baseDamage() * fraction * mult));
                    worldGrid.setEntity(x, y, bp.withEnergy(Math.max(0, bp.energy() - damage)));
                    anyDamage = true;
                }
                case Entity.CompositeMember cm -> {
                    double mult = cfg.resistance().multiplierFor(cm.type());
                    int damage = Math.max(1, (int) Math.round(cfg.baseDamage() * fraction * mult));
                    worldGrid.setEntity(x, y, cm.withEnergy(Math.max(0, cm.energy() - damage)));
                    anyDamage = true;
                }
                default -> { /* Rock/Nutrient immune */ }
            }
        }
    }
    if (anyDamage) markEnvDamageApplied();
}
```
