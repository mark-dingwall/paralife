```java
void spawnToxin(long tick) {
    if (activeToxin != null) return;
    Season currentSeason = seasonTracker.getSeason(tick);
    if (currentSeason == Season.WINTER) return;
    var cfg = config.toxin();
    double lambda = computeSineLambda(tick, currentSeason, cfg.peakSeason(), cfg.peakLambda(), cfg.offSeasonLambda());
    if (rng.nextDouble() < lambda) {
        long seed = rng.nextLong();
        var path = toxinPathGenerator.generatePath(
                worldGrid.getWidth(), worldGrid.getHeight(), cfg, new Random(seed));
        activeToxin = new ToxinEvent(tick, cfg.lifetimeTicks(), path, 0, seed);
        log.info("Toxin event spawned: tick={} seed={} pathLen={}", tick, seed, path.size());
    }
}

double computeSineLambda(long tick, Season currentSeason, Season peakSeason,
                         double peakLambda, double offLambda) {
    if (currentSeason != peakSeason) return offLambda;
    int yearLength = seasonTracker.getConfig().yearLengthTicks();
    int seasonLength = yearLength / 4;
    long tickInSeason = computeTickInSeason(tick, yearLength);
    double x = 2.0 * Math.PI * tickInSeason / seasonLength;
    return 0.5 * (Math.sin(x - Math.PI / 2.0) + 1.0) * (peakLambda - offLambda) + offLambda;
}

private long computeTickInSeason(long tick, int yearLength) {
    long position = Math.floorMod(tick, (long) yearLength);
    long shifted = position + yearLength / 8L;
    int quarter = (int) Math.floorMod(shifted / (yearLength / 4L), 4L);
    long seasonStart = (long) quarter * (yearLength / 4L) - yearLength / 8L;
    return Math.floorMod(position - seasonStart, (long) yearLength);
}
```

```java
private void buildStatusCaches(long tick) {
    int w = worldGrid.getWidth(), h = worldGrid.getHeight();

    int cellVisibilityThreshold = config.toxin().intensityThreshold();
    for (int x = 0; x < w; x++) {
        for (int y = 0; y < h; y++) {
            int intensity = toxinGrid[x][y] & 0xFF;
            Position pos = new Position(x, y);

                cellStatusCache.merge(pos, (byte) 0b0010,
                        (a, b) -> (byte) ((a & 0xFF) | (b & 0xFF)));
```
