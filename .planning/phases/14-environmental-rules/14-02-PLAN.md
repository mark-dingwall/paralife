```java
package com.paralife.engine;

import com.paralife.world.Position;

import java.util.List;

/**
 * Active toxin event (D-06). `prePath` is the arc-length-sampled spline
 * in wrapped grid coordinates; `headIdx` advances by config.speed cells per tick.
 * `seed` is stored for future visualiser replay (Claude's Discretion in CONTEXT.md).
 */
public record ToxinEvent(
        long spawnTick,
        long lifetimeTicks,
        List<Position> prePath,
        int headIdx,
        long seed
) {
    public ToxinEvent withHeadIdx(int newIdx) {
        return new ToxinEvent(spawnTick, lifetimeTicks, prePath, newIdx, seed);
    }

    public boolean hasReachedEnd() {
        return headIdx >= prePath.size() - 1;
    }

    public boolean isExpired(long currentTick) {
        return currentTick - spawnTick >= lifetimeTicks;
    }

    public Position headPosition() {
        return prePath.get(Math.min(headIdx, prePath.size() - 1));
    }
}
```

```java
package com.paralife.engine;

import com.paralife.world.Position;
import org.springframework.stereotype.Component;

import java.util.ArrayList;
import java.util.List;
import java.util.Random;

/**
 * Pure-math helpers for toxin path generation (D-05). Waypoints generated in
 * un-wrapped coords (Pitfall 8); caller wraps at final Position construction.
 * Catmull-Rom evaluation matches p5.js curvePoint (microbes.js reference).
 */
@Component
public class ToxinPathGenerator {

    /**
     * Uniform (tau=0.5) Catmull-Rom. Matches p5.js curvePoint(p0, p1, p2, p3, t).
     * Interpolates from p1 to p2 as t goes 0 -> 1; p0 and p3 shape tangents.
     */
    static double catmullRom(double p0, double p1, double p2, double p3, double t) {
        double t2 = t * t;
        double t3 = t2 * t;
        return 0.5 * (
                (2.0 * p1)
                + (-p0 + p2) * t
                + (2.0 * p0 - 5.0 * p1 + 4.0 * p2 - p3) * t2
                + (-p0 + 3.0 * p1 - 3.0 * p2 + p3) * t3
        );
    }

    /**
     * Generate N waypoints on un-wrapped coordinates: endpoints on two distinct
     * edges, interior points evenly spaced along A->B vector with random perpendicular offset.
     * Returns List<double[2]>. N drawn uniformly in [pointsMin, pointsMax].
     */
    List<double[]> generateWaypoints(int w, int h,
                                      int pointsMin, int pointsMax,
                                      int offsetMin, int offsetMax,
                                      Random rng) {
        int n = pointsMin + rng.nextInt(pointsMax - pointsMin + 1);
        double[] a = randomEdgePoint(w, h, -1, rng);
        double[] b = randomEdgePoint(w, h, edgeOf(a, w, h), rng);
        double dx = b[0] - a[0];
        double dy = b[1] - a[1];
        double length = Math.hypot(dx, dy);
        double nxPerp = -dy / length;
        double nyPerp = dx / length;

        List<double[]> waypoints = new ArrayList<>(n);
        waypoints.add(a);
        for (int i = 1; i < n - 1; i++) {
            double t = i / (double) (n - 1);
            double midX = a[0] + t * dx;
            double midY = a[1] + t * dy;
            int offsetRange = offsetMax - offsetMin + 1;
            double mag = offsetMin + rng.nextInt(offsetRange);
            double sign = rng.nextBoolean() ? 1.0 : -1.0;
            waypoints.add(new double[]{
                    midX + sign * mag * nxPerp,
                    midY + sign * mag * nyPerp
            });
        }
        waypoints.add(b);
        return waypoints;
    }

    /**
     * Produce a densely-sampled wrapped path following the Catmull-Rom spline
     * through generated waypoints. Sampling stride targets ~1 cell per step
     * (arc-length approximation) so head-advance of N cells per tick is meaningful (D-06).
     */
    public List<Position> generatePath(int w, int h, EnvironmentConfig.Toxin cfg, Random rng) {
        List<double[]> wps = generateWaypoints(
                w, h, cfg.pathPointsMin(), cfg.pathPointsMax(),
                cfg.pathOffsetMin(), cfg.pathOffsetMax(), rng);
        int n = wps.size();
        // Arc-length approximation: sum straight-line distances between waypoints, then
        // target ~1 sample per cell. Good enough — the pre-path is consumed integer-wise.
        double arcLen = 0;
        for (int i = 0; i < n - 1; i++) {
            arcLen += Math.hypot(wps.get(i + 1)[0] - wps.get(i)[0],
                                  wps.get(i + 1)[1] - wps.get(i)[1]);
        }
        int samples = Math.max(16, (int) Math.ceil(arcLen));
        List<Position> path = new ArrayList<>(samples);
        for (int s = 0; s <= samples; s++) {
            double overallT = s / (double) samples;
            double scaled = overallT * (n - 1);
            int p1Idx = Math.min((int) scaled, n - 2);
            int p2Idx = p1Idx + 1;
            int c1Idx = Math.max(p1Idx - 1, 0);
            int c2Idx = Math.min(p2Idx + 1, n - 1);
            double segT = scaled - p1Idx;
            double x = catmullRom(wps.get(c1Idx)[0], wps.get(p1Idx)[0], wps.get(p2Idx)[0], wps.get(c2Idx)[0], segT);
            double y = catmullRom(wps.get(c1Idx)[1], wps.get(p1Idx)[1], wps.get(p2Idx)[1], wps.get(c2Idx)[1], segT);
            // Wrap ONLY at final Position construction (Pitfall 8)
            path.add(new Position(Math.floorMod((int) Math.round(x), w),
                                   Math.floorMod((int) Math.round(y), h)));
        }
        // Collapse consecutive duplicates so head-advance always progresses ≥ 1 cell
        List<Position> collapsed = new ArrayList<>(path.size());
        Position prev = null;
        for (Position p : path) {
            if (!p.equals(prev)) collapsed.add(p);
            prev = p;
        }
        return collapsed;
    }

    private double[] randomEdgePoint(int w, int h, int excludeEdge, Random rng) {
        int edge;
        do { edge = rng.nextInt(4); } while (edge == excludeEdge);
        return switch (edge) {
            case 0 -> new double[]{rng.nextInt(w), 0};          // top
            case 1 -> new double[]{rng.nextInt(w), h - 1};      // bottom
            case 2 -> new double[]{0, rng.nextInt(h)};          // left
            default -> new double[]{w - 1, rng.nextInt(h)};     // right
        };
    }

    private int edgeOf(double[] p, int w, int h) {
        if (p[1] == 0) return 0;
        if (p[1] == h - 1) return 1;
        if (p[0] == 0) return 2;
        return 3;
    }
}
```

```java
package com.paralife.engine;

/**
 * Double-buffered cellular automaton helper (RESEARCH Pattern 3).
 * Used for toxin diffusion (Plan 02) and mutagen strain gossip (Plan 03).
 * Every byte read is masked with & 0xFF to handle unsigned intensity > 127 (Pitfall 1).
 *
 * <p>Caller owns src + dst buffers; swap references after each step to avoid per-tick allocation.
 */
public final class CellularAutomaton {

    private CellularAutomaton() {}

    /**
     * One Moore-8 diffusion step with decay and threshold clear.
     *
     * @param src            read buffer (unsigned bytes, 0-255)
     * @param dst            write buffer (same size as src)
     * @param w              grid width
     * @param h              grid height
     * @param diffusionRate  [0,1] — fraction of neighbor-average mixed into self
     * @param decayRate      [0,1] — multiplicative decay applied after mixing
     * @param threshold      cells below threshold (after decay) are cleared to 0
     */
    public static void diffuseStep(byte[][] src, byte[][] dst,
                                    int w, int h,
                                    double diffusionRate, double decayRate,
                                    int threshold) {
        for (int x = 0; x < w; x++) {
            for (int y = 0; y < h; y++) {
                int self = src[x][y] & 0xFF;
                int neighborSum = 0;
                for (int dx = -1; dx <= 1; dx++) {
                    for (int dy = -1; dy <= 1; dy++) {
                        if (dx == 0 && dy == 0) continue;
                        int nx = Math.floorMod(x + dx, w);
                        int ny = Math.floorMod(y + dy, h);
                        neighborSum += src[nx][ny] & 0xFF;
                    }
                }
                double neighborAvg = neighborSum / 8.0;
                double mixed = (1.0 - diffusionRate) * self + diffusionRate * neighborAvg;
                int after = (int) Math.round(mixed * (1.0 - decayRate));
                if (after < threshold) after = 0;
                if (after > 255) after = 255;
                if (after < 0) after = 0;
                dst[x][y] = (byte) after;
            }
        }
    }

    /** Convenience: unsigned-byte read. */
    public static int readUnsigned(byte[][] grid, int x, int y) {
        return grid[x][y] & 0xFF;
    }
}
```

```java
/** D-27 sine-scaled lambda. Winter handled by caller. */
double computeSineLambda(long tick, Season currentSeason, Season peakSeason, double peakLambda, double offLambda) {
    if (currentSeason != peakSeason) return offLambda;
    int yearLength = seasonTracker.getConfig().yearLengthTicks();
    int seasonLength = yearLength / 4;
    long tickInSeason = computeTickInSeason(tick, yearLength);
    double x = 2.0 * Math.PI * tickInSeason / seasonLength;
    return 0.5 * (Math.sin(x - Math.PI / 2.0) + 1.0) * (peakLambda - offLambda) + offLambda;
}

private long computeTickInSeason(long tick, int yearLength) {
    long position = Math.floorMod(tick, (long) yearLength);
    long shifted = position + yearLength / 8L;
    int quarter = (int) Math.floorMod(shifted / (yearLength / 4L), 4L);
    long seasonStart = (long) quarter * (yearLength / 4L) - yearLength / 8L;
    return Math.floorMod(position - seasonStart, (long) yearLength);
}
```
