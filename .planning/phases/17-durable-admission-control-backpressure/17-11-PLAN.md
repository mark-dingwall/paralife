Required API:

```java
package com.paralife.websocket;

public class BlockingWebSocketClient {
    private final List<String> received = Collections.synchronizedList(new ArrayList<>());
    private volatile CountDownLatch holdLatch;   // when non-null, onTextFrame blocks on it
    private Session session;

    public BlockingWebSocketClient() {}

    /** Connect to {@code uri} and block until the WS is open. */
    public void connect(URI uri, Duration timeout) throws Exception { ... }

    /** Send a text frame. */
    public void send(String text) throws IOException { session.getRemote().sendString(text); }

    /** When set true, the next onTextFrame invocation BLOCKS on a latch (drives server queue overflow). */
    public void holdReceive(boolean hold) { this.holdLatch = hold ? new CountDownLatch(1) : null; }

    /** Release the hold so onTextFrame can resume. */
    public void releaseReceive() { CountDownLatch l = holdLatch; holdLatch = null; if (l != null) l.countDown(); }

    /** Snapshot of all received text frames. */
    public List<String> received() { return new ArrayList<>(received); }

    /** Wait until at least {@code count} frames have been received OR timeout. */
    public boolean awaitReceiveCount(int count, Duration timeout) { ... }

    /** Close the session. */
    public void close() { ... }

    @WebSocket
    public class Handler {
        @OnWebSocketConnect public void onConnect(Session session) { BlockingWebSocketClient.this.session = session; }
        @OnWebSocketMessage public void onText(String message) {
            CountDownLatch l = holdLatch;
            if (l != null) {
                try { l.await(); } catch (InterruptedException e) { Thread.currentThread().interrupt(); return; }
            }
            received.add(message);
        }
        @OnWebSocketClose public void onClose(int statusCode, String reason) { ... }
        @OnWebSocketError public void onError(Throwable error) { ... }
    }
}
```
