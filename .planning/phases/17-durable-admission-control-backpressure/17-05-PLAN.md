<objective>
Build the `ResumeTokenRegistry` bean (Phase 17 D-12 / D-13). This is the per-tick grace-window store that lets a STALLED session's entity survive on the grid while its WebSocket is closed, then re-bind the entity to a freshly reconnected session if the client returns a valid token within the grace window.

```java
package com.paralife.admission;

import com.paralife.engine.TickEvent;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.context.event.EventListener;
import org.springframework.core.annotation.Order;
import org.springframework.stereotype.Component;

import java.util.Optional;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.ThreadLocalRandom;
import java.util.function.Consumer;

/**
 * Resume-token grace registry (Phase 17 D-12, D-13).
 *
 * <p>Maps server-issued opaque token → (entityId, originalSessionId, expiresAtTick).
 * On valid {@link #tryRebind}, the consumed token is replaced atomically with
 * a freshly-minted token still bound to the same {@code entityId} (so the client
 * can stall again and recover).
 *
 * <p>Tokens are 18 chars on wire: literal {@code r:} sentinel + 16 lowercase hex.
 * The {@code r:} sentinel disambiguates tokens from effect blocks ({@code f<digit>...})
 * in the {@code S|} sync-frame parser (see Plan 02 / 17-02-SUMMARY.md).
 *
 * <p>Token entropy: 64 bits via {@link ThreadLocalRandom#nextLong()} — sufficient for
 * short-lived grace tokens; replay-resistance is bounded by grace-window expiry.
 */
@Component
public class ResumeTokenRegistry {

    private static final Logger log = LoggerFactory.getLogger(ResumeTokenRegistry.class);

    private final AdmissionConfig admissionConfig;
    private final AdmissionMetrics metrics;
    private final ConcurrentHashMap<String, ResumeEntry> tokenMap = new ConcurrentHashMap<>();

    /** Set by Plan 07 ({@code WorldWebSocketHandler}) wiring. May be null during tests. */
    private volatile Consumer<String> cleanupCallback;

    public ResumeTokenRegistry(AdmissionConfig admissionConfig, AdmissionMetrics metrics) {
        this.admissionConfig = admissionConfig;
        this.metrics = metrics;
    }

    /** Wired by {@code WorldWebSocketHandler} startup to point at {@code cleanupBot(entityId)}. */
    public void setCleanupCallback(Consumer<String> callback) {
        this.cleanupCallback = callback;
    }

    /**
     * Mint a fresh token. Caller (Plan 07 mark-stalled path OR rebind path) provides
     * the entity to associate. {@code expiresAtTick = currentTick + graceWindowTicks}.
     */
    public String issue(String entityId, String sessionId, long currentTick) {
        String token = generateToken();
        long expiresAtTick = currentTick + admissionConfig.backpressure().graceWindowTicks();
        tokenMap.put(token, new ResumeEntry(entityId, sessionId, expiresAtTick));
        metrics.setStalledSessions(tokenMap.size());
        return token;
    }

    /**
     * Attempt to consume {@code token} for re-bind. Returns {@code RebindOutcome}
     * carrying the preserved entityId and a freshly-minted token (so the client
     * can stall again). Returns empty if token unknown or expired.
     */
    public Optional<RebindOutcome> tryRebind(String token, String newSessionId, long currentTick) {
        if (token == null) return Optional.empty();
        ResumeEntry old = tokenMap.remove(token);
        if (old == null) return Optional.empty();
        if (old.expiresAtTick < currentTick) {
            // Expired between sweep windows — treat as not found.
            metrics.setStalledSessions(tokenMap.size());
            return Optional.empty();
        }
        // Mint fresh token for the (now reconnected) entity.
        String freshToken = issue(old.entityId, newSessionId, currentTick);
        return Optional.of(new RebindOutcome(old.entityId, freshToken));
    }

    /** Tick-driven expiry sweep (D-12). Runs at {@code @Order(1)} before sim core. */
    @EventListener
    @Order(1)
    public void onTick(TickEvent event) {
        long currentTick = event.tickNumber();
        Consumer<String> cleanup = this.cleanupCallback;
        tokenMap.entrySet().removeIf(e -> {
            if (e.getValue().expiresAtTick < currentTick) {
                String entityId = e.getValue().entityId;
                if (cleanup != null) {
                    try {
                        cleanup.accept(entityId);
                    } catch (RuntimeException ex) {
                        log.warn("Cleanup callback failed for entity={}: {}", entityId, ex.getMessage());
                    }
                } else {
                    log.warn("BACKPRESSURE expired tick={} entity={} no-cleanup-callback", currentTick, entityId);
                }
                log.info("BACKPRESSURE expired tick={} entity={} session={}",
                        currentTick, entityId, e.getValue().originalSessionId);
                return true;
            }
            return false;
        });
        metrics.setStalledSessions(tokenMap.size());
    }

    private static String generateToken() {
        return String.format("r:%016x", ThreadLocalRandom.current().nextLong());
    }

    /** Stored value: maps token → (entityId, originalSessionId, expiresAtTick). */
    record ResumeEntry(String entityId, String originalSessionId, long expiresAtTick) {}

    /** Result of a successful rebind: preserved entity + freshly-minted token. */
    public record RebindOutcome(String entityId, String freshResumeToken) {}

    // Test-only accessors:
    int size() { return tokenMap.size(); }
    boolean contains(String token) { return tokenMap.containsKey(token); }
    Optional<ResumeEntry> peek(String token) { return Optional.ofNullable(tokenMap.get(token)); }
}
```

```java
package com.paralife.admission;

import com.paralife.engine.TickEvent;
import io.micrometer.core.instrument.simple.SimpleMeterRegistry;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

import java.util.ArrayList;
import java.util.List;
import java.util.Optional;
import java.util.regex.Pattern;

import static org.assertj.core.api.Assertions.assertThat;

class ResumeTokenRegistryTest {

    private static final Pattern TOKEN_FORMAT = Pattern.compile("^r:[0-9a-f]{16}$");

    private AdmissionConfig admissionConfig;
    private AdmissionMetrics metrics;
    private ResumeTokenRegistry registry;

    @BeforeEach
    void setup() {
        admissionConfig = new AdmissionConfig(
                256, false,
                AdmissionConfig.TickOverloadConfig.defaults(),
                new AdmissionConfig.BackpressureConfig(16, 5));   // grace=5
        metrics = new AdmissionMetrics(new SimpleMeterRegistry());
        registry = new ResumeTokenRegistry(admissionConfig, metrics);
    }

    @Test
    void issuedTokenMatchesFormat() {
        String token = registry.issue("entity-1", "session-1", 100L);
        assertThat(token).matches(TOKEN_FORMAT);
        assertThat(registry.size()).isEqualTo(1);
    }

    @Test
    void issuedTokensAreUnique() {
        String t1 = registry.issue("e1", "s1", 100L);
        String t2 = registry.issue("e2", "s2", 100L);
        assertThat(t1).isNotEqualTo(t2);
    }

    @Test
    void rebindWithKnownTokenReturnsEntityAndFreshToken() {
        String t = registry.issue("entity-1", "session-1", 100L);
        Optional<ResumeTokenRegistry.RebindOutcome> result =
                registry.tryRebind(t, "session-2", 102L);
        assertThat(result).isPresent();
        assertThat(result.get().entityId()).isEqualTo("entity-1");
        assertThat(result.get().freshResumeToken())
                .matches(TOKEN_FORMAT)
                .isNotEqualTo(t);                        // fresh token, old consumed
        assertThat(registry.contains(t)).isFalse();      // old single-use consumed
        assertThat(registry.contains(result.get().freshResumeToken())).isTrue();
    }

    @Test
    void rebindWithUnknownTokenReturnsEmpty() {
        Optional<ResumeTokenRegistry.RebindOutcome> r =
                registry.tryRebind("r:0000000000000000", "session-x", 100L);
        assertThat(r).isEmpty();
    }

    @Test
    void rebindWithNullTokenReturnsEmpty() {
        Optional<ResumeTokenRegistry.RebindOutcome> r = registry.tryRebind(null, "session-x", 100L);
        assertThat(r).isEmpty();
    }

    @Test
    void doubleRebindOfSameTokenFails() {
        String t = registry.issue("e1", "s1", 100L);
        Optional<ResumeTokenRegistry.RebindOutcome> first = registry.tryRebind(t, "s2", 102L);
        assertThat(first).isPresent();
        Optional<ResumeTokenRegistry.RebindOutcome> second = registry.tryRebind(t, "s3", 103L);
        assertThat(second).isEmpty();          // single-use; old token gone
    }

    @Test
    void expiredTokenIsRejectedByTryRebind() {
        // grace=5; issued at tick=100; expires at 105.
        String t = registry.issue("e1", "s1", 100L);
        Optional<ResumeTokenRegistry.RebindOutcome> r = registry.tryRebind(t, "s2", 110L);
        assertThat(r).isEmpty();
    }

    @Test
    void sweepRemovesExpiredEntriesAndInvokesCleanup() {
        List<String> reaped = new ArrayList<>();
        registry.setCleanupCallback(reaped::add);

        registry.issue("entity-1", "s1", 100L);     // expires at 105
        registry.issue("entity-2", "s2", 100L);     // expires at 105
        registry.issue("entity-3", "s3", 110L);     // expires at 115

        registry.onTick(new TickEvent(106L));        // sweep: 1 and 2 expire (105 < 106), 3 remains

        assertThat(reaped).containsExactlyInAnyOrder("entity-1", "entity-2");
        assertThat(registry.size()).isEqualTo(1);
    }

    @Test
    void sweepWithoutCallbackStillRemovesEntries() {
        registry.issue("e1", "s1", 100L);
        registry.onTick(new TickEvent(110L));
        assertThat(registry.size()).isEqualTo(0);
    }

    @Test
    void rebindRefreshesGraceWindow() {
        String t = registry.issue("e1", "s1", 100L);    // expires 105
        Optional<ResumeTokenRegistry.RebindOutcome> r = registry.tryRebind(t, "s2", 103L);
        assertThat(r).isPresent();
        // Fresh token has expiresAtTick = 103 + 5 = 108
        registry.onTick(new TickEvent(106L));            // not yet expired
        assertThat(registry.contains(r.get().freshResumeToken())).isTrue();
        registry.onTick(new TickEvent(109L));            // expired (108 < 109)
        assertThat(registry.contains(r.get().freshResumeToken())).isFalse();
    }

    @Test
    void stalledSessionsGaugeReflectsRegistrySize() {
        SimpleMeterRegistry meterReg = new SimpleMeterRegistry();
        AdmissionMetrics m = new AdmissionMetrics(meterReg);
        ResumeTokenRegistry r = new ResumeTokenRegistry(admissionConfig, m);

        r.issue("e1", "s1", 100L);
        r.issue("e2", "s2", 100L);
        assertThat(meterReg.get(AdmissionMetrics.M_STALLED_SESSIONS).gauge().value()).isEqualTo(2.0);

        r.onTick(new TickEvent(110L));
        assertThat(meterReg.get(AdmissionMetrics.M_STALLED_SESSIONS).gauge().value()).isEqualTo(0.0);
    }
}
```
