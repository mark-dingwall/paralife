<objective>
Build the per-session VT-per-queue outbound sender (D-10, D-11) — the structural backpressure mechanism that isolates slow clients from the tick thread.

Purpose: Today `TickBroadcaster.onTick` calls `synchronized(session) { session.sendMessage(...) }` for every connected client. One slow socket holds the lock and blocks the tick thread. Phase 17 replaces this with `outboundSender.offer(sessionId, frame)` (non-blocking) plus a per-session VT that drains the queue. When the queue saturates, `offer` returns false and Plan 07 reacts (markStalled). VTs are cheap (a few KB heap each) so 1000+ connected clients = 1000+ VTs is acceptable per RESEARCH §Pattern 1.

Output: One new `@Component` (`OutboundSender`) and a comprehensive test class. **Plan 07 wires this** into `WorldWebSocketHandler` and `TickBroadcaster`; **Plan 06 only ships the bean and its tests**. The `WorldWebSocketHandler.markStalled` and `TickBroadcaster.onTick` refactors live in Plans 07 and 08 respectively.
</objective>

```java
package com.paralife.admission;

import com.paralife.codec.Frame;
import com.paralife.codec.PerceptionCodec;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.stereotype.Component;
import org.springframework.web.socket.TextMessage;
import org.springframework.web.socket.WebSocketSession;

import java.io.IOException;
import java.util.concurrent.ArrayBlockingQueue;
import java.util.concurrent.ConcurrentHashMap;
import java.util.function.BiConsumer;

/**
 * Per-session virtual-thread outbound sender (Phase 17 D-10, D-11).
 *
 * <p>Each open WebSocket session is paired with one VT that loops:
 * {@code queue.take(); session.sendMessage(...)}. The VT is the single writer to
 * its session — no {@code synchronized(session)} guard is required on the send path.
 * When blocked on {@code take()}, the VT yields its carrier, so 1000+ idle senders
 * cost O(N × few KB) heap and zero carrier threads.
 *
 * <p><b>Outbound concurrency rationale</b> (also documented in {@code CLAUDE.md}
 * "Outbound concurrency" sub-section):
 * <ul>
 *   <li>Matches Paralife's stated philosophy: simple blocking code, virtual threads do concurrency
 *       ({@code spring.threads.virtual.enabled: true}).</li>
 *   <li>Per-session isolation is structural — one slow socket cannot block the tick
 *       thread or any other session.</li>
 *   <li>{@code queue.size()} is the explicit backpressure signal — observable as a gauge.</li>
 *   <li>Java 21 VTs scheduled on shared carriers; per-VT cost is a few KB heap.</li>
 *   <li>Considered alternative — Jetty native async write — was rejected because slow-client
 *       detection becomes implicit (write-Future latency / Jetty internals) and the API surface
 *       differs across Jetty 12 minor versions.</li>
 * </ul>
 */
@Component
public class OutboundSender {

    private static final Logger log = LoggerFactory.getLogger(OutboundSender.class);

    private final ConcurrentHashMap<String, ArrayBlockingQueue<Frame>> queues = new ConcurrentHashMap<>();
    private final ConcurrentHashMap<String, Thread> senderThreads = new ConcurrentHashMap<>();

    /** Set by Plan 07 startup. Receives (sessionId, queueCapacity) on overflow. */
    private volatile BiConsumer<String, Integer> overflowCallback;

    public void setOverflowCallback(BiConsumer<String, Integer> callback) {
        this.overflowCallback = callback;
    }

    /** Spawn a named VT to drain frames for {@code session}. Idempotent — re-attach replaces. */
    public void attachSession(WebSocketSession session, int queueCapacity) {
        if (queueCapacity < 1) throw new IllegalArgumentException("queueCapacity must be >= 1");
        String id = session.getId();
        // If already attached (e.g. test re-attach), detach first.
        detachSession(id);
        ArrayBlockingQueue<Frame> queue = new ArrayBlockingQueue<>(queueCapacity);
        queues.put(id, queue);
        Thread t = Thread.ofVirtual()
                .name("ws-sender-" + id)
                .start(() -> drainLoop(session, queue));
        senderThreads.put(id, t);
    }

    /** Interrupt the sender VT and release its queue. Idempotent. */
    public void detachSession(String sessionId) {
        queues.remove(sessionId);
        Thread t = senderThreads.remove(sessionId);
        if (t != null) t.interrupt();
    }

    /**
     * Non-blocking enqueue. Returns false if no queue exists for {@code sessionId}
     * OR the queue is full. On overflow, the registered callback is invoked once
     * (synchronously) before false is returned.
     */
    public boolean offer(String sessionId, Frame frame) {
        ArrayBlockingQueue<Frame> queue = queues.get(sessionId);
        if (queue == null) return false;
        boolean accepted = queue.offer(frame);
        if (!accepted) {
            BiConsumer<String, Integer> cb = this.overflowCallback;
            if (cb != null) {
                try {
                    cb.accept(sessionId, queue.size());
                } catch (RuntimeException ex) {
                    log.warn("Overflow callback failed for {}: {}", sessionId, ex.getMessage());
                }
            }
        }
        return accepted;
    }

    /** Test/operator hook: current queue depth (or -1 if not attached). */
    public int queueDepth(String sessionId) {
        ArrayBlockingQueue<Frame> q = queues.get(sessionId);
        return q == null ? -1 : q.size();
    }

    /** Number of attached sessions. */
    public int attachedCount() {
        return queues.size();
    }

    private void drainLoop(WebSocketSession session, ArrayBlockingQueue<Frame> queue) {
        try {
            while (!Thread.currentThread().isInterrupted()) {
                Frame frame = queue.take();           // VT yields carrier while empty
                if (!session.isOpen()) continue;       // session closed; drop and continue (clean shutdown shortly)
                try {
                    String encoded = PerceptionCodec.encode(frame);
                    session.sendMessage(new TextMessage(encoded));
                } catch (IOException e) {
                    log.warn("Send failed for {}: {}", session.getId(), e.getMessage());
                    // continue: the next interrupt or session.isOpen() check will exit
                } catch (RuntimeException e) {
                    log.warn("Send error for {}: {}", session.getId(), e.getMessage());
                }
            }
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();        // clean exit on detach
        }
    }
}
```

package com.paralife.admission;

import com.paralife.codec.Frame;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;
import org.springframework.http.HttpHeaders;
import org.springframework.web.socket.CloseStatus;
import org.springframework.web.socket.WebSocketMessage;
import org.springframework.web.socket.WebSocketSession;

import java.net.InetSocketAddress;
import java.net.URI;
import java.security.Principal;
import java.util.*;
import java.util.concurrent.*;
import java.util.concurrent.atomic.AtomicBoolean;
import java.util.concurrent.atomic.AtomicInteger;

import static org.assertj.core.api.Assertions.assertThat;
import static org.awaitility.Awaitility.await;

class OutboundSenderTest {

    private OutboundSender sender;

    @BeforeEach
    void setup() {
        sender = new OutboundSender();
    }

    @Test
    void attachAndOfferEnqueuesAndDelivers() throws Exception {
        FakeSession s = new FakeSession("session-1");
        sender.attachSession(s, 4);

        boolean ok = sender.offer("session-1", new Frame.RegisterFrame('C'));
        assertThat(ok).isTrue();

        await().atMost(2, TimeUnit.SECONDS).until(() -> s.captured.size() >= 1);
        assertThat(s.captured.get(0)).startsWith("r|C");
    }

    @Test
    void offerToUnknownSessionReturnsFalse() {
        boolean ok = sender.offer("nope", new Frame.RegisterFrame('C'));
        assertThat(ok).isFalse();
    }

    @Test
    void offerReturnsFalseWhenQueueFull() throws Exception {
        FakeSession blocking = FakeSession.blocking("session-blocking");
        sender.attachSession(blocking, 2);

        // Block the drain VT by holding the latch in sendMessage.
        blocking.holdNextSend(true);

        // First send is in-flight (held); next two fill the queue (capacity=2).
        sender.offer("session-blocking", new Frame.RegisterFrame('C'));   // dispatched, in-flight
        boolean a = sender.offer("session-blocking", new Frame.RegisterFrame('M'));   // queued
        boolean b = sender.offer("session-blocking", new Frame.RegisterFrame('S'));   // queued (queue=2)
        boolean c = sender.offer("session-blocking", new Frame.RegisterFrame('C'));   // queue full
        assertThat(a).isTrue();
        assertThat(b).isTrue();
        assertThat(c).isFalse();

        blocking.releaseAll();
        sender.detachSession("session-blocking");
    }

    @Test
    void overflowCallbackInvokedOnQueueFull() throws Exception {
        AtomicInteger callbackCount = new AtomicInteger();
        AtomicInteger lastDepth = new AtomicInteger(-1);
        sender.setOverflowCallback((sid, depth) -> { callbackCount.incrementAndGet(); lastDepth.set(depth); });

        FakeSession blocking = FakeSession.blocking("session-cb");
        sender.attachSession(blocking, 1);
        blocking.holdNextSend(true);

        sender.offer("session-cb", new Frame.RegisterFrame('C'));   // in-flight
        sender.offer("session-cb", new Frame.RegisterFrame('M'));   // queued (1/1)
        sender.offer("session-cb", new Frame.RegisterFrame('S'));   // overflow → callback

        assertThat(callbackCount.get()).isEqualTo(1);
        assertThat(lastDepth.get()).isEqualTo(1);

        blocking.releaseAll();
        sender.detachSession("session-cb");
    }

    @Test
    void detachInterruptsVT() throws Exception {
        FakeSession s = new FakeSession("session-d");
        sender.attachSession(s, 4);
        Thread vt = Thread.currentThread();   // placeholder; we read the VT via reflection or via interface — simpler: just verify queueDepth becomes -1.
        sender.detachSession("session-d");
        await().atMost(2, TimeUnit.SECONDS).until(() -> sender.queueDepth("session-d") == -1);
    }

    @Test
    void drainLoopSurvivesIOException() throws Exception {
        FakeSession s = new FakeSession("session-err");
        s.failNextSend = true;
        sender.attachSession(s, 4);

        sender.offer("session-err", new Frame.RegisterFrame('C'));   // throws IOException internally
        sender.offer("session-err", new Frame.RegisterFrame('M'));   // succeeds

        await().atMost(2, TimeUnit.SECONDS).until(() -> s.captured.size() >= 1);
        // Note: the failed send doesn't appear in captured; the second one does.
        sender.detachSession("session-err");
    }

    @Test
    void reattachAfterDetachIsIdempotent() throws Exception {
        FakeSession s1 = new FakeSession("session-r");
        sender.attachSession(s1, 4);
        sender.detachSession("session-r");
        FakeSession s2 = new FakeSession("session-r");   // same sessionId
        sender.attachSession(s2, 4);
        sender.offer("session-r", new Frame.RegisterFrame('C'));
        await().atMost(2, TimeUnit.SECONDS).until(() -> s2.captured.size() >= 1);
        sender.detachSession("session-r");
    }

    /** Minimal FakeSession capturing sendMessage calls. */
    static class FakeSession implements WebSocketSession {
        final String id;
        final List<String> captured = Collections.synchronizedList(new ArrayList<>());
        volatile boolean failNextSend = false;
        private volatile CountDownLatch holdLatch;

        FakeSession(String id) { this.id = id; }

        static FakeSession blocking(String id) { return new FakeSession(id); }

        void holdNextSend(boolean hold) {
            this.holdLatch = hold ? new CountDownLatch(1) : null;
        }

        void releaseAll() {
            CountDownLatch l = this.holdLatch;
            this.holdLatch = null;
            if (l != null) l.countDown();
        }

        @Override public String getId() { return id; }
        @Override public boolean isOpen() { return true; }
        @Override public void sendMessage(WebSocketMessage<?> message) throws java.io.IOException {
            CountDownLatch l = this.holdLatch;
            if (l != null) {
                try { l.await(); } catch (InterruptedException e) { Thread.currentThread().interrupt(); }
            }
            if (failNextSend) {
                failNextSend = false;
                throw new java.io.IOException("simulated send failure");
            }
            captured.add(message.getPayload().toString());
        }
        // The remaining WebSocketSession methods can throw UnsupportedOperationException.
        @Override public URI getUri() { return null; }
        @Override public HttpHeaders getHandshakeHeaders() { return new HttpHeaders(); }
        @Override public Map<String, Object> getAttributes() { return new ConcurrentHashMap<>(); }
        @Override public Principal getPrincipal() { return null; }
        @Override public InetSocketAddress getLocalAddress() { return null; }
        @Override public InetSocketAddress getRemoteAddress() { return null; }
        @Override public String getAcceptedProtocol() { return null; }
        @Override public void setTextMessageSizeLimit(int messageSizeLimit) {}
        @Override public int getTextMessageSizeLimit() { return 0; }
        @Override public void setBinaryMessageSizeLimit(int messageSizeLimit) {}
        @Override public int getBinaryMessageSizeLimit() { return 0; }
        @Override public List<org.springframework.web.socket.WebSocketExtension> getExtensions() { return List.of(); }
        @Override public void close() {}
        @Override public void close(CloseStatus status) {}
    }
}
