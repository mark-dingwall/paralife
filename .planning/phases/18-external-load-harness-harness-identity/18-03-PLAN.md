    ```java
    package com.paralife.admission;

    import com.paralife.bot.BotIdentity;
    import io.micrometer.core.instrument.Tags;
    import org.springframework.web.socket.WebSocketSession;

    import java.util.Set;

    /**
     * Single source of truth for converting a {@link WebSocketSession} into the
     * Phase 18 D-11 two-tag attribution scheme:
     * <ul>
     *   <li>{@code source} ∈ {@link #SOURCE_TAXONOMY}; defaults to {@code "unknown"} when absent</li>
     *   <li>{@code harness} present only when {@code source=harness} (per D-11)</li>
     * </ul>
     *
     * <p>Used by {@link AdmissionMetrics} for tag construction and by
     * {@link com.paralife.websocket.WorldWebSocketHandler} for log-marker field formatting.
     */
    public final class AttributionTagger {

        public static final String ATTR_SOURCE  = "source";
        public static final String ATTR_HARNESS = "harness";
        public static final Set<String> SOURCE_TAXONOMY = BotIdentity.SOURCE_TAXONOMY;

        private AttributionTagger() {}

        /**
         * Build Micrometer tags for a session. Returns a singleton {@code source=unknown}
         * tag set when {@code session} is null (used by metric-emission sites that have no
         * session context, e.g., grid-full placement failure when session is closed).
         */
        public static Tags tagsFor(WebSocketSession session) {
            if (session == null) {
                return Tags.of("source", "unknown");
            }
            String source = sourceOf(session);
            String harness = harnessOf(session);
            return harness == null
                    ? Tags.of("source", source)
                    : Tags.of("source", source, "harness", harness);
        }

        /** Returns the source attribute value or "unknown" if absent / unknown taxonomy. */
        public static String sourceOf(WebSocketSession session) {
            if (session == null) return "unknown";
            Object v = session.getAttributes().get(ATTR_SOURCE);
            if (v instanceof String s && SOURCE_TAXONOMY.contains(s)) return s;
            return "unknown";
        }

        /** Returns the harness id or null if absent. */
        public static String harnessOf(WebSocketSession session) {
            if (session == null) return null;
            Object v = session.getAttributes().get(ATTR_HARNESS);
            return v instanceof String s ? s : null;
        }

        /**
         * Formats source/harness as a space-delimited log-marker field suffix
         * (D-13 grep-friendly style, mirrors Phase 16 D-15).
         * Examples:
         * <ul>
         *   <li>{@code "source=harness harness=harness-A"} (harness-tagged session)</li>
         *   <li>{@code "source=operator"} (no harness)</li>
         *   <li>{@code "source=unknown"} (null session)</li>
         * </ul>
         */
        public static String formatLogFields(WebSocketSession session) {
            String source = sourceOf(session);
            String harness = harnessOf(session);
            return harness == null
                    ? "source=" + source
                    : "source=" + source + " harness=" + harness;
        }
    }
    ```

    ```java
    @Component
    public class AdmissionMetrics {
        // … existing M_ constants …

        private final MeterRegistry registry;
        private final AdmissionConfig admissionConfig;
        private final com.paralife.engine.TickEngine tickEngine;

        // Tag-keyed buckets for per-source[, harness] gauges. SINGLE source of truth — there
        // is no parallel scalar/aggregate counter; the sum across buckets is the total.
        private final ConcurrentHashMap<Tags, AtomicInteger> activeBuckets = new ConcurrentHashMap<>();
        private final ConcurrentHashMap<Tags, AtomicInteger> stalledBuckets = new ConcurrentHashMap<>();
        // Scalar gauges (D-12 stay scalar).
        private final AtomicInteger maintenance    = new AtomicInteger();
        private final AtomicLong    lastTickWorkMs = new AtomicLong();
        // Phase 17 scalar counters (terminal events; SLI denominators) — stay scalar.
        private final Counter rebound;
        private final Counter terminalDropouts;
        private final Counter stalledTotal;
        private final DistributionSummary frameSize;

        // Cardinality-cap state for warn-once.
        private final java.util.concurrent.atomic.AtomicBoolean overflowWarned =
                new java.util.concurrent.atomic.AtomicBoolean(false);

        public AdmissionMetrics(MeterRegistry registry,
                                 AdmissionConfig admissionConfig,
                                 com.paralife.engine.TickEngine tickEngine) {
            this.registry = registry;
            this.admissionConfig = admissionConfig;
            this.tickEngine = tickEngine;
            // … register existing scalar Counters (rebound, terminalDropouts, stalledTotal, frameSize) …
            // Register scalar gauges (D-12 stay scalar).
            Gauge.builder(M_MAINTENANCE, maintenance, AtomicInteger::get)
                    .description("Maintenance flag mirror: 0=off, 1=on (D-18); STAYS SCALAR per Phase 18 D-12")
                    .register(registry);
            Gauge.builder(M_TICK_WORK_MS, lastTickWorkMs, AtomicLong::get)
                    .description("Tick work-time ms; STAYS SCALAR per Phase 18 D-12")
                    .register(registry);
        }

        @PostConstruct
        void registerCardinalityCap() {
            int cap = admissionConfig.attribution().maxHarnessCardinality();
            // Inline mapper that folds over-cap values to "overflow" AND warns once.
            MeterFilter overflowMapper = new MeterFilter() {
                @Override
                public Meter.Id map(Meter.Id id) {
                    String existing = id.getTag("harness");
                    if (existing != null && overflowWarned.compareAndSet(false, true)) {
                        log.warn("HARNESS overflow first-seen tick={} harness-id={}",
                                tickEngine.currentTick(),
                                existing.length() > 32 ? existing.substring(0, 32) : existing);
                    }
                    return id.withTag(Tag.of("harness", "overflow"));
                }
            };
            registry.config().meterFilter(
                    MeterFilter.maximumAllowableTags("paralife.admission", "harness", cap, overflowMapper));
            registry.config().meterFilter(
                    MeterFilter.maximumAllowableTags("paralife.backpressure", "harness", cap, overflowMapper));
        }

        // Two-tag overload (preferred). Single-arg overload (incRejected(String)) defaults
        // session=null → source=unknown, used only by sites with no session context.
        public void incRejected(String reason, WebSocketSession session) {
            Counter.builder(M_REJECTED)
                    .tag("reason", reason)
                    .tags(AttributionTagger.tagsFor(session))
                    .description("Admission rejections by reason token (Phase 17 D-17, Phase 18 D-12)")
                    .register(registry)
                    .increment();
        }
        public void incRejected(String reason) { incRejected(reason, null); }

        public void incIngressOverwrite(WebSocketSession session) {
            Counter.builder(M_INGRESS_OVERWRITES)
                    .tags(AttributionTagger.tagsFor(session))
                    .description("Action-frame ingress overwrites (D-09; tagged per Phase 18 D-12)")
                    .register(registry)
                    .increment();
        }

        // Per-bucket inc/dec API — SINGLE source of truth for active.entities and stalled.sessions.
        // Callers (WorldWebSocketHandler) invoke inc on Allow/markStalled, dec on cleanup/reconnect.
        // First observation of a (source[, harness]) tag combo lazily registers a Gauge for that combo.
        public void incActiveBucket(WebSocketSession session) {
            Tags tags = AttributionTagger.tagsFor(session);
            activeBuckets.computeIfAbsent(tags, t -> {
                AtomicInteger ai = new AtomicInteger();
                Gauge.builder(M_ACTIVE_ENTITIES, ai, AtomicInteger::get)
                        .tags(t)
                        .description("Live cap-relevant occupants per source[, harness] (D-18, Phase 18 D-12)")
                        .register(registry);
                return ai;
            }).incrementAndGet();
        }

        public void decActiveBucket(WebSocketSession session) {
            Tags tags = AttributionTagger.tagsFor(session);
            AtomicInteger bucket = activeBuckets.get(tags);
            if (bucket != null) bucket.decrementAndGet();
        }

        public void incStalledBucket(WebSocketSession session) {
            Tags tags = AttributionTagger.tagsFor(session);
            stalledBuckets.computeIfAbsent(tags, t -> {
                AtomicInteger ai = new AtomicInteger();
                Gauge.builder(M_STALLED_SESSIONS, ai, AtomicInteger::get)
                        .tags(t)
                        .description("STALLED sessions per source[, harness] (D-18, Phase 18 D-12)")
                        .register(registry);
                return ai;
            }).incrementAndGet();
        }

        public void decStalledBucket(WebSocketSession session) {
            Tags tags = AttributionTagger.tagsFor(session);
            AtomicInteger bucket = stalledBuckets.get(tags);
            if (bucket != null) bucket.decrementAndGet();
        }

        // NOTE: No setActiveEntities(int) / setStalledSessions(int) shims. Legacy scalar setters
        // are deleted (not preserved) to lock the single-source-of-truth invariant — every
        // mutation flows through the per-bucket inc/dec API. Call sites must update; the

        // every edit to this file.

        // Scalar setters retained.
        public void setMaintenance(boolean on) { maintenance.set(on ? 1 : 0); }
        public void setLastTickWorkMs(long ms) { lastTickWorkMs.set(ms); }
        // Existing terminal-event counters unchanged.
        public void incRebound() { rebound.increment(); }
        public void incTerminalDropout() { terminalDropouts.increment(); }
        public void incStalledTotal() { stalledTotal.increment(); }
        public void recordFrameSize(int bytes) { frameSize.record(bytes); }
    }
    ```
