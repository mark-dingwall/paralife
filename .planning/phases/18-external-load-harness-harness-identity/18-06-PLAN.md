    import com.paralife.bot.BotClient;
    import com.paralife.bot.BotClientOptions;

    import org.springframework.boot.test.context.SpringBootTest;
    import org.springframework.boot.test.web.server.LocalServerPort;
    import org.springframework.test.context.TestPropertySource;
    import org.springframework.web.socket.WebSocketSession;
    import com.paralife.websocket.SessionRegistry;

    import java.time.Duration;
    import java.util.Random;
    import java.util.concurrent.TimeUnit;

    import static org.assertj.core.api.Assertions.assertThat;

    @SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
    @TestPropertySource(properties = {
            "paralife.admission.cap=8",
            "paralife.admission.backpressure.outbound-queue-size=4",
            "paralife.admission.backpressure.grace-window-ticks=50"
    })
    class AttributionRebindTest {

        @LocalServerPort int port;
        @Autowired WorldWebSocketHandler handler;
        @Autowired SessionRegistry sessionRegistry;
        @Autowired MeterRegistry meterRegistry;
        @Autowired com.paralife.engine.TickEngine tickEngine;

        @Test
        void stalledPivotPreservesSourceAndHarnessAttribution() throws Exception {
            String uri = "ws://localhost:" + port + "/ws/world";
            BotClient bot = new BotClient(new BotClientOptions(
                    uri, 'C', new HeuristicBrain(HeuristicBrain.REPRODUCE_THRESHOLD),
                    100L, 50L, new Random(), BotIdentity.harness("test-attribution")));
            bot.connect();
            assertThat(bot.awaitRegistered(5_000L)).isTrue();
            String firstSessionId = bot.getEntityId();   // proxy: at least one session is up

            // Find the server-side session and force STALLED.
            WebSocketSession session = sessionRegistry.getSessions().stream()
                    .filter(s -> "test-attribution".equals(s.getAttributes().get(AttributionTagger.ATTR_HARNESS)))
                    .findFirst().orElseThrow();
            handler.markStalled(session, tickEngine.currentTick());

            // Bot should receive E|408 and reconnect within ~500ms (jitter 100-300ms).
            Awaitility.await()
                    .atMost(Duration.ofSeconds(3))
                    .pollInterval(Duration.ofMillis(50))
                    .untilAsserted(() -> {
                        assertThat(bot.getE408ReconnectRequiredCount()).isGreaterThanOrEqualTo(1);
                    });

            // Wait for rebind: a NEW server-side session with ATTR_HARNESS=test-attribution.
            Awaitility.await()
                    .atMost(Duration.ofSeconds(5))
                    .pollInterval(Duration.ofMillis(50))
                    .untilAsserted(() -> {
                        long n = sessionRegistry.getSessions().stream()
                                .filter(s -> "test-attribution".equals(s.getAttributes().get(AttributionTagger.ATTR_HARNESS)))
                                .filter(s -> "harness".equals(s.getAttributes().get(AttributionTagger.ATTR_SOURCE)))
                                .filter(s -> s.getAttributes().containsKey("entityId"))
                                .count();
                        assertThat(n).isGreaterThanOrEqualTo(1);
                    });

            // Assert the gauge is tagged with harness=test-attribution and value >= 1.
            Awaitility.await()
                    .atMost(Duration.ofSeconds(2))
                    .untilAsserted(() -> {
                        Double v = meterRegistry.find("paralife.admission.active.entities")
                                .tags("source", "harness", "harness", "test-attribution")
                                .gauge() == null
                                ? null
                                : meterRegistry.find("paralife.admission.active.entities")
                                        .tags("source", "harness", "harness", "test-attribution")
                                        .gauge().value();
                        assertThat(v).isNotNull();
                        assertThat(v).isGreaterThanOrEqualTo(1.0);
                    });

            // Negative assertion: the unknown-source gauge MUST NOT have grown to absorb the rebound bot.
            // (Sanity for T-18-04.)
            Double unknownValue = meterRegistry.find("paralife.admission.active.entities")
                    .tags("source", "unknown")
                    .gauge() == null ? 0.0
                    : meterRegistry.find("paralife.admission.active.entities")
                            .tags("source", "unknown").gauge().value();
            assertThat(unknownValue).isLessThan(1.0);

            bot.disconnect();
        }
    }

    ## §1 Architectural Principles (D-05 / D-21)

    Paralife is a **massively parallel WebSocket architecture**. The following principle is a stated project property:

    > **WS:entity 1:1.** One WebSocket connection carries one entity for its lifetime. Multiplexing / multi-entity-per-session is **strongly discouraged**.

    Rationale: many concurrent WebSocket connections is itself a meaningful architectural property — Paralife's value proposition is that simple blocking code over Java 21 virtual threads scales to thousands of concurrent connections per JVM. Reducing connection count via multiplexing actively undermines that property.

    Exceptions are case-by-case (D-21) and require an ADR or future-phase spec. The default is always 1:1.

    ## §2 Identity Wire Shape (D-06 / D-07 / D-08 / D-09)

    Identity rides on the WebSocket **handshake HTTP headers**:

    | Header | Value | Default when absent |
    |--------|-------|---------------------|
    | `X-Paralife-Source`  | `operator | harness | unknown | overflow | offspring` | `unknown` |
    | `X-Paralife-Harness` | bounded harness id (≤32 chars, alphanum+dash) | absent — only sent when `source=harness` |

    - **Server side:** `WorldWebSocketHandler.afterConnectionEstablished` reads via `session.getHandshakeHeaders()` (Spring's case-insensitive `HttpHeaders`); stashes as session attributes `ATTR_SOURCE` / `ATTR_HARNESS`.
    - **Source taxonomy is bounded.** Wire values not in `{operator, harness, unknown, overflow, offspring}` fold to `unknown` (defensive — wire input is untrusted; T-18-01).
    - **Harness id is truncated to 32 chars** before storage; CR/LF rejected (defense-in-depth, T-18-01).
    - **`r|` grammar is unchanged.** This is intentional: 15-SCHEMA.md is milestone-locked. Phase 18 carries identity *out of band* of the application protocol.

    ## §3 CLI Surface (D-15 / D-16)

    [Document the LoadHarness CLI flags table — copied from CONTEXT.md / Plan 05.]

    ## §4 Attribution Tagging Schema (D-10 / D-11 / D-12)

    **Two-tag scheme** (D-11):
    - `source ∈ {operator, harness, unknown, overflow, offspring}` — bounded taxonomy. `offspring` reserved-now-with-no-producer (D-20).
    - `harness=<id>` — emitted **only when source=harness**. Bounded by 64-cardinality cap (D-10).

    | Metric | Tags | Notes |
    |--------|------|-------|
    | `paralife.admission.rejected`         | `{reason, source[, harness]}` | extends Phase 17 D-17 |
    | `paralife.admission.active.entities`  | `{source[, harness]}`         | extends Phase 17 D-18 |
    | `paralife.backpressure.stalled.sessions` | `{source[, harness]}`     | extends Phase 17 D-18 |
    | `paralife.admission.ingress.overwrites`  | `{source[, harness]}`     | extends Phase 17 D-09 |
    | `paralife.admission.maintenance`         | (scalar, untagged)         | server-global per D-12 |
    | `paralife.tick.health.work-time-ms`      | (scalar, untagged)         | server-global per D-12 |

    **Bounded-cardinality cap** (D-10): cap=64 distinct `harness` tag values per JVM lifetime. Configurable via `paralife.admission.attribution.max-harness-cardinality`. 65th+ folds to `harness=overflow` with one-time WARN log:

    ```
    HARNESS overflow first-seen tick=<n> harness-id=<truncated>
    ```

    Implemented via `MeterFilter.maximumAllowableTags` + `replaceTagValues` (RESEARCH.md verified `deny()` leaks per micrometer issue #4971).

    ## §6 JSON Report Schema (D-17)

    Report writes are **always atomic via temp+rename** for the overwrite mode and the JSONL header line. Counter lines in append mode use `O_SYNC + APPEND` (per-line atomicity sufficient — RESEARCH.md A5).

    **Overwrite mode (default):** single JSON object always reflecting current state. Filename via `--report-out` (default `./harness-<id>-report.json`).

    **Append mode (JSONL):**
    - Line 1 — header: static config (`harnessId`, `serverUri`, `targetCount`, `startWallTime`, `jvmVersion`, `buildSha`)
    - Lines 2..N — counters, one snapshot per `--report-interval` second window

    Counter set:
    - `peak_registered`, `current_registered`
    - `connect_failures_total`, `e408_reconnect_required_total`
    - `respawns_total`
    - `actions_sent_total`, `perceptions_received_total`, `syncs_received_total`
    - `wall_time_seconds_elapsed`
    - `exit_reason` — present only on the final write; values: `signal-int | signal-term | duration-reached | fatal-error`

    ## §8 Forward Notes

    ### Backlog 999.2: bot-driven offspring (D-19 / D-20)

    ### Multi-entity-per-session exception path (D-21)
