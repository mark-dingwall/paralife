    Step 2 — Edit `src/main/java/com/paralife/bot/BotClient.java` to add a CAS-guarded `onClose(Runnable)` hook (Round 2 Codex HIGH amendment — fire-once invariant):

    Read the existing file first to find the WS `@OnWebSocketClose` callback site (likely around line 400). Add the following:

    ```java
    // Field: list of close-callbacks. CopyOnWriteArrayList for safe iteration during fire.
    private final java.util.concurrent.CopyOnWriteArrayList<Runnable> closeCallbacks =
            new java.util.concurrent.CopyOnWriteArrayList<>();

    // Round 2 Codex HIGH amendment: CAS gate so close callbacks fire EXACTLY ONCE per BotClient
    // lifetime. Without this, disconnect() + Jetty's @OnWebSocketClose path could each call
    // fireCloseCallbacks() → liveCount decremented twice → BotFleet.currentRegistered() goes
    // negative.
    private final java.util.concurrent.atomic.AtomicBoolean closedFired =
            new java.util.concurrent.atomic.AtomicBoolean(false);

    /**
     * Register a callback to be invoked when this BotClient closes (either via explicit
     * {@link #disconnect()} or remote close from the server). Phase 18 D-04: BotFleet
     * uses this to track {@code liveCount} for {@link BotFleet#currentRegistered()}.
     *
     * <p><b>Fire-once contract (Round 2 Codex HIGH):</b> regardless of how many close paths
     * trigger ({@code disconnect()} from owning thread + Jetty {@code @OnWebSocketClose}
     * from the read thread), each registered Runnable runs at most once per BotClient
     * lifetime. The gate is an {@link java.util.concurrent.atomic.AtomicBoolean}
     * compare-and-set; only the first close path wins.
     *
     * <p><b>Note:</b> STALLED-pivot reconnects (Phase 17 D-13) are NOT reported via this hook —
     * the BotClient internal reconnect loop handles those transparently.
     */
    public void onClose(Runnable r) {
        closeCallbacks.add(java.util.Objects.requireNonNull(r));
    }

    /** CAS-guarded fire. Returns true if this call won the race; false if already fired. */
    private void fireCloseCallbacks() {
        if (!closedFired.compareAndSet(false, true)) {
            return;   // another close path already fired the callbacks
        }
        for (Runnable r : closeCallbacks) {
            try { r.run(); } catch (RuntimeException e) { log.warn("close callback failed: {}", e.getMessage()); }
        }
    }
    ```

    Wire `fireCloseCallbacks()` into:
    - `disconnect()` — at the END of the method, in a `finally` block so it fires even on exception
    - The Jetty `@OnWebSocketClose` handler path (likely around line 400 in current code) — fire BEFORE the existing reconnect-jitter logic, so callers see a deterministic close-event count

    ```java
    package com.paralife.bot;

    import com.paralife.world.Entity.ParticleType;
    import org.slf4j.Logger;
    import org.slf4j.LoggerFactory;

    import java.util.List;
    import java.util.Map;
    import java.util.Optional;
    import java.util.concurrent.CompletableFuture;
    import java.util.concurrent.ConcurrentHashMap;
    import java.util.concurrent.CopyOnWriteArrayList;
    import java.util.concurrent.TimeUnit;
    import java.util.concurrent.atomic.AtomicBoolean;
    import java.util.concurrent.atomic.AtomicInteger;

    public final class BotFleet {

        private static final Logger log = LoggerFactory.getLogger(BotFleet.class);

        public record RegistrationResult(String botId, boolean registered, Optional<String> failureReason) {}

        private final List<BotClient> bots = new CopyOnWriteArrayList<>();
        private final Map<String, CompletableFuture<RegistrationResult>> futures = new ConcurrentHashMap<>();
        private final AtomicInteger liveCount = new AtomicInteger();
        private final AtomicInteger highWater = new AtomicInteger();
        // Round 2 Claude MEDIUM amendment: idempotent shutdown.
        private final AtomicBoolean shutdownDone = new AtomicBoolean(false);

        public List<BotClient> launch(String serverUri, int count, BotIdentity identity,
                                       RampUpSpec rampUp, SpeciesMix mix, BotFactory factory) {
            for (int i = 0; i < count; i++) {
                rampUp.awaitNext(i);
                char species = mix.pickFor(i, count);
                BotClient bot = factory.create(species, identity, Optional.empty(), Optional.empty());
                bot.onClose(() -> liveCount.decrementAndGet());
                bots.add(bot);
                String botId = "fleet-" + identity.harnessId().orElse("op") + "-" + i;
                CompletableFuture<RegistrationResult> fut = new CompletableFuture<>();
                futures.put(botId, fut);
                Thread.startVirtualThread(() -> {
                    try {
                        bot.connect();
                        boolean ok = bot.awaitRegistered(15_000L);
                        if (ok) {
                            int live = liveCount.incrementAndGet();
                            highWater.updateAndGet(prev -> Math.max(prev, live));
                        }
                        fut.complete(new RegistrationResult(botId, ok, Optional.empty()));
                    } catch (Exception e) {
                        log.warn("Bot {} failed to connect: {}", botId, e.getMessage());
                        fut.complete(new RegistrationResult(botId, false, Optional.of(e.getMessage())));
                    }
                });
            }
            return List.copyOf(bots);
        }

        /** True high-water mark: never decreases. */
        public int peakRegistered() { return highWater.get(); }

        /**
         * Live registered count. Decreases on {@link BotClient#onClose} firing.
         *
         * <p><b>Best-effort for the ramp window only (OpenCode review amendment).</b> A
         * {@code BotClient} that re-registers after a STALLED pivot (Phase 17 D-13) does so
         * through its own internal reconnect loop, NOT through this fleet, so the post-reconnect
         * register-success is invisible to {@code liveCount}. Authoritative steady-state:
         * {@code paralife.admission.active.entities{source=harness, harness=<id>}} (Plan 03)
         * or per-bot {@link BotClient#isRegistered()} polling.
         */
        public int currentRegistered() { return liveCount.get(); }

        public CompletableFuture<Void> awaitAllSettled() {
            return CompletableFuture.allOf(futures.values().toArray(CompletableFuture[]::new));
        }

        public List<BotClient> getBots() { return List.copyOf(bots); }

        /** Idempotent — safe to call from shutdown hook AND main run() (Round 2 Claude amendment). */
        public void shutdown() {
            if (!shutdownDone.compareAndSet(false, true)) {
                return;
            }
            for (BotClient b : bots) b.disconnect();
            log.info("BotFleet: {} bots disconnected", bots.size());
            bots.clear();
            futures.clear();
        }
    }
    ```

    Step 4 — Create `src/main/java/com/paralife/bot/RampUpSpec.java` with **nanosecond-precision Rate** (Round 2 Codex MEDIUM):

    ```java
    package com.paralife.bot;

    import java.util.concurrent.locks.LockSupport;

    public sealed interface RampUpSpec permits RampUpSpec.Instant, RampUpSpec.Rate, RampUpSpec.Wave {
        void awaitNext(int botIndex);

        record Instant() implements RampUpSpec {
            public void awaitNext(int botIndex) { /* no-op */ }
        }

        /**
         * Round 2 Codex MEDIUM amendment — nanosecond precision. The previous shape
         * {@code 1000L / perSecond} integer-truncated to 0 above 1000/s, defeating
         * high-rate ramps. Compute nanoseconds explicitly: {@code 1_000_000_000L / perSecond}
         * gives 500_000ns at 2000/s, 100_000ns at 10_000/s, etc.
         */
        record Rate(int perSecond) implements RampUpSpec {
            public Rate {
                if (perSecond < 1) throw new IllegalArgumentException("perSecond >= 1");
            }
            public void awaitNext(int botIndex) {
                if (botIndex == 0) return;
                long nanos = 1_000_000_000L / perSecond;
                LockSupport.parkNanos(nanos);
            }
        }

        record Wave(int count, long sleepMs) implements RampUpSpec {
            public Wave {
                if (count < 1 || sleepMs < 0) throw new IllegalArgumentException();
            }
            public void awaitNext(int botIndex) {
                if (botIndex == 0) return;
                if (botIndex % count == 0) {
                    try { Thread.sleep(sleepMs); }
                    catch (InterruptedException e) { Thread.currentThread().interrupt(); }
                }
            }
        }

        static RampUpSpec instant() { return new Instant(); }
        static RampUpSpec rate(int perSecond) { return new Rate(perSecond); }
        static RampUpSpec wave(int count, long sleepMs) { return new Wave(count, sleepMs); }
    ```

    ```java
    import com.paralife.world.Entity.ParticleType;

    public record SpeciesMix(double cFrac, double mFrac, double sFrac) {

        // Round 2 OpenCode MEDIUM amendment: hardcode the order so enum reordering of
        // ParticleType.values() can never silently change species distribution.
        private static final ParticleType[] ORDERED_TYPES =
                { ParticleType.CATALYST, ParticleType.MEMBRANE, ParticleType.SPORE };

        public SpeciesMix {
            double sum = cFrac + mFrac + sFrac;
            if (Math.abs(sum - 1.0) > 0.001) {
                throw new IllegalArgumentException("fractions must sum to 1.0 (got " + sum + ")");
            }
        }

        public static SpeciesMix balanced() {
            return new SpeciesMix(1.0/3, 1.0/3, 1.0/3);
        }

        public char pickFor(int i, int count) {
            // Balanced: round-robin over the FIXED array.
            if (Math.abs(cFrac - 1.0/3) < 0.001 && Math.abs(mFrac - 1.0/3) < 0.001) {
                ParticleType t = ORDERED_TYPES[i % ORDERED_TYPES.length];
                return switch (t) {
                    case CATALYST -> 'C';
                    case MEMBRANE -> 'M';
                    case SPORE    -> 'S';
                };
            }
            // Weighted: deterministic position-based partitioning.
            // For (0.4, 0.3, 0.3) and count=10: bots 0..3 → C, 4..6 → M, 7..9 → S.
            double frac = (i + 0.5) / count;
            if (frac < cFrac) return 'C';
            if (frac < cFrac + mFrac) return 'M';
            return 'S';
        }
    }
    ```
