    ### `src/test/java/com/paralife/engine/ZeroTrustFilteringTest.java`

    ```java
    package com.paralife.engine;

    import com.paralife.codec.Frame;
    import com.paralife.codec.PerceptionCodec;
    import com.paralife.websocket.TickBroadcaster;
    import com.paralife.world.Entity;
    import com.paralife.world.Entity.BondedPair;
    import com.paralife.world.Entity.Particle;
    import com.paralife.world.Entity.ParticleType;
    import com.paralife.world.Position;
    import com.paralife.world.WorldGrid;
    import org.junit.jupiter.api.Test;
    import org.springframework.beans.factory.annotation.Autowired;
    import org.springframework.boot.test.context.SpringBootTest;
    import org.springframework.test.context.TestPropertySource;

    import java.util.regex.Matcher;
    import java.util.regex.Pattern;

    import static org.junit.jupiter.api.Assertions.assertFalse;
    import static org.junit.jupiter.api.Assertions.assertTrue;

    /**
     * Verifies SCHEMA §8.1 zero-trust guarantees:
     *   - No entity id strings appear in any encoded T frame.
     *   - BondedPair neighbours reveal only primary type (D/N/T); secondary is hidden.
     *   - Self cell is never emitted in the s block.
     *
     * <p>Assertions use regex anchored to cell-entry boundaries (coord + presence + kind)
     * per SCHEMA §8.1 grammar — NOT coarse substring matches that would false-positive
     * on tick IDs, expiry ticks, or any base64 char that happens to equal a kind code.
     * Per review feedback (2026-04-20).
     */
    @SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.NONE)
    @TestPropertySource(properties = {
            "paralife.tick.auto-start=false",
            "paralife.world.width=16",
            "paralife.world.height=16"
    })
    class ZeroTrustFilteringTest {

        @Autowired TickBroadcaster broadcaster;
        @Autowired WorldGrid worldGrid;
        @Autowired BotRegistry botRegistry;
        @Autowired com.paralife.websocket.SessionRegistry sessionRegistry;

        /**
         * Cell-entry boundary regex: matches the presence + kind chars of any `s`-block
         * entry. Anchored to valid cell-entry start patterns (numpad digit OR signed
         * relative coord) so that kind char matches ONLY inside actual cell entries,
         * never in a tick ID or expiry tick value.
         *
         * <p>Group 1 captures presence (1|2|3); Group 2 captures the kind char when
         * presence bit 0 is set.
         */
        private static final Pattern CELL_ENTRY_KIND = Pattern.compile(
                // coord: numpad (single digit 1-9) OR relative ([+-][base64][+-][base64], 4 chars)
                "(?:[1-9]|[+\\-][0-9A-Za-z_\\-][+\\-][0-9A-Za-z_\\-])"
                // presence: 1 base64 char
                + "([1-3])"
                // optional kind char — present when presence bit 0 is set (1 or 3)
                + "([CMSDNT0-9RF])?");

        @Test
        void encodedFrameCarriesNoEntityIds() {
            String botId = registerBot(null, 5, 5, ParticleType.CATALYST);
            String bondedId = placeBondedPair(5, 6,
                    ParticleType.CATALYST, ParticleType.SPORE);

            Frame.TickFrame frame = broadcaster.buildTickFrame(
                    botRegistry.getBySession("test-s1").orElseThrow(), 1L);
            String wire = PerceptionCodec.encode(frame);

            assertFalse(wire.contains(botId), "Wire must not contain own bot id: " + wire);
            assertFalse(wire.contains(bondedId), "Wire must not contain neighbour id: " + wire);
            assertFalse(wire.contains("SPORE"), "Wire must not reveal bonded secondary type as string");
        }

        @Test
        void bondedNeighbourEmitsOnlyPrimaryKindCode() {
            String botId = registerBot(null, 5, 5, ParticleType.CATALYST);
            placeBondedPair(5, 6, ParticleType.CATALYST, ParticleType.SPORE);

            Frame.TickFrame frame = broadcaster.buildTickFrame(
                    botRegistry.getBySession("test-s1").orElseThrow(), 1L);
            String wire = PerceptionCodec.encode(frame);

            // Extract the s block. Format: ...|s<entry>,<entry>,...|...
            String sBlock = extractSBlock(wire);
            assertTrue(sBlock != null, "Expected s block in wire: " + wire);

            // Walk cell entries via regex; confirm:
            //   (a) At least one entry has kind 'D' (bonded CAT primary).
            //   (b) NO entry has a kind char that exposes bonded secondary (bonded
            //       secondary type is hidden per SCHEMA §8.1.1; bonded-primary codes
            //       are D/N/T — the presence of primary is the only bonded info).
            //   (c) No raw 'S' kind appears for the bonded cell — the neighbour at
            //       (5,6) is a BondedPair primary=CAT, NOT a solo Spore. A solo 'S'
            //       kind for THAT cell would mean the test rig misrendered.
            boolean sawBondedCatPrimary = false;
            Matcher m = CELL_ENTRY_KIND.matcher(sBlock);
            while (m.find()) {
                String kind = m.group(2);
                if (kind == null) continue;
                if (kind.equals("D")) sawBondedCatPrimary = true;
                // The bonded secondary (Spore) must NOT appear as a solo-kind 'S' in
                // the same s block from this test setup — the only Spore-like entity
                // placed is the bonded secondary, which is hidden. If 'S' appears it
                // indicates bonded-secondary leakage.
                assertFalse("S".equals(kind),
                        "Solo 'S' kind appeared in s block — indicates bonded-secondary leak: " + sBlock);
            }
            assertTrue(sawBondedCatPrimary,
                    "Expected at least one cell entry with bonded-CAT-primary kind 'D' in s block: " + sBlock);
        }

        @Test
        void selfCellIsNeverEmittedInSBlock() {
            registerBot(null, 5, 5, ParticleType.CATALYST);

            Frame.TickFrame frame = broadcaster.buildTickFrame(
                    botRegistry.getBySession("test-s1").orElseThrow(), 1L);

            // Self is at (5,5) which would be numpad '5' relative to itself. SCHEMA §8.1
            // says self is never emitted. Confirm by scanning cell entries.
            for (var cell : frame.cells()) {
                if (cell.coord() instanceof com.paralife.codec.Coord.Numpad n) {
                    assertFalse(n.digit() == '5', "Numpad '5' (self) must not appear in s block");
                }
                if (cell.coord() instanceof com.paralife.codec.Coord.Relative r) {
                    assertFalse(r.dx() == 0 && r.dy() == 0, "Relative (0,0) (self) must not appear in s block");
                }
            }
        }

        private String extractSBlock(String wire) {
            int i = wire.indexOf("|s");
            if (i < 0) return null;
            int end = wire.indexOf('|', i + 2);
            return end < 0 ? wire.substring(i + 2) : wire.substring(i + 2, end);
        }

        private String registerBot(String sessionId, int x, int y, ParticleType type) {
            String sid = sessionId != null ? sessionId : "test-s1";
            String entityId = "test-bot-" + x + "-" + y;
            Particle p = Particle.spawn(entityId, type);
            boolean placed = worldGrid.trySetEntity(x, y, p);
            assertTrue(placed, "Failed to place test bot at (" + x + "," + y + ")");
            botRegistry.register(sid, entityId, new Position(x, y));
            return entityId;
    ```
