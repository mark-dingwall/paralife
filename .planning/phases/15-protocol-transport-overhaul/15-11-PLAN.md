    ```java
    package com.paralife.engine;

    import com.paralife.bot.BotClient;
    import com.paralife.bot.HeuristicBrain;
    import io.micrometer.core.instrument.DistributionSummary;
    import io.micrometer.core.instrument.MeterRegistry;
    import org.junit.jupiter.api.Tag;
    import org.junit.jupiter.api.Test;
    import org.springframework.beans.factory.annotation.Autowired;
    import org.springframework.beans.factory.annotation.Value;
    import org.springframework.boot.test.context.SpringBootTest;
    import org.springframework.boot.test.web.server.LocalServerPort;
    import org.springframework.test.context.TestPropertySource;

    import java.util.ArrayList;
    import java.util.List;
    import java.util.concurrent.TimeUnit;

    import static org.junit.jupiter.api.Assertions.assertTrue;

    /**
     * Performance gate for the encode+deflate path under target load.
     *
     * <p>With {@code server_no_context_takeover=true} every tick frame is
     * compressed cold (no cross-frame dictionary reuse), and every per-bot frame
     * is a unique payload — so CPU cost is highest when this combination runs at
     * target scale. The gate asserts tick-drift p99 stays within a generous
     * multiple of {@code paralife.tick.target-millis}. Higher drift = encode+deflate
     * is eating the tick budget; the gate catches that regression.
     *
     * <p>Per review Codex #c (MEDIUM). Complements correctness tests in 15-02 /
     * 15-05 / 15-08.
     */
    @SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
    @TestPropertySource(properties = {
            "paralife.tick.auto-start=true",
            "paralife.tick.target-millis=500",
            "paralife.world.width=64",
            "paralife.world.height=64",
            "paralife.world.rock.seed=42",
            "management.endpoints.web.exposure.include=health,info,metrics"
    })
    @Tag("performance")
    class EncodeDeflatePerformanceGateTest {

        private static final int BOT_COUNT = 100;
        private static final int TARGET_TICKS = 500;

        @LocalServerPort int port;
        @Autowired MeterRegistry meterRegistry;
        @Value("${paralife.tick.target-millis}")
        long targetMillis;

        @Test
        void encodeDeflateUnder100BotsTickDrift() throws Exception {
            List<BotClient> bots = new ArrayList<>(BOT_COUNT);
            for (int i = 0; i < BOT_COUNT; i++) {
                char species = (i % 3 == 0) ? 'C' : (i % 3 == 1 ? 'M' : 'S');
                BotClient bot = new BotClient("ws://localhost:" + port + "/ws/world",
                        species, new HeuristicBrain(70), 200, 100);
                bot.connect();
                bots.add(bot);
            }
            // Wait for registration of at least 90 bots (some may fail under load; that's
            // itself signal — but don't make the test flaky on perfect-100 requirement).
            for (BotClient b : bots) b.awaitConnected(5_000);
            int registered = 0;
            for (BotClient b : bots) {
                if (b.awaitRegistered(5_000)) registered++;
            }
            assertTrue(registered >= 90,
                    "Expected at least 90 of 100 bots registered; got " + registered);

            // Run for TARGET_TICKS. target-millis=500ms → 500 ticks = 250s; too slow for
            // CI. Use a shorter run (500ms × 50 ticks = 25s) if needed. The assertion
            // is p99 drift relative to target, not absolute wallclock.
            long runMillis = targetMillis * Math.min(50, TARGET_TICKS);
            Thread.sleep(runMillis);

            // p99 drift via the tick.frame.bytes DistributionSummary is not the right
            // signal. The right signal is TickEngine's recorded drift — if TickEngine
            // doesn't expose p99 directly, probe it via a tap: record (actualNanos -
            // targetNanos) into a local DistributionSummary each tick. Implementer
            // adds the tap as a @Bean in a TestConfig inner class if TickEngine
            // doesn't already publish drift.
            DistributionSummary drift = meterRegistry.find("paralife.tick.drift.millis").summary();
            if (drift == null) {
                // Fallback: assert that no bot disconnected mid-run (disconnects under
                // load usually mean the tick loop starved).
                int stillConnected = 0;
                for (BotClient b : bots) {
                    if (b.isConnected()) stillConnected++;
                }
                assertTrue(stillConnected >= 90,
                        "Expected ≥90 bots still connected after run; got " + stillConnected
                                + ". (TickEngine does not publish drift metric; using connection-survival proxy.)");
            } else {
                double p99 = drift.percentile(0.99);
                double budget = 2.0 * targetMillis;
                assertTrue(p99 < budget,
                        "Tick drift p99=" + p99 + "ms exceeded 2× target (" + budget + "ms). "
                                + "Encode+deflate path may be over budget.");
            }

            for (BotClient b : bots) b.disconnect();
        }
    }
    ```

    - Test is tagged `@Tag("performance")` so it can be opt-in via `./gradlew test -PincludeTags=performance` if the full suite is too slow. Default `./gradlew test` SHOULD still include it — Phase 15 closes only when the gate passes.
    - Adjust `BOT_COUNT` / `TARGET_TICKS` down if CI timeout becomes an issue, but do NOT disable the test.
