    ```java
    // OLD
    private void sendRegister(WebSocketSession s, String type) throws Exception {
        Messages.Register r = new Messages.Register(type);
        s.sendMessage(new TextMessage(objectMapper.writeValueAsString(r)));
    }

    // NEW
    private void sendRegister(WebSocketSession s, char type) throws Exception {
        String encoded = PerceptionCodec.encode(new Frame.RegisterFrame(type));
        s.sendMessage(new TextMessage(encoded));
    }
    ```
