    import com.paralife.codec.Frame;
    import com.paralife.codec.PerceptionCodec;
    import org.eclipse.jetty.websocket.api.Callback;
    import org.eclipse.jetty.websocket.api.Session;
    import org.eclipse.jetty.websocket.client.ClientUpgradeRequest;
    import org.eclipse.jetty.websocket.client.WebSocketClient;
    import org.slf4j.Logger;
    import org.slf4j.LoggerFactory;

    import java.net.URI;
    import java.util.Optional;
    import java.util.concurrent.*;
    import java.util.concurrent.atomic.AtomicBoolean;
    import java.util.concurrent.atomic.AtomicInteger;

    public class BotClient {

        private static final Logger log = LoggerFactory.getLogger(BotClient.class);

        private final String serverUri;
        private final char entityType;              // stored as char per SCHEMA §6.1 (C/M/S)
        private final HeuristicBrain brain;
        private final long respawnCooldownMs;
        private final long respawnJitterMs;
        private final AtomicInteger actionCount = new AtomicInteger();
        private final AtomicInteger perceptionCount = new AtomicInteger();
        private final CountDownLatch connectedLatch = new CountDownLatch(1);
        private final CountDownLatch registeredLatch = new CountDownLatch(1);
        private final AtomicBoolean alive = new AtomicBoolean(false);

        private WebSocketClient client;
        private Session session;
        private volatile String entityId;
        private volatile char currentType;

        public BotClient(String serverUri, char entityType, HeuristicBrain brain,
                         long respawnCooldownMs, long respawnJitterMs) {
            this.serverUri = serverUri;
            this.entityType = entityType;
            this.currentType = entityType;
            this.brain = brain;
            this.respawnCooldownMs = respawnCooldownMs;
            this.respawnJitterMs = respawnJitterMs;
        }

        public void connect() throws Exception {
            client = new WebSocketClient();
            client.start();
            ClientUpgradeRequest req = new ClientUpgradeRequest();
            req.addExtensions("permessage-deflate; server_no_context_takeover");

            Endpoint endpoint = new Endpoint();
            session = client.connect(endpoint, URI.create(serverUri), req).get(10, TimeUnit.SECONDS);

            // D-33 client-side enforcement — fail fast if server omitted deflate
            String serverExt = session.getUpgradeResponse().getHeaders().get("Sec-WebSocket-Extensions");
            if (serverExt == null || !serverExt.contains("permessage-deflate")) {
                session.close(1002, "Server did not negotiate permessage-deflate", Callback.NOOP);
                throw new IllegalStateException("permessage-deflate not negotiated by server: " + serverExt);
            }
            connectedLatch.countDown();

            // Send the initial register frame immediately
            sendFrame(new Frame.RegisterFrame(entityType));
        }

        public void disconnect() throws Exception {
            if (session != null && session.isOpen()) {
                session.close(1000, "client disconnect", Callback.NOOP);
            }
            if (client != null) client.stop();
        }

        public boolean isConnected() { return session != null && session.isOpen(); }
        public boolean isRegistered() { return entityId != null; }
        public boolean awaitConnected(long timeoutMs) throws InterruptedException {
            return connectedLatch.await(timeoutMs, TimeUnit.MILLISECONDS);
        }
        public boolean awaitRegistered(long timeoutMs) throws InterruptedException {
            return registeredLatch.await(timeoutMs, TimeUnit.MILLISECONDS);
        }
        public int getActionCount() { return actionCount.get(); }
        public int getPerceptionCount() { return perceptionCount.get(); }

        private synchronized void sendFrame(Frame f) {
            if (session == null || !session.isOpen()) return;
            try {
                session.sendText(PerceptionCodec.encode(f), Callback.NOOP);
            } catch (Exception e) {
                log.warn("Failed to send frame: {}", e.getMessage());
            }
        }

        private void handlePayload(String payload) {
            Frame frame;
            try {
                frame = PerceptionCodec.decode(payload);
            } catch (Exception e) {
                log.warn("Failed to decode frame: {}", e.getMessage());
                return;
            }
            switch (frame) {
                case Frame.SyncFrame s -> onSync(s);
                case Frame.TickFrame t -> onTick(t);
                case Frame.ErrorFrame e -> onError(e);
                case Frame.RegisterFrame ignored -> {} // unexpected; server never sends r
                case Frame.ActionFrame ignored -> {}   // unexpected; server never sends a
            }
        }

        private void onSync(Frame.SyncFrame s) {
            entityId = s.entityId();
            alive.set(true);
            registeredLatch.countDown();
        }

        private void onTick(Frame.TickFrame t) {
            perceptionCount.incrementAndGet();
            // death check: any `D` event triggers respawn flow
            boolean died = t.events().stream().anyMatch(ev -> ev.code() == 'D');
            if (died) {
                handleDeath();
                return;
            }
            Frame.ActionFrame decision = brain.decide(t, currentType, ThreadLocalRandom.current());
            if (decision != null) {
                sendFrame(decision);
                actionCount.incrementAndGet();
            }
            // state-change tracking: update cached type if c block present
            t.change().ifPresent(c -> {
                char code = c.code();
                if (code == 'C' || code == 'M' || code == 'S') currentType = code;
                // Other codes (D/N/T bonded-secondary, 0-5 composite, Z dissolve) also update.
                else if (code == 'Z') currentType = entityType;
                else currentType = code;
            });
        }

        private void onError(Frame.ErrorFrame e) {
            log.warn("Server error {}: {}", e.code(), e.message().orElse(""));
            if (e.code() == 429) {
                // Respawn cap — no more respawns possible; disconnect.
                try { disconnect(); } catch (Exception ignored) {}
            }
        }

        private void handleDeath() {
            alive.set(false);
            entityId = null;
            long jitter = respawnJitterMs > 0
                    ? ThreadLocalRandom.current().nextLong(respawnJitterMs)
                    : 0L;
            long waitMs = respawnCooldownMs + jitter;
            CompletableFuture.delayedExecutor(waitMs, TimeUnit.MILLISECONDS).execute(() -> {
                if (session != null && session.isOpen()) {
                    sendFrame(new Frame.RegisterFrame(entityType));
                }
            });
        }

        // Inner endpoint class bridges Jetty lifecycle to handlePayload.
        private class Endpoint {
            @org.eclipse.jetty.websocket.api.annotations.OnWebSocketMessage
            public void onMessage(String message) {
                handlePayload(message);
            }

            @org.eclipse.jetty.websocket.api.annotations.OnWebSocketClose
            public void onClose(int statusCode, String reason) {
                log.info("WS closed: {} {}", statusCode, reason);
            }

            @org.eclipse.jetty.websocket.api.annotations.OnWebSocketError
            public void onError(Throwable cause) {
                log.warn("WS error: {}", cause.getMessage());
            }
        }
    }
