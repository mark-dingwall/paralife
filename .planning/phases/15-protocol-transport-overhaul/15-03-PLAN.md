    ### `src/main/java/com/paralife/websocket/JettyDeflateCustomizer.java`
    ```java
    package com.paralife.websocket;

    import org.eclipse.jetty.ee10.servlet.ServletContextHandler;
    import org.eclipse.jetty.ee10.websocket.server.JettyWebSocketServletContainerInitializer;
    import org.eclipse.jetty.websocket.core.ExtensionConfig;
    import org.eclipse.jetty.websocket.core.exception.UpgradeException;
    import org.slf4j.Logger;
    import org.slf4j.LoggerFactory;
    import org.springframework.boot.web.embedded.jetty.JettyServletWebServerFactory;
    import org.springframework.boot.web.server.WebServerFactoryCustomizer;
    import org.springframework.context.annotation.Bean;
    import org.springframework.context.annotation.Configuration;

    import java.util.List;

    /**
     * Enforces permessage-deflate; server_no_context_takeover on every WebSocket upgrade
     * at /ws/world. Rejects upgrades that omit the extension (D-33 fail-fast).
     *
     * <p>Jetty 12's native WebSocketCreator is the only hook exposing per-upgrade
     * extension negotiation — Spring's WebSocketHandlerRegistry does not. This
     * customizer runs alongside {@link WebSocketConfig} which still handles Spring's
     * handler-registration side.
     */
    @Configuration
    public class JettyDeflateCustomizer {

        private static final Logger log = LoggerFactory.getLogger(JettyDeflateCustomizer.class);
        private static final String PATH = "/ws/world";
        private static final String EXTENSION = "permessage-deflate";
        private static final String NO_CONTEXT = "server_no_context_takeover";

        @Bean
        public WebServerFactoryCustomizer<JettyServletWebServerFactory> forceDeflate(
                WorldWebSocketHandler handler) {
            return factory -> factory.addServerCustomizers(server -> {
                ServletContextHandler ctx = (ServletContextHandler) server.getHandler();
                JettyWebSocketServletContainerInitializer.configure(ctx, (servletCtx, container) -> {
                    container.addMapping(PATH, (req, resp, callback) -> {
                        List<ExtensionConfig> requested = req.getExtensions();
                        boolean hasDeflate = requested.stream()
                                .anyMatch(e -> EXTENSION.equalsIgnoreCase(e.getName()));
                        if (!hasDeflate) {
                            log.warn("Rejecting upgrade from {} — no permessage-deflate",
                                    req.getRemoteSocketAddress());
                            callback.failed(new UpgradeException(
                                    req.getRequestURI(), 400, "permessage-deflate required"));
                            return null;
                        }
                        ExtensionConfig negotiated = new ExtensionConfig(EXTENSION);
                        negotiated.setParameter(NO_CONTEXT);
                        resp.setExtensions(List.of(negotiated));
                        return new JettyWorldWebSocketAdapter(handler);
                    });
                });
            });
        }
    }
    ```

    ### `src/main/java/com/paralife/websocket/JettyWorldWebSocketAdapter.java`
    ```java
    package com.paralife.websocket;

    import org.eclipse.jetty.websocket.api.Session;
    import org.eclipse.jetty.websocket.api.StatusCode;
    import org.eclipse.jetty.websocket.api.Callback;
    import org.slf4j.Logger;
    import org.slf4j.LoggerFactory;
    import org.springframework.web.socket.CloseStatus;
    import org.springframework.web.socket.TextMessage;
    import org.springframework.web.socket.WebSocketSession;
    import org.springframework.web.socket.adapter.jetty.JettyWebSocketSession;

    import java.nio.ByteBuffer;
    import java.util.Map;
    import java.util.concurrent.ConcurrentHashMap;

    /**
     * Bridges Jetty 12 native WebSocket lifecycle events to the Spring WorldWebSocketHandler.
     * This adapter is constructed per-connection by JettyDeflateCustomizer's WebSocketCreator.
     *
     * <p>Each Jetty Session is wrapped in a Spring-compatible WebSocketSession so downstream
     * code (SessionRegistry, TickBroadcaster, WorldWebSocketHandler) keeps the Spring
     * abstraction it already uses.
     */
    public class JettyWorldWebSocketAdapter {

        private static final Logger log = LoggerFactory.getLogger(JettyWorldWebSocketAdapter.class);
        private final WorldWebSocketHandler handler;
        private final Map<Session, WebSocketSession> wrappers = new ConcurrentHashMap<>();

        public JettyWorldWebSocketAdapter(WorldWebSocketHandler handler) {
            this.handler = handler;
        }

        public void onOpen(Session session) {
            try {
                WebSocketSession wrapped = new JettyWebSocketSession(Map.of());
                // wire the wrapped session — exact JettyWebSocketSession wiring is Jetty-version-specific;
                // the implementer consults Spring's JettyRequestUpgradeStrategy source for the latest idiom.
                wrappers.put(session, wrapped);
                handler.afterConnectionEstablished(wrapped);
            } catch (Exception e) {
                log.error("Failed to bridge open event", e);
            }
        }

        public void onText(Session session, String text) {
            try {
                WebSocketSession wrapped = wrappers.get(session);
                if (wrapped != null) handler.handleMessage(wrapped, new TextMessage(text));
            } catch (Exception e) {
                log.warn("Failed to bridge text message", e);
            }
        }

        public void onClose(Session session, int statusCode, String reason) {
            try {
                WebSocketSession wrapped = wrappers.remove(session);
                if (wrapped != null) handler.afterConnectionClosed(wrapped, new CloseStatus(statusCode, reason));
            } catch (Exception e) {
                log.warn("Failed to bridge close event", e);
            }
        }

        public void onError(Session session, Throwable cause) {
            try {
                WebSocketSession wrapped = wrappers.get(session);
                if (wrapped != null) handler.handleTransportError(wrapped, cause);
            } catch (Exception ignored) {}
        }
    }
    ```

    package com.paralife.websocket;

    import org.junit.jupiter.api.Test;
    import org.springframework.boot.test.context.SpringBootTest;
    import org.springframework.boot.test.web.server.LocalServerPort;
    import org.springframework.test.context.TestPropertySource;

    import java.net.URI;
    import java.net.http.HttpClient;
    import java.net.http.HttpRequest;
    import java.net.http.HttpResponse;
    import java.net.http.HttpResponse.BodyHandlers;

    import static org.junit.jupiter.api.Assertions.assertEquals;
    import static org.junit.jupiter.api.Assertions.assertTrue;

    @SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
    @TestPropertySource(properties = {
            "paralife.tick.auto-start=false",
            "paralife.world.width=16",
            "paralife.world.height=16"
    })
    class WebSocketDeflateHandshakeIntegrationTest {

        @LocalServerPort
        private int port;

        @Test
        void serverNegotiatesPermessageDeflateWithNoContextTakeover() throws Exception {
            HttpClient http = HttpClient.newHttpClient();
            HttpRequest upgrade = HttpRequest.newBuilder(
                            URI.create("http://localhost:" + port + "/ws/world"))
                    .header("Connection", "Upgrade")
                    .header("Upgrade", "websocket")
                    .header("Sec-WebSocket-Version", "13")
                    .header("Sec-WebSocket-Key", "dGhlIHNhbXBsZSBub25jZQ==")
                    .header("Sec-WebSocket-Extensions",
                            "permessage-deflate; server_no_context_takeover")
                    .GET().build();

            HttpResponse<Void> resp = http.send(upgrade, BodyHandlers.discarding());

            assertEquals(101, resp.statusCode(), "Server should complete upgrade");
            String ext = resp.headers().firstValue("Sec-WebSocket-Extensions")
                    .orElseThrow(() -> new AssertionError("No Sec-WebSocket-Extensions response header"));
            assertTrue(ext.contains("permessage-deflate"),
                    "Response extensions missing permessage-deflate: " + ext);
            assertTrue(ext.contains("server_no_context_takeover"),
                    "Response extensions missing server_no_context_takeover: " + ext);
        }
    }
