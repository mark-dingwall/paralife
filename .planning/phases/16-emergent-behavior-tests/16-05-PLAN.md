private void resetAllSeedsBetweenRuns() {
    // Order matters:
    // 1. Reset registries that track live entities
    worldGrid.clear();
    compositeRegistry.clear();
    buffRegistry.clear();
    botRegistry.clear();
    environmentEngine.resetForTest();
    deathFinalizer.resetCountForTest();

    // 2. Reset seeded Random state in every bean (REVIEWS HIGH #1 core fix)
    simulationEngine.resetSeed();
    actionResolver.resetSeed();
    compositeEnergyDistributor.resetSeed();
    fertilityInitializer.resetSeed();   // Re-runs seedPatches so fertility grid re-initialises
                                         // — @PostConstruct-only would not fire on a 2nd/3rd iteration

    // VERIFIED pre-revision (WARNING fix): EnvironmentEngine.resetForTest() at lines 1163-1164
    // already re-seeds its internal rng from config.seed() (null→0L). No EnvironmentEngine
    // change is required in this plan. The reassignment `this.rng = new Random(seed)` inside
    // resetForTest() produces byte-identical Poisson roll sequences across runs (cf.

private record RunObservables(int compositeCount, double bondedPairCounterDelta, double compositeCounterDelta) {}

private RunObservables driveRun(int ticks) {
    double bondBefore = emergenceMetrics.bondedPairsFormed();
    double compositeBefore = emergenceMetrics.compositesFormed();

    seedDeterministicScenario();

    for (long t = 1; t <= ticks; t++) {
        publisher.publishEvent(new TickEvent(t));
    }

    return new RunObservables(
            compositeRegistry.size(),
            emergenceMetrics.bondedPairsFormed() - bondBefore,
            emergenceMetrics.compositesFormed() - compositeBefore);
}

private void seedDeterministicScenario() {
    // FIXED placement — identical input state every run. Seeded Random fields
    // deterministically pick which pairs bond + which form composites.
    int[][] pairPositions = {

@Test
@Order(1)
void compositeFormationIsDeterministicAcrossThreeRunsWithSameSeed() {
    List<RunObservables> observations = new ArrayList<>();
    int runs = 3;
    int ticks = 200;
    for (int r = 0; r < runs; r++) {
        resetAllSeedsBetweenRuns();       // REVIEWS HIGH #1: explicit reset, not DirtiesContext
        observations.add(driveRun(ticks));
    }

    Set<Integer> uniqueCompositeCounts = new HashSet<>();
    for (RunObservables o : observations) uniqueCompositeCounts.add(o.compositeCount());

    assertThat(uniqueCompositeCounts)
        .as("seed=42 — all %d runs must produce identical compositeRegistry.size() after explicit resetSeed()", runs)
        .hasSize(1);

    assertThat(observations.get(0).compositeCount())
        .as("If 0, seedDeterministicScenario under-tuned or forced-composite config too weak")
        .isGreaterThan(0);

    seed42CompositeCount.set(observations.get(0).compositeCount());
}

@Test
@Order(2)
void seededRunProducesAtLeastOneComposite() {
    resetAllSeedsBetweenRuns();
    RunObservables obs = driveRun(200);
    assertThat(obs.compositeCount()).isGreaterThan(0);
}
