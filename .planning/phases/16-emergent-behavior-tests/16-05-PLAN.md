Build R15: `CompositeFormationDeterminismTest` — the engine-direct seeded test proving composite formation is reproducible given a fixed master seed. Runs with `@SpringBootTest` (no web env per D-17), `@DirtiesContext(classMode = BEFORE_EACH_TEST_METHOD)` so each `@Test` method gets a fresh Spring context (and therefore fresh per-component `Random` instances derived from the master seed), manual tick stepping via `ApplicationEventPublisher.publishEvent(new TickEvent(n))`, forced-formation `@TestPropertySource` knobs from RESEARCH §Forced-Formation Config.

**Why `@DirtiesContext` is load-bearing:** The `Random rng` fields on `SimulationEngine`, `ActionResolver`, `CompositeEnergyDistributor` are assigned exactly once — at Spring bean construction — via `source.forComponent(tag)`. Once 200 ticks have advanced that `Random`'s internal state, a second run in the same context starts from the advanced state, not the seed. Two consecutive runs in one `@Test` method therefore produce DIFFERENT counter deltas even with identical `paralife.seed.master=42`. `@DirtiesContext(classMode = BEFORE_EACH_TEST_METHOD)` forces Spring to rebuild the context between methods, which re-runs the RandomSource constructor and produces fresh-but-identical `Random` instances seeded from the same master.

Purpose: Closes R15 ("Deterministic seed test for composite formation"). Uses the three Wave 1 infra pieces (EmergenceMetrics counters, RandomSource master seed, TickWorkTimer not required but present on the pipeline) + the Wave 2 helper (PopulationHistory only — TriggerWatcher/fixture/log-capture aren't needed for R15 per CONTEXT).

With `BEFORE_EACH_TEST_METHOD`, Spring rebuilds the `ApplicationContext` before every `@Test` method. ALL beans are re-instantiated — including `RandomSource`, which re-reads `paralife.seed.master=42` and hands fresh per-component `Random` instances to the engines.     package com.paralife.engine;

    ```java
    import static org.assertj.core.api.Assertions.assertThat;

    import com.paralife.metrics.EmergenceMetrics;
    import io.micrometer.core.instrument.MeterRegistry;
    import org.junit.jupiter.api.BeforeEach;
    import org.junit.jupiter.api.Test;
    import org.slf4j.Logger;
    import org.slf4j.LoggerFactory;
    import org.springframework.beans.factory.annotation.Autowired;
    import org.springframework.boot.test.context.SpringBootTest;
    import org.springframework.context.ApplicationEventPublisher;
    import org.springframework.test.annotation.DirtiesContext;
    import org.springframework.test.context.TestPropertySource;

    /**
     * Phase 16 R15: Deterministic seed test for composite formation.
     * Engine-direct per D-17 (@SpringBootTest with no web environment).
     * Manual tick stepping via ApplicationEventPublisher per D-18 inverse.
     *
     * <p><strong>@DirtiesContext(BEFORE_EACH_TEST_METHOD)</strong> is required:
     * RandomSource hands out per-component {@link java.util.Random} instances at
     * bean construction via {@code source.forComponent(tag)}. Once those instances
     * have produced draws in one @Test method, their internal state is advanced;
     * a second @Test method reusing the same context would see different counter
     * values. Rebuilding the context per method re-derives the per-component
     * Random instances from the same master seed ({@code paralife.seed.master=42}),
     * guaranteeing byte-equal counter deltas across methods.
     */
    @SpringBootTest
    @DirtiesContext(classMode = DirtiesContext.ClassMode.BEFORE_EACH_TEST_METHOD)
    @TestPropertySource(properties = {
        "paralife.tick.auto-start=false",
        "paralife.seed.master=42",
        "paralife.simulation.events.seed=42",
        "paralife.world.rock.seed=42",
        "paralife.world.width=32",
        "paralife.world.height=32",
        // Forced-formation knobs per RESEARCH §Forced-Formation Config
        "paralife.bonding.bond-energy-threshold=15",
        "paralife.bonding.bonding-probability=0.5",
        "paralife.composite.can-form-composites=true",
        "paralife.composite.dissolution-chance=0.005",
        "paralife.composite.critical-energy-percent=5",
        "paralife.simulation.nutrient-spawn-probability=0.01",
    })
    class CompositeFormationDeterminismTest {

        private static final Logger log = LoggerFactory.getLogger(CompositeFormationDeterminismTest.class);
        private static final long MASTER_SEED = 42L;
        private static final int TICKS = 200;

        @Autowired ApplicationEventPublisher publisher;
        @Autowired MeterRegistry meterRegistry;
        @Autowired com.paralife.world.WorldGrid worldGrid;
        @Autowired com.paralife.engine.CompositeRegistry compositeRegistry;
        @Autowired com.paralife.engine.BotRegistry botRegistry;
        @Autowired com.paralife.engine.BuffRegistry buffRegistry;
        @Autowired com.paralife.engine.EnvironmentEngine environmentEngine;
        @Autowired com.paralife.engine.DeathFinalizer deathFinalizer;

        @BeforeEach
        void logSeed() {
            log.info("R15 determinism test: master seed = {}", MASTER_SEED);
        }

        /**
         * Reset world + registries to a known empty state. Does NOT reset the
         * Spring context or the per-component Random instances — those are
         * refreshed by @DirtiesContext(BEFORE_EACH_TEST_METHOD) between methods.
         */
        private void resetWorld() {
            worldGrid.clear();
            botRegistry.clear();
            compositeRegistry.clear();
            buffRegistry.clear();
            environmentEngine.resetForTest();
            deathFinalizer.resetCountForTest();
            // Note: do NOT clear MeterRegistry — we read DELTAS around each run (RESEARCH Open Q #3)
        }

        @Test
        void sameSeedYieldsSameCompositeCount() {
            // Run 1
            resetWorld();
            long[] run1 = runOnce();
            log.info("Run 1: bonded={} composites={}", run1[0], run1[1]);

            // Run 2 — same context (same Random instances, now advanced),
            // reset world state only. Deltas should match because starting
            // world is byte-identical and tick loop is deterministic.
            resetWorld();
            long[] run2 = runOnce();
            log.info("Run 2: bonded={} composites={}", run2[0], run2[1]);

            // Run 3 — third invocation, same context + reset world.
            resetWorld();
            long[] run3 = runOnce();
            log.info("Run 3: bonded={} composites={}", run3[0], run3[1]);

            // Three-way equality per checker-revision requirement
            assertThat(run2[0]).as("bonded-pair delta run2 == run1").isEqualTo(run1[0]);
            assertThat(run3[0]).as("bonded-pair delta run3 == run1").isEqualTo(run1[0]);
            assertThat(run2[1]).as("composites delta run2 == run1").isEqualTo(run1[1]);
            assertThat(run3[1]).as("composites delta run3 == run1").isEqualTo(run1[1]);
        }

        @Test
        void compositeFormsFromSeededBonding() {
            resetWorld();
            long[] run = runOnce();
            assertThat(run[0]).as("bonded pairs formed under forced-formation knobs @seed=42, 200 ticks")
                    .isGreaterThan(0L);
            assertThat(run[1]).as("composites formed under forced-formation knobs @seed=42, 200 ticks")
                    .isGreaterThan(0L);
        }

        // ── helpers ─────────────────────────────────────────────────

        /** @return {bondedDelta, compositesDelta} for one 200-tick run. */
        private long[] runOnce() {
            long bondedBefore     = readCounter(EmergenceMetrics.M_BONDED_PAIRS_FORMED);
            long compositesBefore = readCounter(EmergenceMetrics.M_COMPOSITES_FORMED);
            plantParticles();
            for (long t = 1; t <= TICKS; t++) publisher.publishEvent(new TickEvent(t));
            long bondedAfter     = readCounter(EmergenceMetrics.M_BONDED_PAIRS_FORMED);
            long compositesAfter = readCounter(EmergenceMetrics.M_COMPOSITES_FORMED);
            return new long[] { bondedAfter - bondedBefore, compositesAfter - compositesBefore };
        }

        private long readCounter(String name) {
            var m = meterRegistry.find(name).counter();
            return m == null ? 0L : (long) m.count();
        }

        /**
         * Plant ~20 Particles at deterministic positions on the 32x32 world.
         * Use a mix of CATALYST/MEMBRANE/SPORE laid out in predator-prey adjacencies
         * to give forced-formation knobs a fair shot at producing bonds + composites.
         */
        private void plantParticles() {
            com.paralife.world.Entity.ParticleType[] types = {
                com.paralife.world.Entity.ParticleType.CATALYST,
                com.paralife.world.Entity.ParticleType.MEMBRANE,
                com.paralife.world.Entity.ParticleType.SPORE
            };
            int id = 0;
            for (int row = 0; row < 4; row++) {
                for (int col = 0; col < 5; col++) {
                    int x = 5 + col * 5;
                    int y = 5 + row * 5;
                    var type = types[(row + col) % 3];
                    var p = new com.paralife.world.Entity.Particle(
                            "bot-r15-" + (id++), type, 60 /* starting energy */);
                    worldGrid.setEntity(x, y, p);
                }
            }
        }
    }
    ```
