    ```java
    package com.paralife.engine;

    import java.util.Random;
    import org.springframework.beans.factory.annotation.Value;
    import org.springframework.stereotype.Component;

    /**
     * Phase 16 D-09b: central master-seed factory for reproducible statistical tests.
     *
     * <p>Production default ({@code paralife.seed.master=-1}) falls through to
     * {@code new Random()} — identical unseeded behavior to today. Tests set the
     * property via {@code @TestPropertySource} to get deterministic per-component streams.
     *
     * <p><strong>Not cryptographic.</strong> This is {@link java.util.Random} for
     * statistical reproducibility only. NEVER route security-sensitive draws through
     * this source; use {@link java.security.SecureRandom} for tokens, nonces, or keys.
     */
    @Component
    public class RandomSource {

        private final long masterSeed;

        public RandomSource(@Value("${paralife.seed.master:-1}") long masterSeed) {
            this.masterSeed = masterSeed;
        }

        /**
         * Derive a deterministic sub-seed: {@code masterSeed XOR tag.hashCode()}.
         * Returns an unseeded {@link Random} when master is {@code -1} (production default).
         *
         * @param tag stable component identifier (e.g. "simulation", "action", "ced")
         */
        public Random forComponent(String tag) {
            if (masterSeed == -1L) {
                return new Random();
            }
            return new Random(masterSeed ^ (long) tag.hashCode());
        }

        /** @return the master seed (for logging / reproduction) */
        public long masterSeed() { return masterSeed; }
    }
    ```
