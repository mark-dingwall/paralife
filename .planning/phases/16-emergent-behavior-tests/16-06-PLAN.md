    package com.paralife.engine;

    import static org.assertj.core.api.Assertions.assertThat;

    import com.paralife.bot.BotClient;
    import com.paralife.engine.emergence.PopulationHistory;
    import com.paralife.engine.emergence.SeededBotLauncher;
    import com.paralife.engine.emergence.TestLogCapture;
    import com.paralife.engine.emergence.TriggerWatcher;
    import com.paralife.metrics.EmergenceMetrics;
    import com.paralife.world.Cell;
    import com.paralife.world.Entity;
    import com.paralife.world.Position;
    import com.paralife.world.WorldGrid;
    import com.paralife.world.WorldSnapshot;
    import java.net.URI;
    import java.util.List;
    import java.util.Map;
    import org.junit.jupiter.api.AfterEach;
    import org.junit.jupiter.api.Tag;
    import org.junit.jupiter.api.Test;
    import org.slf4j.Logger;
    import org.slf4j.LoggerFactory;
    import org.springframework.beans.factory.annotation.Autowired;

    import org.springframework.boot.test.web.server.LocalServerPort;
    import org.springframework.test.annotation.DirtiesContext;

    /**
     * Phase 16 R16 + R17 + R18: long-run full-stack seeded test.
     * Harness + sampling loop only — assertions land in the companion plan 16-07.
     */
    @SpringBootTest(webEnvironment = SpringBootTest.WebEnvironment.RANDOM_PORT)
    @DirtiesContext(classMode = DirtiesContext.ClassMode.AFTER_CLASS)  // Pitfall 2 mitigation
    @TestPropertySource(properties = {
        "paralife.tick.interval-ms=20",
        "paralife.tick.auto-start=true",
        "paralife.world.width=64",
        "paralife.world.height=64",
        "paralife.seed.master=42",

        "paralife.composite.dissolution-chance=0.005",

        "paralife.simulation.nutrient-spawn-probability=0.01",

    })
    @Tag("slow")  // Per D-08 Claude's Discretion
    class EmergenceStabilityLoadTest {

        private static final Logger log = LoggerFactory.getLogger(EmergenceStabilityLoadTest.class);
        private static final long MASTER_SEED = 42L;
        private static final int BOT_COUNT = 100;
        private static final int TOTAL_TICKS = 1000;
        private static final int WARMUP_TICKS = 200;
        private static final int TRIGGER_WINDOW = 30;
        private static final int TRIGGER_RADIUS = 5;

        @LocalServerPort int port;
        @Autowired WorldGrid worldGrid;
        @Autowired BuffRegistry buffRegistry;
        @Autowired TickEngine tickEngine;
        @Autowired io.micrometer.core.instrument.MeterRegistry meterRegistry;
        @Autowired TickConfig tickConfig;

        private final SeededBotLauncher launcher = new SeededBotLauncher();

        @AfterEach
        void tearDown() {
            launcher.shutdown();
        }

        @Test
        void thousandTickStabilityAndEmergence() throws Exception {
            log.info("R16/R17/R18 long-run test: master seed = {}", MASTER_SEED);

            // 1. Launch 100 seeded bots
            URI uri = URI.create("ws://localhost:" + port + "/ws/world");
            List<BotClient> bots = launcher.launchSeeded(uri, BOT_COUNT, MASTER_SEED,
                    tickConfig.intervalMs(), 50L /* respawnJitterMs */);
            assertThat(bots).hasSize(BOT_COUNT);

            // 2. Attach log capture
            TestLogCapture logCapture = TestLogCapture.attach();

            // 3. Wait through warmup
            waitForTick(WARMUP_TICKS);
            log.info("warmup complete at tick {}", tickEngine.getCurrentTick());

            // 4. Per-tick sampling loop
            PopulationHistory history = new PopulationHistory();
            TriggerWatcher predPressure = new TriggerWatcher(TRIGGER_WINDOW, TRIGGER_RADIUS);  // D-04 #3
            TriggerWatcher fleeFrom     = new TriggerWatcher(TRIGGER_WINDOW, TRIGGER_RADIUS);  // D-04 #5

            long lastProcessedTick = WARMUP_TICKS;
            while (tickEngine.getCurrentTick() < TOTAL_TICKS) {
                Thread.sleep(tickConfig.intervalMs() * 5);  // poll ~every 5 ticks
                long now = tickEngine.getCurrentTick();
                for (long t = lastProcessedTick + 1; t <= now; t++) {
                    WorldSnapshot snap = worldGrid.snapshot();
                    history.append(t, PopulationHistory.countPopulation(snap));

                    // Discover triggers this tick + observe open windows
                    scanAndTriggerStarving(snap, t, predPressure);
                    scanAndTriggerStrongPredators(snap, t, fleeFrom);
                    observeSignal3(predPressure, snap, t);
                    observeSignal5(fleeFrom, snap, t);
                }
                lastProcessedTick = now;
            }

            log.info("Sampling complete: history size={}, pred-windows-closed={}, flee-windows-closed={}",
                    history.size(), predPressure.windowCount(), fleeFrom.windowCount());

            // 5. Detach log capture — Plan 16-07 reads capture.errorCount() / countWithPrefix("EMERGENCE ")
            logCapture.detach();

            // 6. PLACEHOLDER ASSERTION — Plan 16-07 adds the full D-07/D-11/D-04 assertion set
            assertThat(history.size()).as("per-tick samples collected across the run").isGreaterThan(900);
        }

        // ── helpers ─────────────────────────────────────────────────

        private void waitForTick(long target) throws InterruptedException {
            long deadline = System.currentTimeMillis() + 60_000;
            while (tickEngine.getCurrentTick() < target) {
                if (System.currentTimeMillis() > deadline) {
                    throw new AssertionError("tick " + target + " not reached within 60s (currentTick="
                            + tickEngine.getCurrentTick() + ")");
                }
                Thread.sleep(tickConfig.intervalMs());
            }
        }

        /**
         * D-04 #3 trigger: every tick, scan for STARVING prey and open a window.
         * Per RESEARCH §Trigger-Watcher #3 semantics.
         */
        private void scanAndTriggerStarving(WorldSnapshot snap, long tick, TriggerWatcher w) {
            for (int x = 0; x < snap.width(); x++) {
                for (int y = 0; y < snap.height(); y++) {
                    Cell c = snap.getCell(x, y);
                    if (!c.hasFlag(Cell.FLAG_STARVING)) continue;
                    if (c.occupant() instanceof Entity.Particle p) {
                        w.trigger("starving-" + p.id() + "-" + tick, new Position(x, y), tick);
                    }
                }
            }
        }

        /** D-04 #5 trigger: buffed particle OR BondedPair occupant opens a window. */
        private void scanAndTriggerStrongPredators(WorldSnapshot snap, long tick, TriggerWatcher w) {
            for (int x = 0; x < snap.width(); x++) {
                for (int y = 0; y < snap.height(); y++) {
                    Cell c = snap.getCell(x, y);
                    Entity occ = c.occupant();
                    if (occ instanceof Entity.Particle p
                            && !buffRegistry.getBuffs(p.id()).isEmpty()) {
                        w.trigger("strong-" + p.id() + "-" + tick, new Position(x, y), tick);
                    } else if (occ instanceof Entity.BondedPair bp) {
                        w.trigger("strong-" + bp.id() + "-" + tick, new Position(x, y), tick);
                    }
                }
            }
        }

        /** D-04 #3 per-tick sampler — predator density within radius. */
        private void observeSignal3(TriggerWatcher w, WorldSnapshot snap, long tick) {
            w.observe(tick, snap, window -> countPredatorsWithinRadius(snap, window.startPos(), w.radius()));
        }

        /** D-04 #5 per-tick sampler — weaker-type-prey density within radius. */
        private void observeSignal5(TriggerWatcher w, WorldSnapshot snap, long tick) {
            w.observe(tick, snap, window -> countAnyParticlesWithinRadius(snap, window.startPos(), w.radius()));
        }

        private int countPredatorsWithinRadius(WorldSnapshot snap, Position center, int radius) {
            int count = 0;
            for (int dx = -radius; dx <= radius; dx++) {
                for (int dy = -radius; dy <= radius; dy++) {
                    int x = Math.floorMod(center.x() + dx, snap.width());
                    int y = Math.floorMod(center.y() + dy, snap.height());
                    if (snap.getCell(x, y).occupant() instanceof Entity.Particle) count++;
                }
            }
            return count;
        }

        private int countAnyParticlesWithinRadius(WorldSnapshot snap, Position center, int radius) {
            return countPredatorsWithinRadius(snap, center, radius);  // same density signal — Plan 16-07 refines
        }
    }
