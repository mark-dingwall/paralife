    ```java
    package com.paralife.engine;

    import io.micrometer.core.instrument.MeterRegistry;
    import io.micrometer.core.instrument.Timer;
    import java.util.concurrent.TimeUnit;
    import java.util.concurrent.atomic.AtomicLong;
    import org.springframework.context.event.EventListener;
    import org.springframework.core.annotation.Order;
    import org.springframework.stereotype.Component;

    /**
     * Phase 16 D-11: measure wall-clock time from TickEvent dispatch start
     * (BEFORE SimulationEngine @Order(10)) to final listener completion
     * (AFTER TickBroadcaster @Order(100)).
     *
     * <p>Single-threaded tick pipeline (CLAUDE.md) means the two listeners
     * execute on the same publisher thread (virtual-thread tick-engine loop);
     * the AtomicLong is defensive single-writer-single-reader.
     */
    @Component
    public class TickWorkTimer {

        public static final String M_TICK_WORK = "paralife.tick.work";

        private final Timer timer;
        private final AtomicLong tickStartNanos = new AtomicLong();

        public TickWorkTimer(MeterRegistry registry) {
            this.timer = Timer.builder(M_TICK_WORK)
                    .description("Wall-clock time from TickEvent dispatch start to final listener completion")
                    .publishPercentiles(0.5, 0.95, 0.99)
                    .register(registry);
        }

        @EventListener
        @Order(0)                       // BEFORE SimulationEngine @Order(10)
        public void onTickStart(TickEvent event) {
            tickStartNanos.set(System.nanoTime());
        }

        @EventListener
        @Order(101)                     // AFTER TickBroadcaster @Order(100)
        public void onTickEnd(TickEvent event) {
            long elapsed = System.nanoTime() - tickStartNanos.get();
            timer.record(elapsed, TimeUnit.NANOSECONDS);
        }
    }
    ```
