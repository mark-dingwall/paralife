    ```java
    package com.paralife.engine.emergence;

    import com.paralife.world.Position;
    import com.paralife.world.WorldSnapshot;
    import java.util.ArrayList;
    import java.util.List;
    import java.util.function.ToIntFunction;

    /**
     * Phase 16 D-04 #3 / #5: per-entity observation window.
     * trigger() opens a window; observe() samples while any window is active;
     * at endTick each window closes and its mean sample is available.
     */
    public final class TriggerWatcher {

        public record Window(String entityId, Position startPos,
                             long startTick, long endTick, List<Integer> samples) {}

        private final int windowTicks;
        private final int radius;
        private final List<Window> active = new ArrayList<>();
        private final List<Window> closed = new ArrayList<>();

        public TriggerWatcher(int windowTicks, int radius) {
            this.windowTicks = windowTicks;
            this.radius = radius;
        }

        public int radius() { return radius; }

        public void trigger(String entityId, Position pos, long tick) {
            active.add(new Window(entityId, pos, tick, tick + windowTicks, new ArrayList<>()));
        }

        /**
         * Per-tick sampler callback.
         * @param sampler receives (window) and returns an integer count relevant to the signal.
         */
        public void observe(long tick, WorldSnapshot snap, ToIntFunction<Window> sampler) {
            List<Window> toClose = new ArrayList<>();
            for (Window w : active) {
                if (tick > w.endTick()) { toClose.add(w); continue; }
                if (tick >= w.startTick()) {
                    w.samples().add(sampler.applyAsInt(w));
                }
            }
            for (Window w : toClose) {
                active.remove(w);
                closed.add(w);
            }
        }

        public int windowCount() { return closed.size(); }

        public double meanSampleAcrossClosedWindows() {
            if (closed.isEmpty()) return 0.0;
            double sum = 0.0; int n = 0;
            for (Window w : closed) for (int s : w.samples()) { sum += s; n++; }
            return n == 0 ? 0.0 : sum / n;
        }

        public List<Window> closedWindows() { return List.copyOf(closed); }
    }
    ```

    package com.paralife.engine.emergence;

    import com.fasterxml.jackson.databind.ObjectMapper;
    import java.io.IOException;
    import java.nio.file.Files;
    import java.nio.file.Path;
    import java.time.Instant;
    import java.util.Comparator;
    import java.util.List;
    import java.util.Map;
    import java.util.stream.Stream;

    /**
     * Phase 16 D-06b: per-run fixture JSON writer with N=5 rollover.
     * Directory gitignored (Plan 16-03 Task 2); .gitkeep preserved.
     */
    public final class RunFixtureWriter {

        private static final int KEEP = 5;
        private static final ObjectMapper MAPPER = new ObjectMapper();

        private final Path fixturesDir;

        public RunFixtureWriter(Path fixturesDir) throws IOException {
            this.fixturesDir = fixturesDir;
            Files.createDirectories(fixturesDir);
        }

        public Path write(long masterSeed, int totalTicks,
                          Map<String, Object> config,
                          Map<String, Long> counters,
                          List<Map<String, Object>> perTickSamples) throws IOException {
            prune();
            String ts = Instant.now().toString().replace(':', '-');
            Path out = fixturesDir.resolve("run-" + ts + ".json");
            Map<String, Object> doc = Map.of(
                    "masterSeed", masterSeed,
                    "startedAt", ts,
                    "totalTicks", totalTicks,
                    "config", config,
                    "counters", counters,
                    "ticks", perTickSamples);
            MAPPER.writerWithDefaultPrettyPrinter().writeValue(out.toFile(), doc);
            return out;
        }

        /** Delete oldest fixtures beyond KEEP-1 (making room for the next write). */
        private void prune() throws IOException {
            try (Stream<Path> all = Files.list(fixturesDir)) {
                List<Path> jsons = all.filter(p -> p.getFileName().toString().endsWith(".json"))
                        .sorted(Comparator.comparing(p -> {
                            try { return Files.getLastModifiedTime(p); }
                            catch (IOException e) { throw new RuntimeException(e); }
                        }))
                        .toList();
                int excess = jsons.size() - (KEEP - 1);
                for (int i = 0; i < excess; i++) Files.deleteIfExists(jsons.get(i));
            }
        }
    }
    ```

    import ch.qos.logback.classic.Level;
    import ch.qos.logback.classic.Logger;
    import ch.qos.logback.classic.spi.ILoggingEvent;
    import ch.qos.logback.core.read.ListAppender;
    import org.slf4j.LoggerFactory;

    /**
     * Phase 16 D-11 / D-15: attach a Logback ListAppender to the ROOT logger
     * to assert zero ERROR entries and to count EMERGENCE markers.
     */
    public final class TestLogCapture {

        private final ListAppender<ILoggingEvent> appender;

        private TestLogCapture(ListAppender<ILoggingEvent> appender) { this.appender = appender; }

        public static TestLogCapture attach() {
            Logger root = (Logger) LoggerFactory.getLogger(Logger.ROOT_LOGGER_NAME);
            ListAppender<ILoggingEvent> a = new ListAppender<>();
            a.start();
            root.addAppender(a);
            return new TestLogCapture(a);
        }

        public void detach() {
            Logger root = (Logger) LoggerFactory.getLogger(Logger.ROOT_LOGGER_NAME);
            root.detachAppender(appender);
            appender.stop();
        }

        public long errorCount() {
            return appender.list.stream().filter(e -> e.getLevel() == Level.ERROR).count();
        }

        public long countWithPrefix(String prefix) {
            return appender.list.stream().filter(e -> e.getFormattedMessage().startsWith(prefix)).count();
        }
    }

    import com.paralife.bot.BotClient;
    import com.paralife.bot.HeuristicBrain;
    import java.net.URI;
    import java.util.ArrayList;
    import java.util.List;
    import java.util.Random;
    import java.util.concurrent.CountDownLatch;
    import java.util.concurrent.Executors;
    import java.util.concurrent.TimeUnit;

    /**
     * Phase 16 D-09a seeded variant of BotLauncher — gives each BotClient a
     * deterministic per-bot Random(masterSeed ^ botIndex).
     */
    public final class SeededBotLauncher {

        private final List<BotClient> bots = new ArrayList<>();

        public List<BotClient> launchSeeded(URI uri, int count, long masterSeed,
                                             long pollMs, long respawnJitterMs) throws Exception {
            CountDownLatch registered = new CountDownLatch(count);
            char[] species = {'C', 'M', 'S'};
            var exec = Executors.newVirtualThreadPerTaskExecutor();
            for (int i = 0; i < count; i++) {
                final int idx = i;
                char sp = species[i % 3];
                Random rng = new Random(masterSeed ^ (long) idx);
                exec.submit(() -> {
                    try {
                        BotClient b = new BotClient(uri, sp, new HeuristicBrain(), pollMs, respawnJitterMs, rng);
                        synchronized (bots) { bots.add(b); }
                        b.waitForRegistered(30, TimeUnit.SECONDS);
                    } catch (Exception e) { /* surface via test timeout */ }
                    finally { registered.countDown(); }
                });
            }
            registered.await(60, TimeUnit.SECONDS);
            exec.shutdown();
            return List.copyOf(bots);
        }

        public void shutdown() {
            synchronized (bots) { for (BotClient b : bots) try { b.close(); } catch (Exception ignored) {} }
        }
    }
    ```
