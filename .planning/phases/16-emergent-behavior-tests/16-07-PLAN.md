    // ── Heap measurement per RESEARCH §Heap-Measurement Pick ─
    private long heapUsedBytesAveraged(int samples) {
        long sum = 0;
        for (int i = 0; i < samples; i++) {
            System.gc();  // hint, not guarantee
            try { Thread.sleep(50); } catch (InterruptedException ignored) {}
            Runtime rt = Runtime.getRuntime();
            sum += rt.totalMemory() - rt.freeMemory();
        }
        return sum / samples;
    }

    // ── D-07 #2 Per-type floor coverage ──────────────────────
    private static void assertPerTypeFloorCoverage(PopulationHistory history, double floorShare, double coverageRatio) {
        String[] types = {"CATALYST", "MEMBRANE", "SPORE"};
        for (String t : types) {
            int qualifying = 0;
            for (Map<String,Integer> s : history.samples()) {
                int total = s.values().stream().mapToInt(Integer::intValue).sum();
                double share = total == 0 ? 0.0 : s.getOrDefault(t, 0) / (double) total;
                if (share >= floorShare) qualifying++;
            }
            double observedCoverage = qualifying / (double) history.size();
            assertThat(observedCoverage)
                    .as("D-07 #2 %s share >= %.2f for >= %.2f of ticks (observed %.2f)",
                            t, floorShare, coverageRatio, observedCoverage)
                    .isGreaterThanOrEqualTo(coverageRatio);
        }
    }

    // ── D-07 #3 Oscillation amplitude ────────────────────────
    private static void assertOscillationAmplitude(PopulationHistory history, int windowSize, double floor) {
        String[] types = {"CATALYST", "MEMBRANE", "SPORE"};
        boolean any = false;
        double bestObserved = 0.0;
        String bestType = "?";
        for (String t : types) {
            for (int i = 0; i + windowSize <= history.size(); i++) {
                int min = Integer.MAX_VALUE, max = Integer.MIN_VALUE, sum = 0;
                for (int j = i; j < i + windowSize; j++) {
                    int v = history.samples().get(j).getOrDefault(t, 0);
                    min = Math.min(min, v); max = Math.max(max, v); sum += v;
                }
                double mean = sum / (double) windowSize;
                double amp = mean == 0 ? 0.0 : (max - min) / mean;
                if (amp > bestObserved) { bestObserved = amp; bestType = t; }
                if (amp >= floor) { any = true; }
            }
        }
        assertThat(any)
                .as("D-07 #3 oscillation amplitude (max-min)/mean >= %.2f over %d-sample window for some type (best observed: %s @ %.3f)",
                        floor, windowSize, bestType, bestObserved)
                .isTrue();
    }

    // ── D-04 #3 baseline density (non-bonded Particle density across the run) ─
    private double baselineDensity(PopulationHistory history) {
        int width = 64, height = 64;  // Matches @TestPropertySource
        double sum = 0;
        for (Map<String,Integer> s : history.samples()) {
            int total = s.values().stream().mapToInt(Integer::intValue).sum();
            sum += total / (double) (width * height);
        }
        double mean = sum / history.size();
        // Predator density baseline ≈ total / 3 (one-third of types is predator of any given type)
        // over radius 5: area = (2*5+1)^2 = 121 cells
        return mean * (121.0 / 3.0);
    }

    // ── D-04 #4 RPS boom-bust — weak autocorrelation check ───
    private static void assertRpsBoomBustPattern(PopulationHistory history) {
        String[] types = {"CATALYST", "MEMBRANE", "SPORE"};
        int anyTypeChanges = 0;
        for (String t : types) {
            int last = history.samples().get(0).getOrDefault(t, 0);
            int lastDeltaSign = 0;
            int changes = 0;
            for (int i = 1; i < history.size(); i++) {
                int curr = history.samples().get(i).getOrDefault(t, 0);
                int delta = curr - last;
                int sign = delta == 0 ? lastDeltaSign : (delta > 0 ? 1 : -1);
                if (sign != 0 && lastDeltaSign != 0 && sign != lastDeltaSign) changes++;
                lastDeltaSign = sign == 0 ? lastDeltaSign : sign;
                last = curr;
            }
            if (changes >= 2) anyTypeChanges++;
        }
        assertThat(anyTypeChanges)
                .as("D-04 #4 RPS boom-bust: at least one of 3 types shows >=2 direction changes (pattern exists)")
                .isGreaterThanOrEqualTo(1);
    }

    // ── D-04 #5 Flee-from declining density ──────────────────
    private static void assertFleeDensityDeclines(TriggerWatcher fleeFrom) {
        int decliningCount = 0;
        int totalWindows = fleeFrom.closedWindows().size();
        for (TriggerWatcher.Window w : fleeFrom.closedWindows()) {
            if (w.samples().size() < 2) continue;
            int first = w.samples().get(0);
            int last  = w.samples().get(w.samples().size() - 1);
            if (last < first) decliningCount++;
        }
        if (totalWindows == 0) {
            // No buffed/bonded predators ever appeared — soft-observed per D-04 note
            return;
        }
        assertThat(decliningCount / (double) totalWindows)
                .as("D-04 #5 fleeing signal: majority of windows show weaker-type density decline")
                .isGreaterThan(0.50);
    }

    // ── D-11 #1 Tick drift ────────────────────────────────────
    private static void assertTickDriftUnder10Percent(TickEngine engine, int targetTicks,

        long elapsedMs = System.currentTimeMillis() - runStartWallMs;

                .isLessThan(0.10);
    }
